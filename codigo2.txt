menu_criar_avatar = function(_imagem) {

    if (_imagem <= 0) return false;

    // Steamworks 3.0.0:
    // Retorna struct com width e height.
    var _dimensoes = steam_utils_get_image_size(_imagem);

    if (!is_struct(_dimensoes)) return false;

    if (!variable_struct_exists(_dimensoes, "width")) {
        return false;
    }

    if (!variable_struct_exists(_dimensoes, "height")) {
        return false;
    }

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;

    if (_largura <= 0 || _altura <= 0) return false;

    var _tamanho_buffer = _largura * _altura * 4;

    var _buffer = buffer_create(
        _tamanho_buffer,
        buffer_fixed,
        1
    );

    // Steamworks 3.0.0:
    // Esta função recebe somente 2 argumentos.
    var _sucesso = steam_utils_get_image_rgba(
        _imagem,
        _buffer
    );

    if (!_sucesso) {

        buffer_delete(_buffer);
        return false;
    }

    var _surface = surface_create(_largura, _altura);

    if (!surface_exists(_surface)) {

        buffer_delete(_buffer);
        return false;
    }

    // Copiar pixels da Steam para a surface.
    buffer_set_surface(
        _buffer,
        _surface,
        0
    );

    // Criar sprite dinâmico.
    var _novo_sprite = sprite_create_from_surface(
        _surface,
        0,
        0,
        _largura,
        _altura,
        false,
        false,
        0,
        0
    );

    surface_free(_surface);
    buffer_delete(_buffer);

    if (_novo_sprite == -1) return false;

    // Liberar sprite antigo, se existir.
    if (menu_avatar_sprite != -1) {

        if (sprite_exists(menu_avatar_sprite)) {
            sprite_delete(menu_avatar_sprite);
        }
    }

    menu_avatar_sprite = _novo_sprite;
    menu_avatar_carregado = true;

    show_debug_message(
        "Avatar Steam carregado com sucesso!"
    );

    return true;
};
