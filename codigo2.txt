menu_criar_avatar = function(_imagem) {
    if (_imagem <= 0) return false;

    var _dimensoes = steam_utils_get_image_size(_imagem);
    if (!is_struct(_dimensoes)) return false;
    if (!variable_struct_exists(_dimensoes, "width")) return false;
    if (!variable_struct_exists(_dimensoes, "height")) return false;

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;
    if (_largura <= 0 || _altura <= 0) return false;

    var _buffer = buffer_create(_largura * _altura * 4, buffer_fixed, 1);
    var _sucesso = steam_utils_get_image_rgba(_imagem, _buffer);
    if (!_sucesso) {
        buffer_delete(_buffer);
        return false;
    }

    var _surface = surface_create(_largura, _altura);
    if (!surface_exists(_surface)) {
        buffer_delete(_buffer);
        return false;
    }

    buffer_set_surface(_buffer, _surface, 0);
    var _novo_sprite = sprite_create_from_surface(
        _surface, 0, 0, _largura, _altura,
        false, false, 0, 0
    );
    surface_free(_surface);
    buffer_delete(_buffer);

    if (_novo_sprite == -1) return false;
    if (menu_avatar_sprite != -1 && sprite_exists(menu_avatar_sprite)) {
        sprite_delete(menu_avatar_sprite);
    }

    menu_avatar_sprite = _novo_sprite;
    menu_avatar_carregado = true;
    show_debug_message("Avatar Steam carregado com sucesso!");
    return true;
};

net_register_room = function(_room) {

    net_room = _room;

    net_host_session_id = "";

    net_game_started = false;

    net_board_data = [];
    net_board_data_received = false;
    net_board_data_version = 0;

    colyseus_on_join(_room, function(_joined_room) {

        net_connected = true;
        net_busy = false;
        net_status = "Conectado!";
        net_error = "";

        net_room_id = colyseus_room_get_id(_joined_room);
        net_session_id = colyseus_room_get_session_id(_joined_room);

        net_players = [];

        net_callbacks = colyseus_callbacks_create(_joined_room);

        var _state = colyseus_room_get_state(_joined_room);

        if (_state != undefined) {

            var _host_id = colyseus_schema_get(
                _state,
                "hostSessionId"
            );

            if (is_undefined(_host_id)) {
                net_host_session_id = "";
            }
            else {
                net_host_session_id = _host_id;
            }

            net_game_status = colyseus_schema_get(
                _state,
                "status"
            );

            net_game_started = (net_game_status == "playing");

            var _countdown = colyseus_schema_get(
                _state,
                "autoStartCountdown"
            );

            if (is_undefined(_countdown)) {
                net_auto_start_countdown = 0;
            }
            else {
                net_auto_start_countdown = _countdown;
            }
        }

        colyseus_listen(
            net_callbacks,
            "hostSessionId",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_host_session_id = "";
                }
                else {
                    net_host_session_id = _value;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "status",
            function(_value, _previous) {

                net_game_status = _value;

                if (_value == "playing") {
                    net_game_started = true;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "autoStartCountdown",
            function(_value, _previous) {

                if (is_undefined(_value)) {

        const sessionIds = [...this.state.players.keys()];

        if (sessionIds.length === 0) {
            return "";
        }

        const randomIndex = Math.floor(
            Math.random() * sessionIds.length
        );

        return sessionIds[randomIndex];
    }

    // --------------------------------------------------
    // ATUALIZAR JOGADOR SORTEADO
    // --------------------------------------------------

    private updateFirstPlayerAnnouncement(sessionId: string) {
        const player = this.state.players.get(sessionId);

        if (!player) {
            this.state.firstPlayerSessionId = "";
            this.state.firstPlayerName = "";
            return;
        }

        this.state.firstPlayerSessionId = sessionId;
        this.state.firstPlayerName = player.name;

        console.log("==================================");
        console.log("PRIMEIRO JOGADOR SORTEADO!");
        console.log("JOGADOR:", player.name);
        console.log("SESSION ID:", sessionId);
        console.log("==================================");
    }

    // --------------------------------------------------
    // EXIBIR ANÚNCIO DO PRIMEIRO JOGADOR
    // --------------------------------------------------

    private showFirstPlayerAnnouncement() {
        if (this.state.status !== "playing") {
            return;
        }

        const firstPlayerSessionId = this.chooseRandomFirstPlayer();

        if (firstPlayerSessionId === "") {
            return;
        }

        this.updateFirstPlayerAnnouncement(firstPlayerSessionId);

        this.state.turnDeadline = 0;
        this.state.showFirstPlayerMessage = true;

        console.log(
            "Exibindo anúncio do primeiro jogador por 3 segundos."
        );

        this.firstPlayerMessageDelay?.clear();

        this.firstPlayerMessageDelay = this.clock.setTimeout(() => {
            this.firstPlayerMessageDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            let selectedSessionId = this.state.firstPlayerSessionId;

            if (!this.state.players.has(selectedSessionId)) {
                selectedSessionId = this.chooseRandomFirstPlayer();

                if (selectedSessionId === "") {
                    this.state.showFirstPlayerMessage = false;
                    this.state.firstPlayerSessionId = "";
                    this.state.firstPlayerName = "";
                    return;
                }

                this.updateFirstPlayerAnnouncement(selectedSessionId);
            }

            this.state.showFirstPlayerMessage = false;

            console.log(
                "Anúncio encerrado! Liberando o primeiro turno."
            );

            this.nextTurn(selectedSessionId);
        }, FIRST_PLAYER_MESSAGE_DURATION);
    }

    // --------------------------------------------------
    // INICIAR PRAZO DA ROLAGEM EXTRA
    // --------------------------------------------------

    private startExtraRollTimer(sessionId: string) {
        if (this.state.status !== "playing") {
            return;
        }

        if (this.state.currentTurn !== sessionId) {
            return;
        }

        const player = this.state.players.get(sessionId);

        if (!player || !player.extraRollAvailable) {
            return;
        }

        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.state.currentTurn !== sessionId) {
                return;
            }

            const currentPlayer = this.state.players.get(sessionId);

            if (
                currentPlayer?.extraRollAvailable &&
                !this.state.hasRolled
            ) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a rolagem extra.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);
    }

    // --------------------------------------------------
    // GERAR CÓDIGO DE 6 CARACTERES
    // --------------------------------------------------

    private generateCode(): string {
        let code = "";

        for (let i = 0; i < 6; i++) {
            const index = Math.floor(
                Math.random() * this.CODE_CHARACTERS.length
            );

            code += this.CODE_CHARACTERS[index];
        }

        return code;
    }

    // --------------------------------------------------
    // GARANTIR CÓDIGO ÚNICO
    // --------------------------------------------------

    private async generateRoomCode(): Promise<string> {
        const existingCodes = await this.presence.smembers(
            this.CODE_CHANNEL
        );

        let code = "";

        do {
            code = this.generateCode();
        } while (existingCodes.includes(code));

        await this.presence.sadd(this.CODE_CHANNEL, code);

        return code;
    }

    // --------------------------------------------------
    // CANCELAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private clearAutoStartCountdown() {
        this.autoStartInterval?.clear();
        this.autoStartInterval = undefined;

        this.state.autoStartCountdown = 0;
    }

    // --------------------------------------------------
    // REINICIAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private resetAutoStartCountdown() {
        if (this.mode !== "public") {
            this.clearAutoStartCountdown();
            return;
        }

        if (this.state.status !== "waiting") {
            this.clearAutoStartCountdown();
            return;
        }

        this.clearAutoStartCountdown();

        if (this.state.players.size < 2) {
            return;
        }

        this.state.autoStartCountdown = AUTO_START_DURATION;

        console.log(
            `Contagem automática iniciada: ${this.roomId} (${AUTO_START_DURATION}s)`
        );

        this.autoStartInterval = this.clock.setInterval(() => {
            if (this.state.status !== "waiting") {
                this.clearAutoStartCountdown();
                return;
            }

            if (this.state.players.size < 2) {
                this.clearAutoStartCountdown();
                return;
            }

            this.state.autoStartCountdown--;

            console.log(
                `Sala ${this.roomId}: ${this.state.autoStartCountdown}s`
            );

            if (this.state.autoStartCountdown <= 0) {
                this.clearAutoStartCountdown();
                this.startGame();
            }
        }, 1000);
    }

    // --------------------------------------------------
    // INICIAR PARTIDA
    // --------------------------------------------------

    private startGame() {
        if (this.state.status !== "waiting") {
            return;
        }

        if (this.state.players.size < 2) {
            return;
        }

        this.clearAutoStartCountdown();

        this.randomizeSpecialCities();
        this.broadcastBoard();

        this.state.status = "playing";

        this.state.currentTurn = "";
        this.state.turnDeadline = 0;
        this.state.turnCount = 0;

        this.state.hasRolled = false;
        this.state.hasMoved = false;

        this.state.dice1 = 0;
        this.state.dice2 = 0;
        this.state.lastRollBy = "";

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        this.playersResetForNewGame();

        void this.setMetadata({
            mode: this.mode,
            status: this.state.status
        });

        this.lock();

        console.log("Partida iniciada:", this.roomId);
        console.log("Aguardando 3 segundos para iniciar o anúncio...");

        this.startDelay?.clear();

        this.startDelay = this.clock.setTimeout(() => {
            this.startDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            console.log("Animação concluída! Sorteando primeiro jogador.");

            this.showFirstPlayerAnnouncement();
        }, GAME_START_DELAY);
    }

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (player.debtsAvailable <= 0) {
                client.send(
                    "action_error",
                    "Você não possui dívidas disponíveis."
                );
                return;
            }

            player.debtsAvailable--;

            console.log("==================================");
            console.log("JOGADOR USOU UMA DÍVIDA PARA SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log(
                "DÍVIDAS RESTANTES:",
                player.debtsAvailable
            );
            console.log("==================================");

            this.releasePlayerFromPrison(client.sessionId);
        },

        // MOVIMENTAR JOGADOR
        // --------------------------------------------------

        move: (client: Client) => {

            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(
                client.sessionId
            );

            if (!player) {
                return;
            }

            if (player.inPrison) {

                client.send(
                    "action_error",
                    "Você precisa sair da prisão antes de movimentar."
                );

                return;
            }

            if (
                !this.state.hasRolled ||
                this.state.lastRollBy !== client.sessionId
            ) {

                client.send(
                    "action_error",
                    "Você precisa rolar os dados primeiro."
                );

                return;
            }

            if (this.state.hasMoved) {

                client.send(
                    "action_error",
                    "Você já movimentou nesta rodada."
                );

                return;
            }

            const dice1 = this.state.dice1;
            const dice2 = this.state.dice2;

            const rolledDoubles = dice1 === dice2;

            const steps = dice1 + dice2;

            const startingPosition = player.position;
            const newPosition = startingPosition + steps;

            const completedLaps = Math.floor(
                newPosition / BOARD_SIZE
            );

            const landedPosition = newPosition % BOARD_SIZE;

            player.position = landedPosition;
            player.laps += completedLaps;

            const landedCell = this.board[landedPosition];

            player.lastCellType = landedCell.type;
            player.lastCellName = landedCell.name;

            this.state.hasMoved = true;

            console.log(
                `${player.name} avançou ${steps} casas. Posição: ${player.position}`
            );

            console.log(
                `${player.name} caiu na casa ${player.position}: ${landedCell.name} (${landedCell.type})`
            );

            console.log(
                `Dados: ${dice1} + ${dice2} | Duplas: ${rolledDoubles}`
            );

            // --------------------------------------------------
            // RECOMPENSA POR COMPLETAR VOLTAS
            // --------------------------------------------------

            if (completedLaps > 0) {

                console.log(
                    `${player.name} completou ${completedLaps} volta(s). Total: ${player.laps}`
                );

                const totalMoneyReward =
                    START_CELL_MONEY_REWARD * completedLaps;

                player.money += totalMoneyReward;

                console.log(
                    `${player.name} recebeu $${totalMoneyReward.toLocaleString("pt-BR")} por completar ${completedLaps} volta(s).`
                );
            }

            // --------------------------------------------------
            // VERIFICAR SE CAIU NA PRISÃO
            // --------------------------------------------------

            const landedInPrison =
                landedCell.name === "Prisão" &&
                landedCell.type === "island";

            if (landedInPrison) {

                // A prisão será ativada após a chegada.
                // Enquanto caminha, manter a exibição
                // do resultado dos dados.

                player.prisonAttempts = 0;
                player.extraRollAvailable = false;

                this.state.hasMoved = true;

                // Não zerar hasRolled nem lastRollBy aqui.

                console.log(
                    `${player.name} foi enviado para a prisão. O turno será encerrado após o anúncio.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "prison"
                );

                return;
            }

            // --------------------------------------------------
            // CASA INICIAL
            // --------------------------------------------------

            const landedExactlyOnStart =
                landedPosition === 0 &&
                landedCell.type === "start" &&
                landedCell.name === "Início";

            const receivesStartExtraRoll =
                landedExactlyOnStart &&
                completedLaps > 0 &&
                player.laps >= 1;

            let extraRollReason: ExtraRollReason | undefined;

            if (receivesStartExtraRoll) {

                extraRollReason = "start";

            } else if (rolledDoubles) {

                extraRollReason = "doubles";
            }

            const canReceiveExtraRoll =
                extraRollReason !== undefined;

            console.log(
                `[VERIFICAÇÃO ROLAGEM EXTRA] ${player.name} | ` +
                `posição inicial: ${startingPosition} | ` +
                `dados: ${dice1} + ${dice2} | ` +
                `duplas: ${rolledDoubles} | ` +
                `posição final: ${landedPosition} | ` +
                `voltas: ${player.laps} | ` +
                `terminou no início: ${receivesStartExtraRoll} | ` +
                `motivo: ${extraRollReason ?? "nenhum"} | ` +
                `recebe extra: ${canReceiveExtraRoll}`
            );

            // --------------------------------------------------
            // ROLAGEM EXTRA
            // --------------------------------------------------

            if (canReceiveExtraRoll && extraRollReason) {

                player.extraRollAvailable = true;

                this.state.hasMoved = true;

                // Preservar hasRolled e lastRollBy
                // até a chegada do personagem.
                //
                // Isso mantém o resultado visível
                // para todos os clientes.

                console.log(
                    `${player.name} recebeu uma rolagem extra pelo motivo: ${extraRollReason}.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "extra",
                    extraRollReason
                );

            } else {

                // --------------------------------------------------
                // MOVIMENTO NORMAL
                // --------------------------------------------------

                // Mesmo sem popup, aguardar a animação
                // antes de liberar as ações da HUD.

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "none"
                );
            }
        },


        // --------------------------------------------------
        // ENCERRAR TURNO
        // --------------------------------------------------

        end_turn: (client: Client) => {

            const player = this.state.players.get(client.sessionId);

            console.log("==================================");
            console.log("[END_TURN] SOLICITACAO RECEBIDA");
            console.log("[END_TURN] Cliente:", client.sessionId);
            console.log("[END_TURN] Jogador:", player?.name ?? "Nao encontrado");
            console.log("[END_TURN] Status:", this.state.status);
            console.log("[END_TURN] Turno atual:", this.state.currentTurn);
            console.log("[END_TURN] HasRolled:", this.state.hasRolled);

            console.log("[END_TURN] Preso:", player?.inPrison);
            console.log("[END_TURN] Rolagem extra:", player?.extraRollAvailable);
            console.log("[END_TURN] Jogadores na sala:", this.state.players.size);

            // --------------------------------------------------
            // VERIFICAR PARTIDA
            // --------------------------------------------------

            if (this.state.status !== "playing") {

                console.log("[END_TURN] BLOQUEADO: partida nao esta em andamento.");

                client.send(
                    "action_error",
                    "A partida nao esta em andamento."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR POPUPS
            // --------------------------------------------------

            if (this.isPopupActive()) {

                console.log("[END_TURN] BLOQUEADO: popup ativo no servidor.");

                client.send(
                    "action_error",

                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR JOGADOR
            // --------------------------------------------------

            if (!player) {

                console.log("[END_TURN] BLOQUEADO: jogador nao encontrado.");

                client.send(
                    "action_error",
                    "Jogador nao encontrado."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR TURNO
            // --------------------------------------------------

            if (this.state.currentTurn !== client.sessionId) {

                    "Nao e seu turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR MOVIMENTO
            // --------------------------------------------------

            if (!this.state.hasMoved) {

                console.log("[END_TURN] BLOQUEADO: movimento nao concluido.");

                client.send(
                    "action_error",
                    "Voce precisa movimentar antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR PRISAO
            // --------------------------------------------------

            if (player.inPrison) {

                console.log("[END_TURN] BLOQUEADO: jogador esta preso.");

                client.send(
                    "action_error",
                    "Voce ainda esta na prisao."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR ROLAGEM EXTRA
            // --------------------------------------------------

            if (player.extraRollAvailable) {

                console.log("[END_TURN] BLOQUEADO: rolagem extra pendente.");

                client.send(
                    "action_error",
                    "Utilize sua rolagem extra antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // ENCERRAR TURNO
            // --------------------------------------------------

            console.log("[END_TURN] AUTORIZADO!");
            console.log("[END_TURN] Encerrando turno de:", player.name);

            const previousTurn = this.state.currentTurn;

            this.nextTurn();

            console.log("[END_TURN] Turno anterior:", previousTurn);
            console.log("[END_TURN] Novo turno:", this.state.currentTurn);
            console.log("==================================");
        },


        play: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                return;
            }

            player.score++;
            this.nextTurn();
        },

    onAuth(client: Client, options: any) {

        if (this.state.status !== "waiting") {
            throw new ServerError(409, "Esta partida já começou.");
        }

        if (options?.mode !== this.mode) {
            throw new ServerError(
                403,
                "Esta sala não pertence a essa modalidade de partida."
            );
        }

        // --------------------------------------------------
        // VALIDAR NOME STEAM
        // --------------------------------------------------

        if (
            typeof options?.playerName !== "string"
        ) {
            throw new ServerError(400, "Nome Steam inválido.");
        }

        const playerName = String(options.playerName).trim();

        if (playerName.length === 0 || playerName.length > 32) {
            throw new ServerError(
                400,
                "O nome Steam deve possuir entre 1 e 32 caracteres."
            );
        }

        // --------------------------------------------------
        // VALIDAR STEAM ID
        // --------------------------------------------------

        const steamId = String(options?.steamId ?? "").trim();

        if (!/^\d{17}$/.test(steamId)) {
            throw new ServerError(
                400,
                "Steam ID inválido ou não recebido."
            );
        }

        // --------------------------------------------------
        // VALIDAR PERSONAGEM
        // --------------------------------------------------

        const selectedCharacter = Number(
            options?.selectedCharacter ?? 0
        );

        const validCharacter =
            Number.isInteger(selectedCharacter) &&
            selectedCharacter >= 0 &&
            selectedCharacter <= 3;

        if (!validCharacter) {
            throw new ServerError(400, "Personagem inválido.");
        }

        // --------------------------------------------------
        // VALIDAR COR
        // --------------------------------------------------

        const preferredColor =
            typeof options?.selectedColor === "string" &&
            this.isValidColor(options.selectedColor)
                ? options.selectedColor
                : "red";

        return {
            playerName,
            steamId,
            preferredColor,
            selectedCharacter
        };
    }

    onLeave(client: Client, code: CloseCode) {
        console.log(client.sessionId, "saiu da sala!", code);

            this.state.currentTurn === client.sessionId;

            this.state.extraRollPlayerSessionId === client.sessionId;

            this.state.prisonPlayerSessionId === client.sessionId;

        this.state.players.delete(client.sessionId);

        this.playerJoinOrder = this.playerJoinOrder.filter(
            sessionId => sessionId !== client.sessionId
        );

        if (this.pendingMovementAnnouncement) {
            if (
                this.pendingMovementAnnouncement.sessionId ===
                client.sessionId
            ) {
                this.pendingMovementAnnouncement = undefined;
                this.state.movementAnnouncementPending = false;
            }
            else {
                this.pendingMovementAnnouncement.arrivedClients.delete(
                    client.sessionId
                );

                this.finishMovementAnnouncementIfReady(
                    client.sessionId
                );
            }
        }

        if (
            this.state.showFirstPlayerMessage &&
            wasSelectedFirstPlayer
        ) {
            const newFirstPlayer = this.chooseRandomFirstPlayer();

            if (newFirstPlayer !== "") {
                this.updateFirstPlayerAnnouncement(newFirstPlayer);
            } else {
                this.state.firstPlayerSessionId = "";
                this.state.firstPlayerName = "";
            }
        }

        if (
            this.state.showExtraRollMessage &&
            wasExtraRollPlayer
        ) {
            this.extraRollMessageDelay?.clear();
            this.extraRollMessageDelay = undefined;

            this.state.showExtraRollMessage = false;
            this.state.extraRollPlayerName = "";
            this.state.extraRollPlayerSessionId = "";
            this.state.extraRollReason = "";

            if (this.state.status === "playing" && wasTheirTurn) {
                this.nextTurn();
            }
        }

        if (
            this.state.showPrisonMessage &&
            wasPrisonMessagePlayer
        ) {
            this.prisonMessageDelay?.clear();
            this.prisonMessageDelay = undefined;

            this.state.showPrisonMessage = false;
            this.state.prisonPlayerName = "";
            this.state.prisonPlayerSessionId = "";
        }

        if (
            this.state.status === "waiting" &&
            wasHost
        ) {
            const nextHost = this.playerJoinOrder.find(
                sessionId => this.state.players.has(sessionId)
            );

            this.state.hostSessionId = nextHost ?? "";

            if (nextHost) {
                const nextHostPlayer = this.state.players.get(nextHost);

            wasTheirTurn &&
            !(
                this.state.showExtraRollMessage &&
                wasExtraRollPlayer
            )
        ) {
            this.nextTurn();
        }
    }
