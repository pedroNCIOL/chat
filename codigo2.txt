menu_amigos_gerar_codigo = function(_steam_id) {

    var _identidade = steam_user_decode_steam_id(_steam_id);

    if (!is_struct(_identidade)) return "------";

    if (!variable_struct_exists(_identidade, "account_id")) {
        return "------";
    }

    if (variable_struct_exists(_identidade, "is_valid")) {
        if (!_identidade.is_valid) return "------";
    }

    var _numero = floor(_identidade.account_id);

    if (_numero < 0 || _numero > 4294967295) {
        return "------";
    }

    var _alfabeto =
        "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

    var _codigo = "";

    repeat (6) {
        var _digito = _numero mod 62;

        _codigo =
            string_char_at(_alfabeto, _digito + 1) + _codigo;

        _numero = floor(_numero / 62);
    }

    return _codigo;
};

// --------------------------------------------------
// CONVERTER IMAGEM STEAM EM SPRITE
// --------------------------------------------------

menu_amigos_criar_sprite = function(_imagem) {

    if (_imagem <= 0) return -1;

    var _dimensoes = steam_utils_get_image_size(_imagem);

    if (!is_struct(_dimensoes)) return -1;

    if (
        !variable_struct_exists(_dimensoes, "width") ||
        !variable_struct_exists(_dimensoes, "height")
    ) return -1;

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;

    if (_largura <= 0 || _altura <= 0) return -1;

    var _buffer = buffer_create(
        _largura * _altura * 4,
        buffer_fixed,
        1
    );

    var _ok = steam_utils_get_image_rgba(
        _imagem,
        _buffer
    );

    if (!_ok) {
        buffer_delete(_buffer);
        return -1;
    }

    var _surf = surface_create(_largura, _altura);

    if (!surface_exists(_surf)) {
        buffer_delete(_buffer);
        return -1;
    }

    buffer_set_surface(_buffer, _surf, 0);

    var _sprite = sprite_create_from_surface(
        _surf, 0, 0, _largura, _altura,
        false, false, 0, 0
    );

    surface_free(_surf);
    buffer_delete(_buffer);

    return _sprite;
};

menu_amigos_atualizar = function() {

    menu_amigos_steam_disponivel = steam_api_is_initialized();

    if (!menu_amigos_steam_disponivel) return;

    // App ID do Money Town em execucao.
    var _meu_app_id = steam_utils_get_app_id();

    var _flags = SteamFriendsFriendFlag.Immediate;
    var _total = steam_friends_get_friend_count(_flags);

    if (_total < 0) return;

    var _anteriores = menu_amigos;
    var _atuais = [];

    for (var _i = 0; _i < _total; _i++) {

        var _steam_id = steam_friends_get_friend_by_index(
            _i,
            _flags
        );

        var _nome =
            steam_friends_get_friend_persona_name(_steam_id);

        // ----------------------------------------------
        // IDENTIFICAR STATUS
        //
        // 0 = Jogando Money Town
        // 1 = Online (inclusive jogando outros jogos)
        // 2 = Offline
        // ----------------------------------------------

        var _estado =
            steam_friends_get_friend_persona_state(_steam_id);

        var _grupo = 2;

        if (_estado != SteamFriendsPersonaState.Offline) {

            // Esta online na Steam.
            _grupo = 1;

            var _jogo =
                steam_friends_get_friend_game_played(_steam_id);

            // So entra em "Jogando" quando o jogo
            // do amigo for exatamente Money Town.
            if (
                _meu_app_id > 0 &&
                is_struct(_jogo) &&
                variable_struct_exists(_jogo, "game_id")
            ) {

                if (_jogo.game_id == _meu_app_id) {
                    _grupo = 0;
                }
            }
        }

        // ----------------------------------------------
        // REAPROVEITAR AVATAR
        // ----------------------------------------------

        var _sprite = -1;

        for (var _j = 0; _j < array_length(_anteriores); _j++) {

            if (_anteriores[_j].steam_id == _steam_id) {
                _sprite = _anteriores[_j].sprite;
                break;
            }
        }

        if (_sprite == -1) {

            var _imagem =
                steam_friends_get_large_friend_avatar(_steam_id);

            if (_imagem > 0) {
                _sprite = menu_amigos_criar_sprite(_imagem);
            }
        }

        array_push(_atuais, {
            steam_id: _steam_id,
            nome: _nome,
            sprite: _sprite,
            grupo: _grupo,
            origem: "steam"
        });
    }

    // ----------------------------------------------
    // LIBERAR AVATARES REMOVIDOS
    // ----------------------------------------------

    for (var _j = 0; _j < array_length(_anteriores); _j++) {

        var _antigo = _anteriores[_j];
        var _existe = false;

        for (var _k = 0; _k < array_length(_atuais); _k++) {

            if (_atuais[_k].steam_id == _antigo.steam_id) {
                _existe = true;
                break;
            }
        }

        if (!_existe && _antigo.sprite != -1) {

            if (sprite_exists(_antigo.sprite)) {
                sprite_delete(_antigo.sprite);
            }
        }
    }

    menu_amigos = _atuais;
};

// FILTRAR E MONTAR LISTA COM CATEGORIAS
// --------------------------------------------------

menu_amigos_filtrar = function() {

    menu_amigos_filtrados = [];
    menu_amigos_exibicao = [];

    var _busca = string_lower(
        string_trim(menu_amigos_busca)
    );

    // Juntar Steam e amigos internos do jogo.
    var _todos = [];

    for (var _i = 0; _i < array_length(menu_amigos); _i++) {
        array_push(_todos, menu_amigos[_i]);
    }

    for (var _i = 0; _i < array_length(menu_amigos_jogo); _i++) {

        var _interno = menu_amigos_jogo[_i];
        var _duplicado = false;

        // Nao mostrar duas vezes o mesmo SteamID.
        if (_interno.steam_id != 0) {

            for (var _j = 0; _j < array_length(_todos); _j++) {

                if (_todos[_j].steam_id == _interno.steam_id) {
                    _duplicado = true;
                    break;
                }
            }
        }

        if (!_duplicado) {
            array_push(_todos, _interno);
        }
    }

    for (var _i = 0; _i < array_length(_todos); _i++) {

        var _amigo = _todos[_i];

        if (
            _busca == "" ||
            string_pos(_busca, string_lower(_amigo.nome)) > 0
        ) {

            array_push(menu_amigos_filtrados, _amigo);
        }
    }

    // ----------------------------------------------
    // MONTAR CATEGORIAS NA ORDEM CORRETA
    // ----------------------------------------------

    var _nomes_grupos = [
        "Jogando",
        "Online",
        "Offline"
    ];

    var _pos_y = 0;

    for (var _grupo = 0; _grupo < 3; _grupo++) {

        array_push(menu_amigos_exibicao, {
            tipo: "titulo",
            texto: _nomes_grupos[_grupo],
            pos_y: _pos_y,
            altura: menu_amigos_titulo_altura
        });

        _pos_y += menu_amigos_titulo_altura;

        for (
            var _i = 0;
            _i < array_length(menu_amigos_filtrados);
            _i++
        ) {

            var _amigo = menu_amigos_filtrados[_i];

            if (_amigo.grupo == _grupo) {

                array_push(menu_amigos_exibicao, {
                    tipo: "amigo",
                    amigo: _amigo,
                    pos_y: _pos_y,
                    altura: menu_amigos_item_altura
                });

                _pos_y +=
                    menu_amigos_item_altura +
                    menu_amigos_item_espaco;
            }
        }

        _pos_y += 5;
    }

    menu_amigos_conteudo_altura = _pos_y;
};

net_register_room = function(_room) {

    net_room = _room;
    net_busy = true;
    net_connected = false;

    net_host_session_id = "";
    net_game_status = "waiting";
    net_game_started = false;
    net_auto_start_countdown = 0;

    // Reiniciar dados do tabuleiro ao entrar em outra sala.
    net_board_data = [];
    net_board_data_received = false;
    net_board_data_version = 0;

    colyseus_on_join(_room, function(_joined_room) {

        net_connected = true;
        net_busy = false;
        net_status = "Conectado!";
        net_error = "";

        net_room_id = colyseus_room_get_id(_joined_room);
        net_session_id = colyseus_room_get_session_id(_joined_room);

        net_players = [];

        net_callbacks = colyseus_callbacks_create(_joined_room);

        var _state = colyseus_room_get_state(_joined_room);

        if (_state != undefined) {

            var _host_id = colyseus_schema_get(
                _state,
                "hostSessionId"
            );

            if (is_undefined(_host_id)) {
                net_host_session_id = "";
            }
            else {
                net_host_session_id = _host_id;
            }

            net_game_status = colyseus_schema_get(
                _state,
                "status"
            );

            net_game_started = (net_game_status == "playing");

            var _countdown = colyseus_schema_get(
                _state,
                "autoStartCountdown"
            );

            if (is_undefined(_countdown)) {
                net_auto_start_countdown = 0;
            }
            else {
                net_auto_start_countdown = _countdown;
            }
        }

        colyseus_listen(
            net_callbacks,
            "hostSessionId",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_host_session_id = "";
                }
                else {
                    net_host_session_id = _value;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "status",
            function(_value, _previous) {

                net_game_status = _value;

                if (_value == "playing") {
                    net_game_started = true;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "autoStartCountdown",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_auto_start_countdown = 0;
                }
                else {
                    net_auto_start_countdown = _value;
                }
            }
        );

        // Observar jogadores adicionados.
        colyseus_on_add(
            net_callbacks,
            "players",
            function(_instance, _key) {

                var _player_name = colyseus_schema_get(
                    _instance,
                    "name"
                );

                var _player_color = colyseus_schema_get(
                    _instance,
                    "color"
                );

                if (is_undefined(_player_color) || _player_color == "") {
                    _player_color = "red";
                }

                var _player_data = {
                    session_id: _key,
                    name: _player_name,
                    color: _player_color,
                    schema: _instance
                };

                net_players[array_length(net_players)] = _player_data;

                if (_key == net_session_id) {
                    net_player_color = _player_color;
                }

                show_debug_message(
                    "PLAYER ADICIONADO | ID: " + string(_key) +
                    " | NOME: " + string(_player_name) +
                    " | COR: " + string(_player_color)
                );

                show_debug_message(
                    "TOTAL APOS ADICAO: " +
                    string(array_length(net_players))
                );
            }
        );

        // Observar jogadores removidos.
        colyseus_on_remove(
            net_callbacks,
            "players",
            function(_instance, _key) {

                show_debug_message("=== ANTES DA REMOCAO ===");

                show_debug_message(
                    "ID REMOVIDO: " + string(_key)
                );

                show_debug_message(
                    "TOTAL ANTES: " +
                    string(array_length(net_players))
                );

                for (var i = 0; i < array_length(net_players); i++) {

                    show_debug_message(
                        "JOGADOR " + string(i) +
                        " | ID: " + string(net_players[i].session_id) +
                        " | NOME: " + string(net_players[i].name)
                    );
                }

                var _remaining_players = [];

                for (var j = 0; j < array_length(net_players); j++) {

                    if (net_players[j].session_id != _key) {

                        _remaining_players[
                            array_length(_remaining_players)
                        ] = net_players[j];
                    }
                }

                net_players = _remaining_players;

                show_debug_message(
                    "PLAYER REMOVIDO | ID: " + string(_key)
                );

                show_debug_message("=== JOGADORES RESTANTES ===");

                for (var k = 0; k < array_length(net_players); k++) {

                    show_debug_message(
                        string(net_players[k].session_id) + " | " +
                        string(net_players[k].name)
                    );
                }

                show_debug_message(
                    "TOTAL: " + string(array_length(net_players))
                );

                show_debug_message("===========================");
            }
        );

        // Receber mensagens do servidor.
        colyseus_on_message(
            _joined_room,
            function(_room_message, _type, _data) {

                // ------------------------------------------
                // ERRO DE COR
                // ------------------------------------------

                if (_type == "color_error") {

                    net_error = string(_data);
                }

                // ------------------------------------------
                // DADOS DO TABULEIRO
                // ------------------------------------------

                else if (_type == "board_data") {

                    var _parsed_data = undefined;

                    // O servidor envia os dados como JSON string.
                    if (is_string(_data)) {

                        _parsed_data = json_parse(_data);
                    }
                    else {

                        _parsed_data = _data;
                    }

                    if (is_array(_parsed_data)) {

                        net_board_data = _parsed_data;
                        net_board_data_received = true;

                        // Registrar nova versão do tabuleiro.
                        net_board_data_version++;

                        show_debug_message(
                            "DADOS DO TABULEIRO RECEBIDOS | VERSÃO: " +
                            string(net_board_data_version)
                        );

                        show_debug_message(
                            "QUANTIDADE DE CASAS: " +
                            string(array_length(net_board_data))
                        );

                    }
                    else {

                        show_debug_message(
                            "ERRO AO CONVERTER board_data. Tipo recebido: " +
                            typeof(_data)
                        );
                    }
                }
            }
        );

        show_debug_message("=== SALA CONECTADA ===");
        show_debug_message("Sala: " + net_room_id);
        show_debug_message("Sessão: " + net_session_id);
        show_debug_message("Líder: " + string(net_host_session_id));
        show_debug_message("Modalidade: " + net_room_mode);
    });

    colyseus_on_error(_room, function(_code, _message) {

        net_connected = false;
        net_busy = false;
        net_status = "Erro ao entrar na sala";
        net_error = _message;

        show_debug_message("ERRO COLYSEUS [" + string(_code) + "]");
        show_debug_message(_message);
    });
};

                show_debug_message(
                    "TOTAL DE PARTIDAS: " +

// --------------------------------------------------
// CARREGAR AVATAR DE UM PEDIDO
// --------------------------------------------------

popup_carregar_avatar = function(_pedido) {

    // Avatar recebido diretamente.
    if (_pedido.avatar_sprite != -1) {
        return;
    }

    var _menu = instance_find(obj_menu, 0);

    if (_menu == noone) return;

    // ----------------------------------------------
    // REAPROVEITAR AVATAR JA CARREGADO
    // ----------------------------------------------

    for (var _i = 0; _i < array_length(_menu.menu_amigos); _i++) {

        var _amigo = _menu.menu_amigos[_i];

        if (_amigo.steam_id == _pedido.steam_id) {

            if (
                _amigo.sprite != -1 &&
                sprite_exists(_amigo.sprite)
            ) {

                _pedido.avatar_sprite = _amigo.sprite;
                _pedido.avatar_proprio = false;

                return;
            }
        }
    }

    // ----------------------------------------------
    // SOLICITAR AVATAR PELA STEAM
    // ----------------------------------------------

    if (!steam_api_is_initialized()) return;

    if (
        _pedido.steam_id == 0 ||
        _pedido.steam_id == ""
    ) {
        return;
    }

    var _imagem =
        steam_friends_get_large_friend_avatar(
            _pedido.steam_id
        );

    if (_imagem <= 0) return;

    var _novo_sprite =
        _menu.menu_amigos_criar_sprite(_imagem);

    if (_novo_sprite != -1) {

        _pedido.avatar_sprite = _novo_sprite;

        // Este sprite pertence ao popup.
        _pedido.avatar_proprio = true;
    }
};

// --------------------------------------------------
// LIBERAR AVATAR DE UM PEDIDO
// --------------------------------------------------

popup_liberar_avatar = function(_pedido) {

    if (_pedido.avatar_proprio) {

        if (
            _pedido.avatar_sprite != -1 &&
            sprite_exists(_pedido.avatar_sprite)
        ) {

            sprite_delete(_pedido.avatar_sprite);
        }

        _pedido.avatar_sprite = -1;
        _pedido.avatar_proprio = false;
    }
};

// --------------------------------------------------
// RESPONDER SOLICITACAO
// --------------------------------------------------

popup_responder = function(_pedido, _aceitou) {

    if (_pedido.resposta_enviada) return;

    _pedido.resposta_enviada = true;
    _pedido.estado = "saindo";

    // Entrega a resposta ao sistema de amizades,
    // caso o callback esteja conectado.
    if (is_callable(popup_ao_responder)) {

        popup_ao_responder(
            _pedido.pedido_id,
            _aceitou
        );

    } else {

        show_debug_message(
            "[AMIZADE] Pedido: " +
            string(_pedido.pedido_id) +
            " | Resposta: " +
            (_aceitou ? "ACEITAR" : "RECUSAR") +
            " | Sem callback do servidor"
        );
    }
};
