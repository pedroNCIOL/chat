menu_amigos_gerar_codigo = function(_steam_id) {

    var _identidade = steam_user_decode_steam_id(_steam_id);

    if (!is_struct(_identidade)) return "------";

    if (!variable_struct_exists(_identidade, "account_id")) {
        return "------";
    }

    if (variable_struct_exists(_identidade, "is_valid")) {
        if (!_identidade.is_valid) return "------";
    }

    var _numero = floor(_identidade.account_id);

    if (_numero < 0 || _numero > 4294967295) {
        return "------";
    }

    var _alfabeto =
        "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

    var _codigo = "";

    repeat (6) {
        var _digito = _numero mod 62;

        _codigo =
            string_char_at(_alfabeto, _digito + 1) + _codigo;

        _numero = floor(_numero / 62);
    }

    return _codigo;
};

// --------------------------------------------------
// CONVERTER IMAGEM STEAM EM SPRITE
// --------------------------------------------------

menu_amigos_criar_sprite = function(_imagem) {

    if (_imagem <= 0) return -1;

    var _dimensoes = steam_utils_get_image_size(_imagem);

    if (!is_struct(_dimensoes)) return -1;

    if (
        !variable_struct_exists(_dimensoes, "width") ||
        !variable_struct_exists(_dimensoes, "height")
    ) return -1;

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;

    if (_largura <= 0 || _altura <= 0) return -1;

    var _buffer = buffer_create(
        _largura * _altura * 4,
        buffer_fixed,
        1
    );

    var _ok = steam_utils_get_image_rgba(
        _imagem,
        _buffer
    );

    if (!_ok) {
        buffer_delete(_buffer);
        return -1;
    }

    var _surf = surface_create(_largura, _altura);

    if (!surface_exists(_surf)) {
        buffer_delete(_buffer);
        return -1;
    }

    buffer_set_surface(_buffer, _surf, 0);

    var _sprite = sprite_create_from_surface(
        _surf, 0, 0, _largura, _altura,
        false, false, 0, 0
    );

    surface_free(_surf);
    buffer_delete(_buffer);

    return _sprite;
};

menu_amigos_atualizar = function() {

    menu_amigos_steam_disponivel = steam_api_is_initialized();

    if (!menu_amigos_steam_disponivel) return;

    // App ID do Money Town em execucao.
    var _meu_app_id = steam_utils_get_app_id();

    var _flags = SteamFriendsFriendFlag.Immediate;
    var _total = steam_friends_get_friend_count(_flags);

    if (_total < 0) return;

    var _anteriores = menu_amigos;
    var _atuais = [];

    for (var _i = 0; _i < _total; _i++) {

        var _steam_id = steam_friends_get_friend_by_index(
            _i,
            _flags
        );

        var _nome =
            steam_friends_get_friend_persona_name(_steam_id);

        // ----------------------------------------------
        // IDENTIFICAR STATUS
        //
        // 0 = Jogando Money Town
        // 1 = Online (inclusive jogando outros jogos)
        // 2 = Offline
        // ----------------------------------------------

        var _estado =
            steam_friends_get_friend_persona_state(_steam_id);

        var _grupo = 2;

        if (_estado != SteamFriendsPersonaState.Offline) {

            // Esta online na Steam.
            _grupo = 1;

            var _jogo =
                steam_friends_get_friend_game_played(_steam_id);

            // So entra em "Jogando" quando o jogo
            // do amigo for exatamente Money Town.
            if (
                _meu_app_id > 0 &&
                is_struct(_jogo) &&
                variable_struct_exists(_jogo, "game_id")
            ) {

                if (_jogo.game_id == _meu_app_id) {
                    _grupo = 0;
                }
            }
        }

        // ----------------------------------------------
        // REAPROVEITAR AVATAR
        // ----------------------------------------------

        var _sprite = -1;

        for (var _j = 0; _j < array_length(_anteriores); _j++) {

            if (_anteriores[_j].steam_id == _steam_id) {
                _sprite = _anteriores[_j].sprite;
                break;
            }
        }

        if (_sprite == -1) {

            var _imagem =
                steam_friends_get_large_friend_avatar(_steam_id);

            if (_imagem > 0) {
                _sprite = menu_amigos_criar_sprite(_imagem);
            }
        }

        array_push(_atuais, {
            steam_id: _steam_id,
            nome: _nome,
            sprite: _sprite,
            grupo: _grupo,
            origem: "steam"
        });
    }

    // ----------------------------------------------
    // LIBERAR AVATARES REMOVIDOS
    // ----------------------------------------------

    for (var _j = 0; _j < array_length(_anteriores); _j++) {

        var _antigo = _anteriores[_j];
        var _existe = false;

        for (var _k = 0; _k < array_length(_atuais); _k++) {

            if (_atuais[_k].steam_id == _antigo.steam_id) {
                _existe = true;
                break;
            }
        }

        if (!_existe && _antigo.sprite != -1) {

            if (sprite_exists(_antigo.sprite)) {
                sprite_delete(_antigo.sprite);
            }
        }
    }

    menu_amigos = _atuais;
};

// FILTRAR E MONTAR LISTA COM CATEGORIAS
// --------------------------------------------------

menu_amigos_filtrar = function() {

    menu_amigos_filtrados = [];
    menu_amigos_exibicao = [];

    var _busca = string_lower(
        string_trim(menu_amigos_busca)
    );

    // Juntar Steam e amigos internos do jogo.
    var _todos = [];

    for (var _i = 0; _i < array_length(menu_amigos); _i++) {
        array_push(_todos, menu_amigos[_i]);
    }

    for (var _i = 0; _i < array_length(menu_amigos_jogo); _i++) {

        var _interno = menu_amigos_jogo[_i];
        var _duplicado = false;

        // Nao mostrar duas vezes o mesmo SteamID.
        if (_interno.steam_id != 0) {

            for (var _j = 0; _j < array_length(_todos); _j++) {

                if (_todos[_j].steam_id == _interno.steam_id) {
                    _duplicado = true;
                    break;
                }
            }
        }

        if (!_duplicado) {
            array_push(_todos, _interno);
        }
    }

    for (var _i = 0; _i < array_length(_todos); _i++) {

        var _amigo = _todos[_i];

        if (
            _busca == "" ||
            string_pos(_busca, string_lower(_amigo.nome)) > 0
        ) {

            array_push(menu_amigos_filtrados, _amigo);
        }
    }

    // ----------------------------------------------
    // MONTAR CATEGORIAS NA ORDEM CORRETA
    // ----------------------------------------------

    var _nomes_grupos = [
        "Jogando",
        "Online",
        "Offline"
    ];

    var _pos_y = 0;

    for (var _grupo = 0; _grupo < 3; _grupo++) {

        array_push(menu_amigos_exibicao, {
            tipo: "titulo",
            texto: _nomes_grupos[_grupo],
            pos_y: _pos_y,
            altura: menu_amigos_titulo_altura
        });

        _pos_y += menu_amigos_titulo_altura;

        for (
            var _i = 0;
            _i < array_length(menu_amigos_filtrados);
            _i++
        ) {

            var _amigo = menu_amigos_filtrados[_i];

            if (_amigo.grupo == _grupo) {

                array_push(menu_amigos_exibicao, {
                    tipo: "amigo",
                    amigo: _amigo,
                    pos_y: _pos_y,
                    altura: menu_amigos_item_altura
                });

                _pos_y +=
                    menu_amigos_item_altura +
                    menu_amigos_item_espaco;
            }
        }

        _pos_y += 5;
    }

    menu_amigos_conteudo_altura = _pos_y;
};

net_register_room = function(_room) {

    net_room = _room;
    net_busy = true;
    net_connected = false;

    net_host_session_id = "";
    net_game_status = "waiting";
    net_game_started = false;
    net_auto_start_countdown = 0;

    // Reiniciar dados do tabuleiro ao entrar em outra sala.
    net_board_data = [];
    net_board_data_received = false;
    net_board_data_version = 0;

    colyseus_on_join(_room, function(_joined_room) {

        net_connected = true;
        net_busy = false;
        net_status = "Conectado!";
        net_error = "";

        net_room_id = colyseus_room_get_id(_joined_room);
        net_session_id = colyseus_room_get_session_id(_joined_room);

        net_players = [];

        net_callbacks = colyseus_callbacks_create(_joined_room);

        var _state = colyseus_room_get_state(_joined_room);

        if (_state != undefined) {

            var _host_id = colyseus_schema_get(
                _state,
                "hostSessionId"
            );

            if (is_undefined(_host_id)) {
                net_host_session_id = "";
            }
            else {
                net_host_session_id = _host_id;
            }

            net_game_status = colyseus_schema_get(
                _state,
                "status"
            );

            net_game_started = (net_game_status == "playing");

            var _countdown = colyseus_schema_get(
                _state,
                "autoStartCountdown"
            );

            if (is_undefined(_countdown)) {
                net_auto_start_countdown = 0;
            }
            else {
                net_auto_start_countdown = _countdown;
            }
        }

        colyseus_listen(
            net_callbacks,
            "hostSessionId",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_host_session_id = "";
                }
                else {
                    net_host_session_id = _value;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "status",
            function(_value, _previous) {

                net_game_status = _value;

                if (_value == "playing") {
                    net_game_started = true;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "autoStartCountdown",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_auto_start_countdown = 0;
                }
                else {
                    net_auto_start_countdown = _value;
                }
            }
        );

        // Observar jogadores adicionados.
        colyseus_on_add(
            net_callbacks,
            "players",
            function(_instance, _key) {

                var _player_name = colyseus_schema_get(
                    _instance,
                    "name"
                );

                var _player_color = colyseus_schema_get(
                    _instance,
                    "color"
                );

                if (is_undefined(_player_color) || _player_color == "") {
                    _player_color = "red";
                }

                var _player_data = {
                    session_id: _key,
                    name: _player_name,
                    color: _player_color,
                    schema: _instance
                };

                net_players[array_length(net_players)] = _player_data;

                if (_key == net_session_id) {
                    net_player_color = _player_color;
                }

                show_debug_message(
                    "PLAYER ADICIONADO | ID: " + string(_key) +
                    " | NOME: " + string(_player_name) +
                    " | COR: " + string(_player_color)
                );

                show_debug_message(
                    "TOTAL APOS ADICAO: " +
                    string(array_length(net_players))
                );
            }
        );

        // Observar jogadores removidos.
        colyseus_on_remove(
            net_callbacks,
            "players",
            function(_instance, _key) {

                show_debug_message("=== ANTES DA REMOCAO ===");

                show_debug_message(
                    "ID REMOVIDO: " + string(_key)
                );

                show_debug_message(
                    "TOTAL ANTES: " +
                    string(array_length(net_players))
                );

                for (var i = 0; i < array_length(net_players); i++) {

                    show_debug_message(
                        "JOGADOR " + string(i) +
                        " | ID: " + string(net_players[i].session_id) +
                        " | NOME: " + string(net_players[i].name)
                    );
                }

                var _remaining_players = [];

                for (var j = 0; j < array_length(net_players); j++) {

                    if (net_players[j].session_id != _key) {

                        _remaining_players[
                            array_length(_remaining_players)
                        ] = net_players[j];
                    }
                }

                net_players = _remaining_players;

                show_debug_message(
                    "PLAYER REMOVIDO | ID: " + string(_key)
                );

                show_debug_message("=== JOGADORES RESTANTES ===");

                for (var k = 0; k < array_length(net_players); k++) {

                    show_debug_message(
                        string(net_players[k].session_id) + " | " +
                        string(net_players[k].name)
                    );
                }

                show_debug_message(
                    "TOTAL: " + string(array_length(net_players))
                );

                show_debug_message("===========================");
            }
        );

        // Receber mensagens do servidor.
        colyseus_on_message(
            _joined_room,
            function(_room_message, _type, _data) {

                // ------------------------------------------
                // ERRO DE COR
                // ------------------------------------------

                if (_type == "color_error") {

                    net_error = string(_data);
                }

                // ------------------------------------------
                // DADOS DO TABULEIRO
                // ------------------------------------------

                else if (_type == "board_data") {

                    var _parsed_data = undefined;

                    // O servidor envia os dados como JSON string.
                    if (is_string(_data)) {

                        _parsed_data = json_parse(_data);
                    }
                    else {

                        _parsed_data = _data;
                    }

                    if (is_array(_parsed_data)) {

                        net_board_data = _parsed_data;
                        net_board_data_received = true;

                        // Registrar nova versão do tabuleiro.
                        net_board_data_version++;

                        show_debug_message(
                            "DADOS DO TABULEIRO RECEBIDOS | VERSÃO: " +
                            string(net_board_data_version)
                        );

                        show_debug_message(
                            "QUANTIDADE DE CASAS: " +
                            string(array_length(net_board_data))
                        );

                    }
                    else {

                        show_debug_message(
                            "ERRO AO CONVERTER board_data. Tipo recebido: " +
                            typeof(_data)
                        );
                    }
                }
            }
        );

        show_debug_message("=== SALA CONECTADA ===");
        show_debug_message("Sala: " + net_room_id);
        show_debug_message("Sessão: " + net_session_id);
        show_debug_message("Líder: " + string(net_host_session_id));
        show_debug_message("Modalidade: " + net_room_mode);
    });

    colyseus_on_error(_room, function(_code, _message) {

        net_connected = false;
        net_busy = false;
        net_status = "Erro ao entrar na sala";
        net_error = _message;

        show_debug_message("ERRO COLYSEUS [" + string(_code) + "]");
        show_debug_message(_message);
    });
};

                net_lobby_rooms = [];

            else if (_type == "public_room") {

                if (is_struct(_data)) {

                    var _found = false;

                    for (var i = 0; i < array_length(net_lobby_rooms); i++) {

                        if (net_lobby_rooms[i].roomId == _data.roomId) {

                            net_lobby_rooms[i] = _data;
                            _found = true;
                            break;
                        }
                    }

                    if (!_found) {

                        net_lobby_rooms[
                            array_length(net_lobby_rooms)
                        ] = _data;
                    }

                    show_debug_message(
                        "PARTIDA RECEBIDA: " + _data.roomId
                    );

                } else {

                    show_debug_message(
                        "ERRO: Dados da partida não são um struct."
                    );
                }
            }
            else if (_type == "rooms_end") {

                show_debug_message(
                    "TOTAL DE PARTIDAS: " +
                    string(array_length(net_lobby_rooms))
                );
            }
        }
    );
};

        }

// Lista de avatares por Steam ID.
avatar_cache = [];

// --------------------------------------------------
// PROCURAR AVATAR
// --------------------------------------------------

avatar_buscar = function(_steam_id_texto) {

    for (var _i = 0; _i < array_length(avatar_cache); _i++) {

        if (avatar_cache[_i].steam_id == _steam_id_texto) {
            return _i;
        }
    }

    return -1;
};

// --------------------------------------------------
// REGISTRAR AVATAR
// --------------------------------------------------

avatar_registrar = function(_steam_id_texto) {

    if (_steam_id_texto == "") return;

    if (avatar_buscar(_steam_id_texto) != -1) {
        return;
    }

    var _dados = {
        steam_id: _steam_id_texto,
        sprite: -1,
        carregado: false,
        solicitado: false,
        timer: 0
    };

    array_push(avatar_cache, _dados);
};

// --------------------------------------------------
// CRIAR SPRITE DA IMAGEM STEAM
// --------------------------------------------------

avatar_criar_sprite = function(_imagem) {

    if (_imagem <= 0) return -1;

    var _dimensoes = steam_utils_get_image_size(_imagem);

    if (!is_struct(_dimensoes)) return -1;

    if (!variable_struct_exists(_dimensoes, "width")) {
        return -1;
    }

    if (!variable_struct_exists(_dimensoes, "height")) {
        return -1;
    }

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;

    if (_largura <= 0 || _altura <= 0) {
        return -1;
    }

    var _tamanho_buffer = _largura * _altura * 4;

    var _buffer = buffer_create(
        _tamanho_buffer,
        buffer_fixed,
        1
    );

    var _sucesso = steam_utils_get_image_rgba(
        _imagem,
        _buffer
    );

    if (!_sucesso) {

        buffer_delete(_buffer);
        return -1;
    }

    var _surface = surface_create(
        _largura,
        _altura
    );

    if (!surface_exists(_surface)) {

        buffer_delete(_buffer);
        return -1;
    }

    buffer_set_surface(
        _buffer,
        _surface,
        0
    );

    var _sprite = sprite_create_from_surface(
        _surface,
        0,
        0,
        _largura,
        _altura,
        false,
        false,
        0,
        0
    );

    surface_free(_surface);
    buffer_delete(_buffer);

    return _sprite;
};

// --------------------------------------------------
// ATUALIZAR AVATAR DE UM JOGADOR
// --------------------------------------------------

avatar_atualizar = function(_indice) {

    if (_indice < 0) return;
    if (_indice >= array_length(avatar_cache)) return;

    var _dados = avatar_cache[_indice];

    if (_dados.carregado) return;

    if (_dados.timer > 0) {

        _dados.timer--;
        return;
    }

    _dados.timer = avatar_intervalo;

    // Converter o ID recebido do Colyseus
    // para um inteiro de 64 bits.
    var _steam_id = int64(_dados.steam_id);

    // --------------------------------------------------
    // SOLICITAR INFORMAÇÕES DO USUÁRIO
    // --------------------------------------------------

    if (!_dados.solicitado) {

        steam_friends_request_user_information(
            _steam_id,
            false
        );

        _dados.solicitado = true;
    }

    // --------------------------------------------------
    // BUSCAR FOTO GRANDE DA STEAM
    // --------------------------------------------------

    var _imagem = steam_friends_get_large_friend_avatar(
        _steam_id
    );

    // Ainda não disponível.
    if (_imagem <= 0) return;

    // --------------------------------------------------
    // CONVERTER IMAGEM EM SPRITE
    // --------------------------------------------------

    var _sprite = avatar_criar_sprite(_imagem);

    if (_sprite == -1) return;

    _dados.sprite = _sprite;
    _dados.carregado = true;

    show_debug_message(
        "Avatar Steam carregado no lobby: " +
        _dados.steam_id
    );
};

friends_connect = function() {

    var _bridge = friends_get_bridge();

    if (_bridge == noone) {

        friends_room = 0;

        friends_connected = false;
        friends_connecting = false;
        friends_registered = false;

        friends_status =
            "Aguardando sistema de amizades.";

        return false;
    }

    // ----------------------------------------------
    // BRIDGE AINDA NAO CONECTADO
    // ----------------------------------------------

    if (
        !_bridge.friend_conectado ||
        _bridge.friend_room == 0
    ) {

        friends_room = 0;

        friends_connected = false;
        friends_connecting =
            _bridge.friend_conectando;

        friends_registered = false;

        friends_status =
            "Conectando ao sistema de amizades...";

        return false;
    }

    // ----------------------------------------------
    // VALIDAR SALA
    // ----------------------------------------------

    if (
        !colyseus_room_is_connected(
            _bridge.friend_room
        )
    ) {

        friends_room = 0;

        friends_connected = false;
        friends_connecting = false;
        friends_registered = false;

        friends_status =
            "Aguardando reconexao...";

        return false;
    }

    // ----------------------------------------------
    // UTILIZAR A CONEXAO DO BRIDGE
    // ----------------------------------------------

    friends_room =
        _bridge.friend_room;

    friends_connected = true;
    friends_connecting = false;
    friends_registered = true;

    friends_status =
        "Conectado ao sistema de amizades.";

    // ----------------------------------------------
    // SINCRONIZAR IDENTIDADE
    // ----------------------------------------------

    var _steam = instance_find(
        obj_steam,
        0
    );

    if (_steam != noone) {

        friends_steam_id = string(
            _steam.steam_player_id_string
        );
    }

    var _menu = instance_find(
        obj_menu,
        0
    );

    if (_menu != noone) {

        // No modo de teste, este e o codigo
        // aleatorio exclusivo da instancia.
        //
        // Nao substituir pelo codigo da Steam.

        if (
            _menu.menu_codigo_amigo_modo_teste
        ) {

            friends_code = string(
                _menu.menu_codigo_amigo_teste
            );

        } else {

            friends_code = string(
                _menu.menu_amigos_texto_codigo
            );
        }
    }

    return true;
};

// --------------------------------------------------
// ENVIAR PEDIDO DE AMIZADE
// --------------------------------------------------

friends_send_request = function(_code) {

    _code = string_trim(
        string(_code)
    );

    // ----------------------------------------------
    // VALIDAR CODIGO
    // ----------------------------------------------

    if (string_length(_code) != 6) {

        friends_error =
            "O codigo deve ter 6 caracteres.";

        return false;
    }

    // ----------------------------------------------
    // SINCRONIZAR CONEXAO
    // ----------------------------------------------

    if (!friends_connect()) {

        friends_error =
            "Servidor de amizades desconectado.";

        return false;
    }

    var _bridge = friends_get_bridge();

    if (_bridge == noone) {

        friends_error =
            "Sistema de amizades indisponivel.";

        return false;
    }

    // ----------------------------------------------
    // IDENTIFICAR INSTANCIA DE TESTE
    // ----------------------------------------------

    var _instance_id = "";

    if (variable_instance_exists(
        _bridge,
        "friend_obter_instancia"
    )) {

        _instance_id =
            string(
                _bridge.friend_obter_instancia()
            );
    }

    // ----------------------------------------------
    // ENVIAR PELA CONEXAO PRINCIPAL
    // ----------------------------------------------

    colyseus_send(
        _bridge.friend_room,
        "friend_request",
        {
            code: _code,
            instanceId: _instance_id
        }
    );

    friends_error = "";

    friends_status =
        "Pedido de amizade enviado.";

    friends_last_request_message =
        "Pedido de amizade enviado.";

    show_debug_message(
        "[FRIENDS] Pedido enviado pelo bridge | " +
        "Codigo: " + _code +
        " | Instancia: " + _instance_id
    );

    return true;
};

        friends_error =

        return false;
    }

    var _bridge = friends_get_bridge();

    if (_bridge == noone) {

        friends_error =
            "Sistema de amizades indisponivel.";

        return false;
    }

    var _id = string(
        _request_id
    );

    if (_id == "") {
        return false;
    }

    // ----------------------------------------------
    // ENVIAR RESPOSTA
    // ----------------------------------------------

        _bridge.friend_room,
        "friend_response",
        {
            requestId: _id,
            accept: _accept
        }
    );

    show_debug_message(
        "[FRIENDS] Resposta enviada pelo bridge | " +
        "Pedido: " + _id +
        " | Aceitou: " + string(_accept)
    );

    return true;
};

// --------------------------------------------------

popup_carregar_avatar = function(_pedido) {

    // Avatar recebido diretamente.
    if (_pedido.avatar_sprite != -1) {
        return;
    }

    var _menu = instance_find(obj_menu, 0);

    if (_menu == noone) return;

    // ----------------------------------------------
    // REAPROVEITAR AVATAR JA CARREGADO
    // ----------------------------------------------

    for (var _i = 0; _i < array_length(_menu.menu_amigos); _i++) {

        var _amigo = _menu.menu_amigos[_i];

        if (_amigo.steam_id == _pedido.steam_id) {

            if (
                _amigo.sprite != -1 &&
                sprite_exists(_amigo.sprite)
            ) {

                _pedido.avatar_sprite = _amigo.sprite;
                _pedido.avatar_proprio = false;

                return;
            }
        }
    }

    // ----------------------------------------------
    // SOLICITAR AVATAR PELA STEAM
    // ----------------------------------------------

    if (!steam_api_is_initialized()) return;

    if (
        _pedido.steam_id == 0 ||
        _pedido.steam_id == ""
    ) {
        return;
    }

    var _imagem =
        steam_friends_get_large_friend_avatar(
            _pedido.steam_id
        );

    if (_imagem <= 0) return;

    var _novo_sprite =
        _menu.menu_amigos_criar_sprite(_imagem);

    if (_novo_sprite != -1) {

        _pedido.avatar_sprite = _novo_sprite;

        // Este sprite pertence ao popup.
        _pedido.avatar_proprio = true;

// GERAR CODIGO DE AMIGO
// --------------------------------------------------

friend_gerar_codigo = function(_steam_id) {

    var _identidade =
        steam_user_decode_steam_id(_steam_id);

    if (!is_struct(_identidade)) return "";

    if (!variable_struct_exists(
        _identidade,
        "account_id"
    )) return "";

    if (
        variable_struct_exists(
            _identidade,
            "is_valid"
        ) &&
        !_identidade.is_valid
    ) return "";

    var _numero = floor(_identidade.account_id);

    if (_numero < 0 || _numero > 4294967295) {
        return "";
    }

    var _alfabeto =
        "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

    var _codigo = "";

    repeat (6) {

        var _digito = _numero mod 62;

        _codigo =
            string_char_at(_alfabeto, _digito + 1) +
            _codigo;

        _numero = floor(_numero / 62);
    }

    return _codigo;
};

                // --------------------------------------

                    if (

                    // identifica cada amizade.

                    // ----------------------------------
                    // ADICIONAR AMIGO
                    // ----------------------------------

                    array_push(
                        _nova_lista,
                        {

                            steam_id: 0,

                            steam_id_texto: _steam_id,

                            nome: _nome,

                            codigo: _codigo,

    // Identifica a instancia do jogo que originou este pedido.
    originInstanceId?: string;
};

type StoredData = {
    accounts: Record<string, Account>;
    requests: FriendRequest[];
};

type JoinOptions = {
    steamId?: string;
    playerName?: string;
    friendCode?: string;
    instanceId?: string;
};

// CONFIGURACOES
// --------------------------------------------------

const CODE_PATTERN = /^[0-9A-Za-z]{6}$/;
const STEAM_PATTERN = /^[0-9]{17}$/;

// Usa a mesma conversao Base62 do obj_menu no GameMaker.
// SteamID64 contem o AccountID nos 32 bits menos significativos.
// BigInt impede perda de precisao com SteamIDs de 17 digitos.
const FRIEND_CODE_ALPHABET =
    "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

function friendCodeFromSteamId(steamId: string): string {
    if (!STEAM_PATTERN.test(steamId)) {
        throw new Error("SteamID invalido.");
    }

    let accountId = BigInt(steamId) & 0xFFFFFFFFn;
    let code = "";

    for (let i = 0; i < 6; i++) {
        const digit = Number(accountId % 62n);
        code = FRIEND_CODE_ALPHABET[digit] + code;
        accountId /= 62n;
    }

    return code;
}

// Identidades virtuais nao sao contas Steam reais.
// O prefixo nao pode ser digitado como codigo de amigo.
const TEST_ACCOUNT_PREFIX = "dev:";

const DATA_PATH = resolve(
    process.cwd(),
    "data",
    "money_town_friends.json"
);

// Desativar os testes quando o servidor estiver
// configurado para producao.
const DEVELOPMENT_MODE =
    process.env.NODE_ENV !== "production";

// --------------------------------------------------
// BANCO DE DADOS
// --------------------------------------------------

class FriendsDatabase {

    private data: StoredData = {
        accounts: {},
        requests: []
    };

    constructor() {
        this.load();
    }

    // --------------------------------------------------
    // CARREGAR DADOS
    // --------------------------------------------------

    private load() {

        if (!existsSync(DATA_PATH)) {
            return;
        }

        try {

            const raw = readFileSync(
                DATA_PATH,
                "utf8"
            );

            const parsed = JSON.parse(raw);

            this.data = {
                accounts: parsed.accounts ?? {},
                requests: parsed.requests ?? []
            };

            console.log(
                "[FRIENDS] Dados carregados."
            );

        } catch (error) {

            console.error(
                "[FRIENDS] Erro ao carregar dados:",
                error
            );

            throw error;
        }
    }

    // --------------------------------------------------
    // SALVAR DADOS
    // --------------------------------------------------

    save() {

        mkdirSync(
            dirname(DATA_PATH),
            {
                recursive: true
            }
        );

        const temporary =
            DATA_PATH + ".tmp";

        writeFileSync(
            temporary,
            JSON.stringify(
                this.data,
                null,
                2
            ),
            "utf8"
        );

        renameSync(
            temporary,
            DATA_PATH
        );
    }

    // --------------------------------------------------
    // OBTER CONTA
    // --------------------------------------------------

    getAccount(steamId: string) {

        return this.data.accounts[steamId];
    }

    // --------------------------------------------------
    // BUSCAR CONTA PELO CODIGO
    // --------------------------------------------------

    getByCode(code: string) {
        // A conta DEV existe de verdade no banco e e encontrada
        // pelo codigo gerado pela SUA janela, nao pela Steam.
        // Preferir uma conta de desenvolvimento conectada.
        return Object.values(this.data.accounts).find(account =>
            account.code === code &&
            (account.steamId.startsWith(TEST_ACCOUNT_PREFIX) ||
             (STEAM_PATTERN.test(account.steamId) &&
              friendCodeFromSteamId(account.steamId) === code))
        );
    }

    // --------------------------------------------------
    // REGISTRAR CONTA
    // --------------------------------------------------

    register(
        steamId: string,
        name: string,
        code: string
    ): Account {

        const existing =
            this.getAccount(steamId);

        const owner =
            this.getByCode(code);

        if (
            owner &&
            owner.steamId !== steamId
        ) {

            throw new Error(
                "Este codigo ja pertence a outra conta."
            );
        }

        if (existing) {

            existing.name = name;

            // Atualizar registros antigos para o codigo correto.
            existing.code = code;

            this.save();

            return existing;
        }

        const account: Account = {
            steamId,
            code,
            name,
            friends: []
        };

        this.data.accounts[steamId] =
            account;

        this.save();

        return account;
    }

    // --------------------------------------------------
    // OBTER AMIGOS
    // --------------------------------------------------

    getFriends(steamId: string): Account[] {

        const account =
            this.getAccount(steamId);

        if (!account) {
            return [];
        }

        return account.friends
            .map(
                id => this.getAccount(id)
            )
            .filter(
                (friend): friend is Account =>
                    friend !== undefined
            );
    }

    // --------------------------------------------------
    // OBTER PEDIDOS PENDENTES
    // --------------------------------------------------

    getPending(steamId: string) {

        return this.data.requests.filter(
            request =>
                request.to === steamId
        );
    }

    // --------------------------------------------------
    // IDENTIDADE VIRTUAL DE DESENVOLVIMENTO
    // --------------------------------------------------

    private getTestAccountId(
        steamId: string,
        code?: string
    ) {
        // Codigo aleatorio cria uma identidade virtual independente.
        // Para o proprio codigo, preserva a identidade de teste antiga.
        return code
            ? `${TEST_ACCOUNT_PREFIX}${steamId}:${code}`
            : `${TEST_ACCOUNT_PREFIX}${steamId}`;
    }

    // --------------------------------------------------
    // CRIAR CONTA VIRTUAL DE TESTE
    // --------------------------------------------------

    private ensureTestAccount(
        realAccount: Account,
        code?: string
    ): Account {

        const virtualId = this.getTestAccountId(
            realAccount.steamId,
            code
        );

        const existing = this.getAccount(virtualId);

        if (existing) {
            existing.name = code
                ? `Jogador ${code} (Teste)`
                : `${realAccount.name} (Teste)`;
            return existing;
        }

        const virtualAccount: Account = {
            steamId: virtualId,
            code: code ?? "__DEV__",
            name: code
                ? `Jogador ${code} (Teste)`
                : `${realAccount.name} (Teste)`,
            friends: []
        };

        this.data.accounts[virtualId] = virtualAccount;
        return virtualAccount;
    }

    // --------------------------------------------------
    // CRIAR PEDIDO SIMULADO
    // --------------------------------------------------

    private createTestRequest(
        realAccount: Account,
        code?: string
    ) {

        if (!DEVELOPMENT_MODE) {
            throw new Error("Testes desativados no servidor.");
        }

        const virtualAccount = this.ensureTestAccount(
            realAccount,
            code
        );

        if (realAccount.friends.includes(virtualAccount.steamId)) {
            throw new Error("Esse jogador de teste ja e seu amigo.");
        }

        const duplicate = this.data.requests.find(request =>
            request.from === virtualAccount.steamId &&
            request.to === realAccount.steamId
        );

        if (duplicate) {
            throw new Error("Ja existe um pedido de teste pendente.");
        }

        const request: FriendRequest = {
            id: randomUUID(),
            from: virtualAccount.steamId,
            to: realAccount.steamId,
            createdAt: Date.now()
        };

        this.data.requests.push(request);
        this.save();

        console.log(
            "[FRIENDS DEV] Pedido simulado:",
            virtualAccount.steamId,
            "->",
            realAccount.steamId
        );

        return {
            request,
            target: realAccount,
            simulated: true
        };
    }

    // --------------------------------------------------
    // CRIAR PEDIDO DE AMIZADE
    // --------------------------------------------------

    createRequest(
        from: string,
        code: string
    ) {

        const sender =
            this.getAccount(from);

        const target =
            this.getByCode(code);

        if (!sender) {

            throw new Error(
                "Conta nao registrada."
            );
        }

        if (!target) {
            throw new Error(
                "Codigo nao encontrado. A outra instancia deve estar conectada."
            );
        }

        // ==============================================
        // PEDIDO PARA A PROPRIA CONTA
        // ==============================================

        if (target.steamId === from) {
            throw new Error("Voce nao pode adicionar sua propria conta.");
        }

        // ==============================================
        // PEDIDO NORMAL
        // ==============================================

        if (
            sender.friends.includes(
                target.steamId
            )
        ) {

            throw new Error(
                "Esse jogador ja e seu amigo."
            );
        }

        const duplicate =
            this.data.requests.find(
                request =>
                    (
                        request.from === from &&
                        request.to === target.steamId
                    ) ||
                    (
                        request.from === target.steamId &&
                        request.to === from
                    )
            );

        if (duplicate) {

            throw new Error(
                "Ja existe um pedido de amizade pendente."
            );
        }

        const request: FriendRequest = {

            id: randomUUID(),

            from,

            to: target.steamId,

            createdAt: Date.now()
        };

        this.data.requests.push(
            request
        );

        this.save();

        return {
            request,
            target,
            simulated: false
        };
    }

    // --------------------------------------------------
    // RESPONDER PEDIDO
    // --------------------------------------------------

    answerRequest(
        steamId: string,
        requestId: string,
        accept: boolean
    ) {

        const index =
            this.data.requests.findIndex(
                request =>
                    request.id === requestId &&
                    request.to === steamId
            );

        if (index === -1) {

            throw new Error(
                "Pedido nao encontrado."
            );
        }

        const request =
            this.data.requests[index];

        const sender =
            this.getAccount(
                request.from
            );

        const target =
            this.getAccount(
                request.to
            );

        // Validar antes de remover o pedido.
        if (
            accept &&
            (!sender || !target)
        ) {

            throw new Error(
                "Uma das contas nao existe."
            );
        }

        // Remover pedido pendente.
        this.data.requests.splice(
            index,
            1
        );

        // ==============================================
        // ACEITAR PEDIDO
        // ==============================================

        if (
            accept &&
            sender &&
            target
        ) {

            if (
                !sender.friends.includes(
                    target.steamId
                )
            ) {

                sender.friends.push(
                    target.steamId
                );
            }

            if (
                !target.friends.includes(
                    sender.steamId
                )
            ) {

                target.friends.push(
                    sender.steamId
                );
            }
        }

        this.save();

        return request;
    }

    // --------------------------------------------------
    // VERIFICAR CONTA DE TESTE
    // --------------------------------------------------

    removeFriend(ownerId: string, targetId: string) {
        const owner = this.getAccount(ownerId);
        const target = this.getAccount(targetId);
        if (!owner || !target || !owner.friends.includes(targetId)) {
            throw new Error("Amizade nao encontrada.");
        }
        const beforeOwner = [...owner.friends];
        const beforeTarget = [...target.friends];
        owner.friends = owner.friends.filter(id => id !== targetId);
        target.friends = target.friends.filter(id => id !== ownerId);
        try { this.save(); } catch (error) {
            owner.friends = beforeOwner;
            target.friends = beforeTarget;
            throw error;
        }
    }

    isTestAccount(steamId: string) {

        return steamId.startsWith(
            TEST_ACCOUNT_PREFIX
        );
    }
}

const database =
    new FriendsDatabase();

// --------------------------------------------------
// SALA DE AMIZADES
// --------------------------------------------------

export class FriendsRoom extends Room {

    maxClients = 1000;

    autoDispose = false;

    private sessions =
        new Map<string, string>();

    // Uma mesma instancia GameMaker pode abrir varias conexoes FriendsRoom.
    // Todas compartilham o mesmo instanceId/codigo aleatorio de teste.
    private sessionInstances = new Map<string, string>();

    // Identifica qual instancia enviou cada pedido nesta execucao.
    // Necessario nos testes com duas instancias usando a mesma SteamID.
    private requestOriginSessions =
        new Map<string, string>();

    // --------------------------------------------------
    // CRIAR SALA
    // --------------------------------------------------

    onCreate() {

        console.log(
            "[FRIENDS] Sala de amizades iniciada."
        );

        console.log(
            "[FRIENDS] Modo de desenvolvimento:",
            DEVELOPMENT_MODE
        );

        // ==============================================
        // RECEBER PEDIDO DE AMIZADE
        // ==============================================

        this.onMessage(
            "friend_request",
            (client, message) => {

                try {

                    const senderId =
                        this.sessions.get(
                            client.sessionId
                        );

                    if (!senderId) {

                        throw new Error(
                            "Conta nao identificada."
                        );
                    }

                    const code =
                        String(
                            message?.code ?? ""
                        ).trim();

                    if (
                        !CODE_PATTERN.test(
                            code
                        )
                    ) {

                        throw new Error(
                            "O codigo deve ter 6 caracteres."
                        );
                    }

                    const result =
                        database.createRequest(
                            senderId,
                            code
                        );

                    // ----------------------------------
                    // CONFIRMAR ENVIO
                    // ----------------------------------

                    client.send(
                        "friend_request_result",
                        {

                            ok: true,

                            message:
                                result.simulated
                                    ? "Pedido de teste enviado."
                                    : "Pedido de amizade enviado.",

                            requestId:
                                result.request.id
                        }
                    );

                    // ----------------------------------
                    // ENTREGAR APENAS AO DESTINATARIO
                    // ----------------------------------

                    // Nao notificar a instancia que enviou o pedido.
                    // No modo de teste, duas instancias podem usar a
                    // mesma SteamID, mas possuem sessionIds distintos.
                    const originInstanceId =
                        String(message?.instanceId ?? "").trim() ||
                        this.sessionInstances.get(client.sessionId) ||
                        client.sessionId;

                    // Salvar a origem no pedido para o filtro continuar
                    // funcionando mesmo depois de reconectar/reiniciar.
                    result.request.originInstanceId = originInstanceId;
                    database.save();

                    this.requestOriginSessions.set(
                        result.request.id,
                        client.sessionId
                    );

                    this.sendToAccount(
                        result.target.steamId,
                        "friend_request_received",
                        this.serializeRequest(result.request),
                        client.sessionId,
                        originInstanceId
                    );

                    console.log(
                        "[FRIENDS] Origem da solicitacao:",
                        originInstanceId
                    );

                    console.log(
                        "[FRIENDS] Pedido enviado:",
                        result.request.from,
                        "->",
                        result.request.to
                    );

                } catch (error) {

                    client.send(
                        "friend_request_result",
                        {

                            ok: false,

                            message:
                                this.errorText(
                                    error
                                )
                        }
                    );
                }
            }
        );

        // ==============================================
        // ACEITAR OU REJEITAR
        // ==============================================

        this.onMessage(
            "friend_response",
            (client, message) => {

                try {

                    const steamId =
                        this.sessions.get(
                            client.sessionId
                        );

                    if (!steamId) {

                        throw new Error(
                            "Conta nao identificada."
                        );
                    }

                    const requestId =
                        String(
                            message?.requestId ?? ""
                        );

                    const accept =
                        message?.accept === true;

                    const request =
                        database.answerRequest(
                            steamId,
                            requestId,
                            accept
                        );

                    this.requestOriginSessions.delete(request.id);

                    // ----------------------------------
                    // CONFIRMAR RESPOSTA
                    // ----------------------------------

                    client.send(
                        "friend_response_result",
                        {

                            ok: true,

                            requestId,

                            accepted: accept
                        }
                    );

                    // ----------------------------------
                    // NOTIFICAR REMETENTE
                    // ----------------------------------

                    this.sendToAccount(
                        request.from,
                        "friend_request_answered",
                        { requestId, accepted: accept, by: request.to }
                    );

                    // ----------------------------------
                    // ATUALIZAR LISTAS
                    // ----------------------------------

                    this.broadcastFriends(
                        request.from
                    );

                    this.broadcastFriends(
                        request.to
                    );

                    console.log(
                        "[FRIENDS] Pedido",
                        requestId,
                        accept
                            ? "ACEITO"
                            : "RECUSADO"
                    );

                } catch (error) {

                    client.send(
                        "friend_response_result",
                        {

                            ok: false,

                            message:
                                this.errorText(
                                    error
                                )
                        }
                    );
                }
            }
        );

        // ==============================================
        // REMOVER AMIZADE DOS DOIS LADOS
        // ==============================================
        this.onMessage("friend_remove", (client, message) => {
            const ownerId = this.sessions.get(client.sessionId);
            const targetId = String(message?.steamId ?? "").trim();
            try {
                if (!ownerId) throw new Error("Conta nao identificada.");
                database.removeFriend(ownerId, targetId);
                client.send("friend_remove_result", {ok:true, steamId:targetId});
                this.broadcastFriends(ownerId);
                this.broadcastFriends(targetId);
                console.log("[FRIENDS] Amizade removida:",ownerId,targetId);
            } catch(error) {
                client.send("friend_remove_result", {
                    ok:false, steamId:targetId, message:this.errorText(error)
                });
            }
        });

        // ==============================================
        // CONSULTAR LISTA DE AMIGOS
        // ==============================================

        this.onMessage(
            "friend_list",
            client => {

                const steamId =
                    this.sessions.get(
                        client.sessionId
                    );

                if (steamId) {

                    this.sendFriends(
                        client,
                        steamId
                    );
                }
            }
        );
    }

    // --------------------------------------------------
    // JOGADOR ENTROU NA SALA DE AMIZADES
    // --------------------------------------------------

    onJoin(
        client: Client,
        options: JoinOptions
    ) {

        try {

            const steamId =
                String(
                    options?.steamId ?? ""
                ).trim();

            const playerName =
                String(
                    options?.playerName ??
                    "Jogador"
                ).slice(0, 64);

            const suppliedFriendCode =
                String(
                    options?.friendCode ?? ""
                ).trim();

            // ------------------------------------------
            // VALIDAR STEAM ID
            // ------------------------------------------

            if (
                !STEAM_PATTERN.test(
                    steamId
                )
            ) {

                throw new Error(
                    "SteamID invalido."
                );
            }

            // --------------------------------------------------
            // IDENTIDADE DE DESENVOLVIMENTO: CONTA POR JANELA
            // --------------------------------------------------
            const realCode = friendCodeFromSteamId(steamId);
            const explicitInstanceId = String(options?.instanceId ?? "").trim();
            const testMode = DEVELOPMENT_MODE &&
                CODE_PATTERN.test(suppliedFriendCode) &&
                suppliedFriendCode !== realCode;

            // Duas janelas com a mesma SteamID e codigos distintos
            // representam DUAS contas diferentes no banco.
            const accountId = testMode
                ? `dev:${steamId}:${suppliedFriendCode}`
                : steamId;
            const friendCode = testMode ? suppliedFriendCode : realCode;
            const instanceId = testMode
                ? suppliedFriendCode
                : (explicitInstanceId || client.sessionId);

            const account = database.register(
                accountId, playerName, friendCode
            );

            this.sessions.set(client.sessionId, accountId);
            this.sessionInstances.set(client.sessionId, instanceId);

            console.log(
                "[FRIENDS] Identidade:", accountId,
                "| codigo:", friendCode,
                "| dev:", testMode
            );

            console.log(
                "[FRIENDS] Sessao registrada:",
                client.sessionId,
                "| Instancia:", instanceId
            );

            // ------------------------------------------
            // CONFIRMAR CADASTRO
            // ------------------------------------------

            client.send(
                "friend_registered",
                {

                    ok: true,

                    code:
                        account.code,

                    steamId:
                        account.steamId
                }
            );

            // ------------------------------------------
            // ENVIAR LISTA DE AMIGOS
            // ------------------------------------------

            this.sendFriends(
                client,
                accountId
            );

            // ------------------------------------------
            // ENVIAR PEDIDOS PENDENTES
            // ------------------------------------------

            const pending =
                database.getPending(
                    accountId
                );

            for (const request of pending) {

                // Nunca reenviar um pedido de teste legado sem
                // origem rastreavel: isso mostrava o popup ao remetente.

                // Ao reconectar, manter a exclusao da INSTANCIA que
                // enviou, mesmo se ela recebeu novo sessionId.
                if (
                    request.originInstanceId === instanceId ||
                    this.requestOriginSessions.get(request.id) === client.sessionId
                ) {
                    continue;
                }

                client.send(
                    "friend_request_received",
                    this.serializeRequest(
                        request
                    )
                );
            }

            // ------------------------------------------
            // ATUALIZAR STATUS ONLINE
            // ------------------------------------------

            this.broadcastFriends(
                accountId
            );

            console.log(
                "[FRIENDS] Conta conectada:",
                account.name,
                "| Codigo:",
                account.code
            );

        } catch (error) {

            client.send(
                "friend_registered",
                {

                    ok: false,

                    message:
                        this.errorText(
                            error
                        )
                }
            );

            console.error(
                "[FRIENDS] Erro no cadastro:",
                error
            );
        }
    }

    // --------------------------------------------------
    // JOGADOR SAIU
    // --------------------------------------------------

    onLeave(client: Client) {

        const steamId =
            this.sessions.get(
                client.sessionId
            );

        this.sessions.delete(
            client.sessionId
        );
        this.sessionInstances.delete(client.sessionId);

        if (steamId) {

            this.broadcastFriends(
                steamId
            );
        }
    }

    // --------------------------------------------------
    // VERIFICAR SE CONTA ESTA ONLINE
    // --------------------------------------------------

    private isOnline(
        steamId: string
    ) {

        for (
            const id of
            this.sessions.values()
        ) {

            if (id === steamId) {
                return true;
            }
        }

        return false;
    }

    // --------------------------------------------------
    // CONVERTER PEDIDO PARA O GAMEMAKER
    // --------------------------------------------------

    private serializeRequest(
        request: FriendRequest
    ) {

        const sender =
            database.getAccount(
                request.from
            );

        const simulated =
            database.isTestAccount(
                request.from
            );

        return {
            requestId: request.id,
            // Identidade real dentro do Money Town, inclusive em testes.
            steamId: request.from,
            name: sender?.name ?? "Jogador",
            code: sender?.code ?? "------",
            simulated
        };
    }

    // --------------------------------------------------
    // ENVIAR LISTA DE AMIGOS
    // --------------------------------------------------

    private sendFriends(
        client: Client,
        steamId: string
    ) {

        const friends =
            database.getFriends(
                steamId
            ).map(
                account => {

                    const simulated =
                        database.isTestAccount(
                            account.steamId
                        );

                    return {

                        steamId:
                            account.steamId,

                        name:
                            account.name,

                        code:
                            account.code,

                        online: this.isOnline(account.steamId),

                        simulated
                    };
                }
            );

        // Enviar tambem em JSON para clientes GameMaker cujas versoes
        // da extensao Colyseus nao preservam arrays aninhados.
        client.send(
            "friend_list_result",
            {
                friends,
                friendsJson: JSON.stringify(friends),
                count: friends.length
            }
        );

        console.log(
            "[FRIENDS] Lista enviada:",
            steamId,
            "| amigos:",
            friends.length
        );
    }

    // --------------------------------------------------
    // ENVIAR MENSAGEM A UMA CONTA
    // --------------------------------------------------

    private sendToAccount(
        steamId: string,
        type: string,
        data: unknown,
        excludeSessionId?: string,
        excludeInstanceId?: string
    ) {

        for (const client of this.clients) {

            if (
                client.sessionId !== excludeSessionId &&
                (!excludeInstanceId ||
                    this.sessionInstances.get(client.sessionId) !== excludeInstanceId) &&
                this.sessions.get(
                    client.sessionId
                ) === steamId
            ) {

                client.send(
                    type,
                    data
                );
            }
        }
    }

    // --------------------------------------------------
    // ATUALIZAR LISTAS DE AMIZADES
    // --------------------------------------------------

    private broadcastFriends(
        steamId: string
    ) {

        const affected =
            new Set<string>();

        affected.add(
            steamId
        );

        for (
            const account of
            database.getFriends(
                steamId
            )
        ) {

            affected.add(
                account.steamId
            );
        }

        for (const client of this.clients) {

            const connectedId =
                this.sessions.get(
                    client.sessionId
                );

            if (
                connectedId &&
                affected.has(
                    connectedId
                )
            ) {

                this.sendFriends(
                    client,
                    connectedId
                );
            }
        }
    }

    ) {

        return error instanceof Error
            ? error.message
            : "Erro desconhecido.";
    }

"
partyroom:"
import { Room, Client, matchMaker } from "colyseus";

// --------------------------------------------------

type Member = {

type Party = {

type Invitation = {

type GameMode = "public" | "private";

type RoomCandidate = {

type PartyReservation = {

export class PartyRoom extends Room {

    maxClients = 1000;
    autoDispose = false;

    private members = new Map<string, Member>();
    private parties = new Map<string, Party>();
    private invitations = new Map<string, Invitation>();

    // --------------------------------------------------
    // IDENTIDADE DO JOGADOR
    // --------------------------------------------------

    private keyFor(
        steamId: string,
        instanceId: string
    ): string {

        return steamId + "|" + instanceId;
    }

    private memberOf(
        client: Client
    ): Member | undefined {

        return this.members.get(client.sessionId);
    }

    private memberByKey(
        key: string
    ): Member | undefined {

        return [...this.members.values()].find(
            member => member.key === key
        );
    }

    private partyOf(
        key: string
    ): Party | undefined {

        for (const party of this.parties.values()) {

            if (party.members.includes(key)) {
                return party;
            }
        }

        return undefined;
    }

    private clientOf(
        key: string
    ): Client | undefined {

        const member = this.memberByKey(key);

        if (!member) return undefined;

        return this.clients.find(
            client =>
                client.sessionId === member.sessionId
        );
    }

    // --------------------------------------------------
    // ICONE MONEY TOWN
    // --------------------------------------------------
    //
    // Cada identidade recebe um indice de 0 a 3.
    //
    // GameMaker:
    // 0 -> spr_icon_player1
    // 1 -> spr_icon_player2
    // 2 -> spr_icon_player3
    // 3 -> spr_icon_player4
    //
    // A foto Steam e escolhida no GameMaker
    // quando o integrante e amigo pela Steam.
    // --------------------------------------------------

    private iconFor(
        key: string
    ): number {

        let hash = 0;

        for (let i = 0; i < key.length; i++) {

            hash = (
                hash * 31 + key.charCodeAt(i)
            ) >>> 0;
        }

        return hash % 4;
    }

    // --------------------------------------------------
    // PUBLICAR ATUALIZACAO DO GRUPO
    // --------------------------------------------------

    private publish(
        party: Party
    ): void {

        const result = party.members
            .map(key => this.memberByKey(key))
            .filter(
                (member): member is Member =>
                    member !== undefined
            )
            .map(member => ({

                key: member.key,

                steamId: member.steamId,

                name: member.name,

                iconIndex: this.iconFor(
                    member.key
                )
            }));

        for (const key of party.members) {

            const client = this.clientOf(key);

            if (!client) continue;

            client.send(
                "party_group",
                {
                    partyId: party.id,

                    leaderKey: party.leader,

                    yourKey: key,

                    membersJson: JSON.stringify(
                        result
                    )
                }
            );
        }

        console.log(
            "[PARTY] Grupo atualizado:",
            party.id,
            "| Integrantes:",
            result.length,
            "| Lider:",
            party.leader
        );
    }

    // --------------------------------------------------
    // CRIAR GRUPO INDIVIDUAL
    // --------------------------------------------------

    private createSolo(
        member: Member
    ): Party {

        const party: Party = {

            id: randomUUID(),

            leader: member.key,

            members: [member.key],

            launching: false
        };

        this.parties.set(
            party.id,
            party
        );

        this.publish(party);

        return party;
    }

    // --------------------------------------------------
    // SAIR DO GRUPO ATUAL
    // --------------------------------------------------

    private detach(
        key: string
    ): void {

        const party = this.partyOf(key);

        if (!party) return;

        party.members = party.members.filter(
            memberKey => memberKey !== key
        );

        if (party.members.length === 0) {

            this.parties.delete(party.id);

            console.log(
                "[PARTY] Grupo vazio removido:",
                party.id
            );

            return;
        }

        if (party.leader === key) {

            party.leader = party.members[0];
        }

        party.launching = false;

        this.publish(party);
    }

    // --------------------------------------------------
    // ENTREGAR CONVITE
    // --------------------------------------------------
    //
    // Esta funcao NAO cria uma partida.
    //
    // O convite e entregue ao GameMaker
    // pela mensagem "party_invitation".
    //
    // O GameMaker utiliza essa mensagem
    // para exibir o popup por 10 segundos.
    // --------------------------------------------------

    private deliver(
        invite: Invitation
    ): boolean {

        // ----------------------------------------------
        // LOCALIZAR DESTINATARIO
        // ----------------------------------------------

        const target = [...this.members.values()].find(
            member => {

                if (
                    member.steamId !==
                    invite.targetSteamId
                ) {
                    return false;
                }

                if (
                    invite.targetInstanceId !== "" &&
                    member.instanceId !==
                    invite.targetInstanceId
                ) {
                    return false;
                }

                return true;
            }
        );

        if (!target) {

            console.log(
                "[PARTY] Destinatario ainda nao conectado:",
                invite.targetSteamId,
                "| Instancia:",
                invite.targetInstanceId || "qualquer"
            );

            return false;
        }

        // ----------------------------------------------
        // LOCALIZAR REMETENTE
        // ----------------------------------------------

        const sender = this.memberByKey(
            invite.senderKey
        );

        const targetClient = this.clientOf(
            target.key
        );

        if (!sender || !targetClient) {

            console.warn(
                "[PARTY] Convite sem remetente " +
                "ou destinatario conectado:",
                invite.id
            );

            return false;
        }

        const party = this.parties.get(
            invite.partyId
        );

        if (!party) {

            console.warn(
                "[PARTY] Grupo do convite nao existe:",
                invite.partyId
            );

            return false;
        }

        if (
            party.launching ||
            party.members.length >= MAX_PARTY_MEMBERS
        ) {

            return false;
        }

        if (
            party.members.includes(target.key)
        ) {

            return false;
        }

        // ----------------------------------------------
        // INICIAR PRAZO DE 10 SEGUNDOS
        // ----------------------------------------------

        invite.delivered = true;

        invite.expiresAt =
            Date.now() + INVITATION_DURATION;

        // ----------------------------------------------
        // ENVIAR POPUP AO DESTINATARIO
        // ----------------------------------------------

        targetClient.send(
            "party_invitation",
            {

                invitationId: invite.id,

                senderName: sender.name,

                // Permite identificar a foto Steam
                // do jogador que enviou o convite.
                senderSteamId: sender.steamId,

                senderInstanceId: sender.instanceId,

                partyId: invite.partyId,

                durationMs: INVITATION_DURATION
            }
        );

        console.log(
            "[PARTY] Convite entregue:",
            sender.name,
            "->",
            target.name,
            "| SteamID:",
            sender.steamId,
            "| Convite:",
            invite.id,
            "| Prazo: 10 segundos"
        );

        return true;
    }

    // --------------------------------------------------
    // EXCLUIR CONVITES EXPIRADOS
    // --------------------------------------------------

    private cleanupInvitations(): void {

        const now = Date.now();

        for (
            const [id, invite] of
            this.invitations.entries()
        ) {

            if (now < invite.expiresAt) {
                continue;
            }

            this.invitations.delete(id);

            console.log(
                "[PARTY] Convite expirado:",
                id
            );
        }
    }

    // ==================================================
    // BUSCA DE PARTIDAS
    // ==================================================

    // --------------------------------------------------
    // LOCALIZAR SALAS PUBLICAS COMPATIVEIS
    // --------------------------------------------------
    //
    // Grupo de 1:
    // pode entrar em sala com ate 3 jogadores.
    //
    // Grupo de 2:
    // pode entrar em sala com ate 2 jogadores.
    //
    // Grupo de 3:
    // pode entrar em sala com ate 1 jogador.
    //
    // Grupo de 4:
    // somente sala vazia.
    // --------------------------------------------------

    private async findCompatibleRooms(
        partySize: number
    ): Promise<RoomCandidate[]> {

        const rooms = await matchMaker.query({
            name: "my_room",
            mode: "public",
            locked: false,
            private: false
        });

        const compatible: RoomCandidate[] = [];

        for (const room of rooms) {

            const clients = Number(
                room.clients ?? 0
            );

            const maxClients = Number(
                room.maxClients ?? MAX_GAME_PLAYERS
            );

            const capacity = Math.min(
                maxClients,
                MAX_GAME_PLAYERS
            );

            const available =
                capacity - clients;

            // Ignorar salas bloqueadas.
            if (room.locked) continue;

            // Ignorar salas privadas.
            if (room.private) continue;

            // Todos devem caber na mesma sala.
            if (available < partySize) continue;

            if (capacity < partySize) continue;

            compatible.push({

                roomId: room.roomId,

                clients,

                maxClients,

                locked: room.locked,

                private: room.private
            });
        }

        // Priorizar salas mais cheias,
        // desde que tenham vagas suficientes.

        compatible.sort(
            (a, b) => b.clients - a.clients
        );

        console.log(
            "[PARTY] Busca publica:",
            partySize,
            "jogadores no grupo |",
            compatible.length,
            "salas compativeis"
        );

        return compatible;
    }

    // OPCOES DE ENTRADA NA PARTIDA

    private gameOptions(
        member: Member,
        mode: GameMode
    ) {

        return {

            mode,

            playerName: member.name.slice(0, 32),

            steamId: member.steamId,

            selectedColor: member.color,

            selectedCharacter: member.character
        };
    }

    // --------------------------------------------------
    // RESERVAR VAGAS PARA O GRUPO
    // --------------------------------------------------
    //
    // Todas as reservas precisam estar prontas
    // antes que o servidor envie "party_launch".
    // --------------------------------------------------

    private async reserveParty(
        roomId: string,
        participants: Member[],
        mode: GameMode
    ): Promise<PartyReservation[]> {

        const reservations: PartyReservation[] = [];

        for (const member of participants) {

            const reservation = await matchMaker.joinById(
                roomId,
                this.gameOptions(member, mode)
            );

            reservations.push({

                member,

                json: JSON.stringify(
                    reservation
                )
            });
        }

        return reservations;
    }

    // --------------------------------------------------
    // PROCURAR SALA OU CRIAR UMA NOVA
    // --------------------------------------------------

    private async findAndReserve(
        participants: Member[],
        mode: GameMode
    ): Promise<{
        roomId: string;
        reservations: PartyReservation[];
    }> {

        const partySize = participants.length;

        if (
            partySize < 1 ||
            partySize > MAX_PARTY_MEMBERS
        ) {

            throw new Error(
                "Quantidade de jogadores invalida."
            );
        }

        // ----------------------------------------------
        // BUSCA PUBLICA
        // ----------------------------------------------

        if (mode === "public") {

            const candidates =
                await this.findCompatibleRooms(
                    partySize
                );

            for (const candidate of candidates) {

                // Consultar novamente, porque
                // a ocupacao pode ter mudado.

                const rooms = await matchMaker.query({
                    name: "my_room",
                    mode: "public",
                    roomId: candidate.roomId
                });

                const currentRoom = rooms[0];

                if (!currentRoom) continue;

                if (currentRoom.locked) continue;

                if (currentRoom.private) continue;

                const clients = Number(
                    currentRoom.clients ?? 0
                );

                const capacity = Math.min(
                    MAX_GAME_PLAYERS,
                    Number(
                        currentRoom.maxClients ??
                        MAX_GAME_PLAYERS
                    )
                );

                const available =
                    capacity - clients;

                if (available < partySize) {

                    console.log(
                        "[PARTY] Sala ignorada:",
                        candidate.roomId,
                        "| Vagas:",
                        available,
                        "| Grupo:",
                        partySize
                    );

                    continue;
                }

                try {

                    const reservations =
                        await this.reserveParty(
                            candidate.roomId,
                            participants,
                            mode
                        );

                    console.log(
                        "[PARTY] Sala publica encontrada:",
                        candidate.roomId,
                        "| Jogadores anteriores:",
                        clients,
                        "| Grupo:",
                        partySize,
                        "| Total esperado:",
                        clients + partySize
                    );

                    return {
                        roomId: candidate.roomId,
                        reservations
                    };

                } catch (error) {

                    // A sala pode ter sido ocupada
                    // por outros jogadores enquanto
                    // as reservas eram criadas.
                    //
                    // Nenhum party_launch e enviado
                    // para este conjunto incompleto.

                    console.warn(
                        "[PARTY] Falha ao reservar grupo em:",
                        candidate.roomId,
                        error
                    );
                }
            }
        }

        // ----------------------------------------------
        // NOVA PARTIDA
        // ----------------------------------------------
        //
        // Publica:
        // criar quando nenhuma sala existente
        // comportar o grupo inteiro.
        //
        // Privada:
        // criar sala nova para o grupo.
        // ----------------------------------------------

        const newRoom =
            await matchMaker.createRoom(
                "my_room",
                {
                    mode
                }
            );

        const reservations =
            await this.reserveParty(
                newRoom.roomId,
                participants,
                mode
            );

        console.log(
            "[PARTY] Nova partida criada:",
            newRoom.roomId,
            "| Modalidade:",
            mode,
            "| Integrantes do grupo:",
            partySize
        );

        return {

            roomId: newRoom.roomId,

            reservations
        };
    }

    // --------------------------------------------------
    // INICIAR PARTIDA COM GRUPO
    // --------------------------------------------------
    //
    // Esta e a unica funcao que inicia
    // matchmaking de partida.
    //
    // O convite "party_invite" NAO chama
    // esta funcao.
    // --------------------------------------------------

    private async launch(
        party: Party,
        mode: GameMode
    ): Promise<void> {

        if (party.launching) return;

        party.launching = true;

        try {

            const memberKeys = [
                ...party.members
            ];

            const users = memberKeys.map(
                key => this.memberByKey(key)
            );

            if (
                users.some(
                    member => member === undefined
                )
            ) {

                throw new Error(
                    "Um integrante do grupo desconectou."
                );
            }

            const participants =
                users as Member[];

            // ------------------------------------------
            // VALIDAR QUANTIDADE
            // ------------------------------------------

            if (
                participants.length < 1 ||
                participants.length > MAX_PARTY_MEMBERS
            ) {

                throw new Error(
                    "O grupo precisa ter de 1 a 4 jogadores."
                );
            }

            console.log(
                "[PARTY] Iniciando busca:",
                mode,
                "| Grupo:",
                party.id,
                "| Integrantes:",
                participants.length
            );

            // ------------------------------------------
            // LOCALIZAR SALA E RESERVAR
            // ------------------------------------------

            const result =
                await this.findAndReserve(
                    participants,
                    mode
                );

            // ------------------------------------------
            // VERIFICAR SE O GRUPO MUDOU
            // ------------------------------------------

            if (
                party.members.length !== memberKeys.length ||
                party.members.some(
                    (key, index) =>
                        key !== memberKeys[index]
                )
            ) {

                throw new Error(
                    "O grupo mudou durante a busca."
                );
            }

            // ------------------------------------------
            // CONFERIR CONEXOES
            // ------------------------------------------

            for (const member of participants) {

                if (!this.clientOf(member.key)) {

                    throw new Error(
                        "Um integrante desconectou " +
                        "durante a busca."
                    );
                }
            }

            // ------------------------------------------
            // ENVIAR ENTRADA A TODOS
            // ------------------------------------------

            for (const item of result.reservations) {

                const client = this.clientOf(
                    item.member.key
                );

                if (!client) continue;

                client.send(
                    "party_launch",
                    {

                        reservationJson: item.json,

                        roomId: result.roomId,

                        mode
                    }
                );
            }

            console.log(
                "[PARTY] Grupo enviado para partida:",
                party.id,
                "| Jogadores:",
                participants.length,
                "| Sala:",
                result.roomId
            );

        } catch (error) {

            console.error(
                "[PARTY] Erro ao iniciar partida:",
                error
            );

            const message =
                error instanceof Error
                    ? error.message
                    : "Erro ao iniciar partida.";

            for (const key of party.members) {

                const client =
                    this.clientOf(key);

                client?.send(
                    "party_error",
                    {
                        message
                    }
                );
            }

        } finally {

            // Liberar nova tentativa caso
            // todos voltem ao menu.
            party.launching = false;
        }
    }

    // ==================================================
    // EVENTOS DA PARTYROOM
    // ==================================================

    onCreate() {

        console.log(
            "[PARTY] Sistema de grupos iniciado."
        );

        // --------------------------------------------------
        // LIMPAR CONVITES EXPIRADOS
        // --------------------------------------------------

        this.clock.setInterval(
            () => {
                this.cleanupInvitations();
            },
            1000
        );

        // --------------------------------------------------
        // ATUALIZAR PERFIL DO JOGADOR
        // --------------------------------------------------

        this.onMessage(
            "party_profile",
            (client, data) => {

                const member =
                    this.memberOf(client);

                if (!member) return;

                const color = String(
                    data?.color ?? "red"
                );

                member.color = [
                    "red",
                    "blue",
                    "green",
                    "yellow"
                ].includes(color)
                    ? color
                    : "red";

                const character = Number(
                    data?.character ?? 0
                );

                member.character =
                    Number.isInteger(character)
                        ? Math.max(
                            0,
                            Math.min(3, character)
                        )
                        : 0;
            }
        );

        // --------------------------------------------------
        // CONVIDAR JOGADOR PARA O GRUPO
        // --------------------------------------------------
        //
        // Nao cria sala publica.
        // Nao cria sala privada.
        // Apenas envia um convite.
        // --------------------------------------------------

        this.onMessage(
            "party_invite",
            (client, data) => {

                const sender =
                    this.memberOf(client);

                if (!sender) return;

                const party =
                    this.partyOf(sender.key);

                if (!party) return;

                // ------------------------------------------
                // VALIDAR GRUPO
                // ------------------------------------------

                if (party.launching) {

                    this.sendError(
                        client,
                        "O grupo esta iniciando uma partida."
                    );

                    return;
                }

                if (party.leader !== sender.key) {

                    this.sendError(
                        client,
                        "Somente o lider pode convidar."
                    );

                    return;
                }

                if (
                    party.members.length >=
                    MAX_PARTY_MEMBERS
                ) {

                    this.sendError(
                        client,
                        "O grupo ja esta cheio."
                    );

                    return;
                }

                // ------------------------------------------
                // IDENTIFICAR DESTINATARIO
                // ------------------------------------------

                const rawId = String(
                    data?.targetId ?? ""
                ).trim();

                let targetSteamId = rawId;
                let targetInstanceId = "";

                // Conta simulada de desenvolvimento:
                // dev:SteamID:CodigoDaInstancia

                if (rawId.startsWith("dev:")) {

                    const parts =
                        rawId.split(":");

                    targetSteamId =
                        parts[1] ?? "";

                    targetInstanceId =
                        parts[2] ?? "";
                }

                if (
                    !/^\d{17}$/.test(targetSteamId)
                ) {

                    this.sendError(
                        client,
                        "Identidade do amigo invalida."
                    );

                    return;
                }

                if (
                    targetSteamId === sender.steamId &&
                    (
                        targetInstanceId === "" ||
                        targetInstanceId ===
                        sender.instanceId
                    )
                ) {

                    this.sendError(
                        client,
                        "Voce nao pode convidar " +
                        "sua propria janela."
                    );

                    return;
                }

                // ------------------------------------------
                // VERIFICAR SE JA ESTA NO GRUPO
                // ------------------------------------------

                const targetKey = targetInstanceId !== ""
                    ? this.keyFor(
                        targetSteamId,
                        targetInstanceId
                    )
                    : "";

                if (
                    targetKey !== "" &&
                    party.members.includes(targetKey)
                ) {

                    this.sendError(
                        client,
                        "Esse jogador ja esta no grupo."
                    );

                    return;
                }

                // ------------------------------------------
                // EVITAR CONVITES DUPLICADOS
                // ------------------------------------------

                for (
                    const invite of
                    this.invitations.values()
                ) {

                    if (
                        invite.partyId === party.id &&
                        invite.targetSteamId === targetSteamId &&
                        invite.targetInstanceId ===
                        targetInstanceId &&
                        Date.now() < invite.expiresAt
                    ) {

                        this.sendError(
                            client,
                            "Ja existe um convite pendente " +
                            "para esse jogador."
                        );

                        return;
                    }
                }

                // ------------------------------------------
                // REGISTRAR CONVITE
                // ------------------------------------------

                const invite: Invitation = {

                    id: randomUUID(),

                    partyId: party.id,

                    senderKey: sender.key,

                    targetSteamId,

                    targetInstanceId,

                    expiresAt:
                        Date.now() +
                        OFFLINE_INVITATION_DURATION,

                    delivered: false
                };

                this.invitations.set(
                    invite.id,
                    invite
                );

                // ------------------------------------------
                // TENTAR ENTREGAR AGORA
                // ------------------------------------------

                const delivered =
                    this.deliver(invite);

                // ------------------------------------------
                // RESPONDER AO REMETENTE
                // ------------------------------------------

                client.send(
                    "party_invite_result",
                    {

                        ok: true,

                        delivered,

                        invitationId: invite.id,

                        partyId: party.id
                    }
                );

                console.log(
                    "[PARTY] Convite criado:",
                    invite.id,
                    "| De:",
                    sender.name,
                    "| Para SteamID:",
                    targetSteamId,
                    "| Instancia:",
                    targetInstanceId || "qualquer",
                    "| Entregue:",
                    delivered
                );
            }
        );

        // --------------------------------------------------
        // ACEITAR OU RECUSAR CONVITE
        // --------------------------------------------------

        this.onMessage(
            "party_reply",
            (client, data) => {

                const member =
                    this.memberOf(client);

                if (!member) return;

                const invitationId = String(
                    data?.invitationId ?? ""
                );

                const accept =

                    invite.targetInstanceId !== "" &&
                    invite.targetInstanceId !==

                        "Este convite pertence a outra instancia."
                    );

                    return;

                // ------------------------------------------

                if (
                    !invite.delivered ||
                    Date.now() >= invite.expiresAt
                ) {

                    this.invitations.delete(
                        invitationId
                    );

                    this.sendError(
                        client,
                        "O convite expirou."
                    );

                    return;
                }

                // RECUSAR

                if (!accept) {

                    this.invitations.delete(
                        invitationId

                        "party_reply_result",

                            accepted: false,

                        invitationId
                    );

                    return;

                // ACEITAR

                    this.parties.get(
                        invite.partyId

                    this.invitations.delete(
                        invitationId
                    );

                    this.sendError(
                        client,
                        "O grupo nao existe mais."

                        "O grupo ja esta entrando numa partida."
                    );

                    return;

                    party.members.length >=

                        client,
                        "O grupo esta cheio."

                    this.memberByKey(

                    );

                    invitationId
                );

                // Remover do grupo anterior.
                this.detach(member.key);

                    party.members.push(
                        member.key
                    );

                this.publish(party);

                    "party_reply_result",
                    {
                        ok: true,
                        accepted: true,
                        invitationId,
                        partyId: party.id
                    }
                );

                    "[PARTY] CONVITE ACEITO:",
                    member.name,
                    "| Grupo:",
                    party.id,
                    "| Integrantes:",
                    party.members.length

            "party_leave",
            client => {

                const member =
                    this.memberOf(client);

                if (!member) return;

                this.detach(member.key);

                this.createSolo(member);

                console.log(
                    "[PARTY] Jogador saiu do grupo:",
                    member.name
                );
            }
        );

        // INICIAR PARTIDA
        // --------------------------------------------------
        //
        // Somente o lider pode executar.
        //
        // Este evento e separado de party_invite.
        // --------------------------------------------------

        this.onMessage(
            "party_start",
            (client, data) => {

                const member =
                    this.memberOf(client);

                if (!member) return;

                const party =
                    this.partyOf(member.key);

                if (!party) return;

                if (
                    party.leader !==
                    member.key
                ) {

                    this.sendError(
                        client,
                        "Somente o lider pode iniciar a partida."
                    );

                    return;
                }

                if (party.launching) return;

                const mode: GameMode =
                    data?.mode === "private"
                        ? "private"
                        : "public";

                console.log(
                    "[PARTY] Lider iniciou partida:",
                    member.name,
                    "| Modalidade:",
                    mode,
                    "| Integrantes:",
                    party.members.length
                );

                void this.launch(
                    party,
                    mode
                );
            }
        );
    }

    onJoin(

            );

            return;
        }

            "Jogador"
        ).slice(0, 32);

        const key = this.keyFor(
            steamId,
            instanceId
        );

        // ----------------------------------------------
        // EVITAR DUPLICACAO
        // ----------------------------------------------

        for (

            this.members.values()
        ) {

            if (existing.key === key) {

                this.detach(

                this.members.delete(
                    existing.sessionId
                );
            }
        }

        // ----------------------------------------------
        // CRIAR REGISTRO
        // ----------------------------------------------

        const member: Member = {

            steamId,

            color: "red",

            character: 0,

            sessionId: client.sessionId
        };

            "[PARTY] Jogador conectado:",
            name,
            "| SteamID:",
            steamId,

        // ENTREGAR CONVITES OFFLINE
        // ----------------------------------------------
        //
        // Caso um amigo abra o jogo depois

            }

            if (
                invite.targetInstanceId !== "" &&
                invite.targetInstanceId !==

            ) {
                continue;
            }

            this.deliver(invite);
        }
    }

        );

            "[PARTY] Jogador desconectado:",

            member.key
        );
    }

    // --------------------------------------------------
    // INICIAR PRAZO DA ROLAGEM EXTRA
    // --------------------------------------------------

    private startExtraRollTimer(sessionId: string) {
        if (this.state.status !== "playing") {
            return;
        }

        if (this.state.currentTurn !== sessionId) {
            return;
        }

        const player = this.state.players.get(sessionId);

        if (!player || !player.extraRollAvailable) {
            return;
        }

        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.state.currentTurn !== sessionId) {
                return;
            }

            const currentPlayer = this.state.players.get(sessionId);

            if (
                currentPlayer?.extraRollAvailable &&
                !this.state.hasRolled
            ) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a rolagem extra.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);
    }


 

    // --------------------------------------------------
    // EXIBIR ANÚNCIO DA ROLAGEM EXTRA
    // --------------------------------------------------

    private showExtraRollAnnouncement(
        sessionId: string,
        reason: ExtraRollReason
    ) {

        const player = this.state.players.get(sessionId);

        if (!player) {
            return;
        }

        this.extraRollMessageDelay?.clear();

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;

        // --------------------------------------------------
        // FINALIZAR ROLAGEM ANTERIOR APÓS A CHEGADA
        // --------------------------------------------------

        this.state.hasRolled = false;
        this.state.lastRollBy = "";

        // Preservar dice1 e dice2 para o popup de duplas.
        // Somente uma nova rolagem substituirá esses valores.

        this.state.extraRollPlayerSessionId = sessionId;
        this.state.extraRollPlayerName = player.name;
        this.state.extraRollReason = reason;
        this.state.showExtraRollMessage = true;

        console.log("==================================");
        console.log("ROLAGEM EXTRA CONCEDIDA!");
        console.log("JOGADOR:", player.name);
        console.log("SESSION ID:", sessionId);
        console.log(
            "MOTIVO:",
            reason === "start" ? "CASA INICIAL" : "DUPLAS"
        );
        console.log("==================================");

        this.extraRollMessageDelay = this.clock.setTimeout(() => {

            this.extraRollMessageDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            this.state.showExtraRollMessage = false;
            this.state.extraRollPlayerName = "";
            this.state.extraRollPlayerSessionId = "";
            this.state.extraRollReason = "";

            console.log(
                "Popup da rolagem extra encerrado. Liberando a partida."
            );

            this.startExtraRollTimer(sessionId);

        }, EXTRA_ROLL_MESSAGE_DURATION);
    }




    // --------------------------------------------------
    // GERAR CÓDIGO DE 6 CARACTERES
    // --------------------------------------------------

    private generateCode(): string {
        let code = "";

        for (let i = 0; i < 6; i++) {
            const index = Math.floor(
                Math.random() * this.CODE_CHARACTERS.length
            );

            code += this.CODE_CHARACTERS[index];
        }

        return code;
    }

    // --------------------------------------------------
    // GARANTIR CÓDIGO ÚNICO
    // --------------------------------------------------

    private async generateRoomCode(): Promise<string> {
        const existingCodes = await this.presence.smembers(
            this.CODE_CHANNEL
        );

        let code = "";

        do {
            code = this.generateCode();
        } while (existingCodes.includes(code));

        await this.presence.sadd(this.CODE_CHANNEL, code);

        return code;
    }

    // --------------------------------------------------
    // CANCELAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private clearAutoStartCountdown() {
        this.autoStartInterval?.clear();
        this.autoStartInterval = undefined;

        this.state.autoStartCountdown = 0;
    }

    // --------------------------------------------------
    // REINICIAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private resetAutoStartCountdown() {
        if (this.mode !== "public") {
            this.clearAutoStartCountdown();
            return;
        }

        if (this.state.status !== "waiting") {
            this.clearAutoStartCountdown();
            return;
        }

        this.clearAutoStartCountdown();

        if (this.state.players.size < 2) {
            return;
        }

        this.state.autoStartCountdown = AUTO_START_DURATION;

        console.log(
            `Contagem automática iniciada: ${this.roomId} (${AUTO_START_DURATION}s)`
        );

        this.autoStartInterval = this.clock.setInterval(() => {
            if (this.state.status !== "waiting") {
                this.clearAutoStartCountdown();
                return;
            }

            if (this.state.players.size < 2) {
                this.clearAutoStartCountdown();
                return;
            }

            this.state.autoStartCountdown--;

            console.log(
                `Sala ${this.roomId}: ${this.state.autoStartCountdown}s`
            );

            if (this.state.autoStartCountdown <= 0) {
                this.clearAutoStartCountdown();
                this.startGame();
            }
        }, 1000);
    }

    // INICIAR PARTIDA
    // --------------------------------------------------

    private startGame() {
        if (this.state.status !== "waiting") {
            return;
        }

        if (this.state.players.size < 2) {
            return;
        }

        this.clearAutoStartCountdown();

        this.randomizeSpecialCities();
        this.broadcastBoard();

        this.state.status = "playing";

        this.state.currentTurn = "";
        this.state.turnDeadline = 0;
        this.state.turnCount = 0;

        this.state.hasRolled = false;
        this.state.hasMoved = false;

        this.state.dice1 = 0;
        this.state.dice2 = 0;
        this.state.lastRollBy = "";

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        this.playersResetForNewGame();

        void this.setMetadata({
            mode: this.mode,
            status: this.state.status
        });

        this.lock();

        console.log("Partida iniciada:", this.roomId);
        console.log("Aguardando 3 segundos para iniciar o anúncio...");

        this.startDelay?.clear();

        this.startDelay = this.clock.setTimeout(() => {
            this.startDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            console.log("Animação concluída! Sorteando primeiro jogador.");

            this.showFirstPlayerAnnouncement();
        }, GAME_START_DELAY);
    }

            const rolledDoubles = dice1 === dice2;

            const steps = dice1 + dice2;

            const startingPosition = player.position;
            const newPosition = startingPosition + steps;

            const completedLaps = Math.floor(
                newPosition / BOARD_SIZE
            );

            const landedPosition = newPosition % BOARD_SIZE;

            player.position = landedPosition;
            player.laps += completedLaps;

            const landedCell = this.board[landedPosition];

            player.lastCellType = landedCell.type;
            player.lastCellName = landedCell.name;

            this.state.hasMoved = true;

            console.log(

            // --------------------------------------------------

            if (completedLaps > 0) {

                console.log(
                    `${player.name} completou ${completedLaps} volta(s). Total: ${player.laps}`
                );

                const totalMoneyReward =
                    START_CELL_MONEY_REWARD * completedLaps;

                player.money += totalMoneyReward;

                console.log(
                    `${player.name} recebeu $${totalMoneyReward.toLocaleString("pt-BR")} por completar ${completedLaps} volta(s).`
                );
            }

            // --------------------------------------------------
            // VERIFICAR SE CAIU NA PRISÃO
            // --------------------------------------------------

            const landedInPrison =
                landedCell.name === "Prisão" &&
                landedCell.type === "island";

            if (landedInPrison) {

                // A prisão será ativada após a chegada.
                // Enquanto caminha, manter a exibição
                // do resultado dos dados.

                player.prisonAttempts = 0;
                player.extraRollAvailable = false;

                    `${player.name} foi enviado para a prisão. O turno será encerrado após o anúncio.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "prison"
                );

                return;
            }

            // --------------------------------------------------
            // CASA INICIAL
            // --------------------------------------------------

            const landedExactlyOnStart =
                landedPosition === 0 &&
                landedCell.type === "start" &&
                landedCell.name === "Início";

            const receivesStartExtraRoll =
                landedExactlyOnStart &&
                completedLaps > 0 &&
                player.laps >= 1;

            let extraRollReason: ExtraRollReason | undefined;

            if (receivesStartExtraRoll) {

                extraRollReason = "start";

            } else if (rolledDoubles) {

                extraRollReason = "doubles";
            }

            const canReceiveExtraRoll =
                extraRollReason !== undefined;

            console.log(
                `[VERIFICAÇÃO ROLAGEM EXTRA] ${player.name} | ` +
                `posição inicial: ${startingPosition} | ` +
                `dados: ${dice1} + ${dice2} | ` +
                `duplas: ${rolledDoubles} | ` +
                `posição final: ${landedPosition} | ` +
                `voltas: ${player.laps} | ` +
                `terminou no início: ${receivesStartExtraRoll} | ` +
                `motivo: ${extraRollReason ?? "nenhum"} | ` +
                `recebe extra: ${canReceiveExtraRoll}`
            );

            // --------------------------------------------------
            // ROLAGEM EXTRA
            // --------------------------------------------------

            if (canReceiveExtraRoll && extraRollReason) {

                player.extraRollAvailable = true;

                this.state.hasMoved = true;

                // Preservar hasRolled e lastRollBy
                // até a chegada do personagem.
                //
                // Isso mantém o resultado visível
                // para todos os clientes.

                console.log(
                    `${player.name} recebeu uma rolagem extra pelo motivo: ${extraRollReason}.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "extra",
                    extraRollReason
                );

            } else {

                // --------------------------------------------------
                // MOVIMENTO NORMAL
                // --------------------------------------------------

                // Mesmo sem popup, aguardar a animação
                // antes de liberar as ações da HUD.

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "none"
                );
            }
        },


        // --------------------------------------------------
        // ENCERRAR TURNO
        // --------------------------------------------------

        end_turn: (client: Client) => {

            const player = this.state.players.get(client.sessionId);

            console.log("==================================");
            console.log("[END_TURN] SOLICITACAO RECEBIDA");
            console.log("[END_TURN] Cliente:", client.sessionId);
            console.log("[END_TURN] Jogador:", player?.name ?? "Nao encontrado");
            console.log("[END_TURN] Status:", this.state.status);
            console.log("[END_TURN] Turno atual:", this.state.currentTurn);
            console.log("[END_TURN] HasRolled:", this.state.hasRolled);
            console.log("[END_TURN] HasMoved:", this.state.hasMoved);
            console.log("[END_TURN] Primeiro jogador popup:", this.state.showFirstPlayerMessage);
            console.log("[END_TURN] Rolagem extra popup:", this.state.showExtraRollMessage);
            console.log("[END_TURN] Prisao popup:", this.state.showPrisonMessage);
            console.log("[END_TURN] Preso:", player?.inPrison);
            console.log("[END_TURN] Rolagem extra:", player?.extraRollAvailable);
            console.log("[END_TURN] Jogadores na sala:", this.state.players.size);

            // --------------------------------------------------
            // VERIFICAR PARTIDA
            // --------------------------------------------------

            if (this.state.status !== "playing") {

                console.log("[END_TURN] BLOQUEADO: partida nao esta em andamento.");

                client.send(
                    "action_error",
                    "A partida nao esta em andamento."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR POPUPS
            // --------------------------------------------------

            if (this.isPopupActive()) {

                console.log("[END_TURN] BLOQUEADO: popup ativo no servidor.");

                client.send(
                    "action_error",
                    "Aguarde o encerramento do popup."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR JOGADOR
            // --------------------------------------------------

            if (!player) {

                console.log("[END_TURN] BLOQUEADO: jogador nao encontrado.");

                client.send(
                    "action_error",
                    "Jogador nao encontrado."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR TURNO
            // --------------------------------------------------

            if (this.state.currentTurn !== client.sessionId) {

                console.log("[END_TURN] BLOQUEADO: nao e o turno deste cliente.");

                client.send(
                    "action_error",
                    "Nao e seu turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR MOVIMENTO
            // --------------------------------------------------

            if (!this.state.hasMoved) {

                console.log("[END_TURN] BLOQUEADO: movimento nao concluido.");

                client.send(
                    "action_error",
                    "Voce precisa movimentar antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR PRISAO
            // --------------------------------------------------

            if (player.inPrison) {

                console.log("[END_TURN] BLOQUEADO: jogador esta preso.");

                client.send(
                    "action_error",
                    "Voce ainda esta na prisao."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR ROLAGEM EXTRA
            // --------------------------------------------------

            if (player.extraRollAvailable) {

                console.log("[END_TURN] BLOQUEADO: rolagem extra pendente.");

                client.send(
                    "action_error",
                    "Utilize sua rolagem extra antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // ENCERRAR TURNO
            // --------------------------------------------------

            console.log("[END_TURN] AUTORIZADO!");
            console.log("[END_TURN] Encerrando turno de:", player.name);

            const previousTurn = this.state.currentTurn;

            this.nextTurn();

            console.log("[END_TURN] Turno anterior:", previousTurn);
            console.log("[END_TURN] Novo turno:", this.state.currentTurn);
            console.log("==================================");
        },


        play: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                return;
            }

            player.score++;
            this.nextTurn();
        },
    };

    async onCreate(options: any) {
        this.roomId = await this.generateRoomCode();

        this.mode = options.mode === "private" ? "private" : "public";

        this.state.status = "waiting";
        this.state.autoStartCountdown = 0;

        this.state.movementAnnouncementPending = false;
        this.pendingMovementAnnouncement = undefined;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        await this.setMatchmaking({
            metadata: {
                mode: this.mode,
                status: this.state.status
            },
            private: this.mode === "private"
        });

        console.log("==================================");
        console.log("SALA CRIADA:", this.roomId);
        console.log("MODO:", this.mode);
        console.log("PRIVADA:", this.mode === "private");
        console.log(
            "LISTAGEM:",
            this.mode === "public" ? "PUBLICA" : "OCULTA"
        );
        console.log("==================================");

        console.log("Tabuleiro carregado:", this.board.length, "casas.");

        if (this.board.length !== BOARD_SIZE) {
            throw new Error(
                `Tabuleiro inválido: esperado ${BOARD_SIZE}, encontrado ${this.board.length}.`
            );
        }
    }

    onAuth(client: Client, options: any) {

        if (this.state.status !== "waiting") {
            throw new ServerError(409, "Esta partida já começou.");
        }

        if (options?.mode !== this.mode) {
            throw new ServerError(
                403,
                "Esta sala não pertence a essa modalidade de partida."
            );
        }

        // --------------------------------------------------
        // VALIDAR NOME STEAM
        // --------------------------------------------------

        if (
            typeof options?.playerName !== "string"
        ) {
            throw new ServerError(400, "Nome Steam inválido.");
        }

        const playerName = String(options.playerName).trim();

        if (playerName.length === 0 || playerName.length > 32) {
            throw new ServerError(
                400,
                "O nome Steam deve possuir entre 1 e 32 caracteres."
            );
        }

        // --------------------------------------------------
        // VALIDAR STEAM ID
        // --------------------------------------------------

        const steamId = String(options?.steamId ?? "").trim();

        if (!/^\d{17}$/.test(steamId)) {
            throw new ServerError(
                400,
                "Steam ID inválido ou não recebido."
            );
        }

        // --------------------------------------------------
        // VALIDAR PERSONAGEM
        // --------------------------------------------------

        const selectedCharacter = Number(
            options?.selectedCharacter ?? 0
        );

        const validCharacter =
            Number.isInteger(selectedCharacter) &&
            selectedCharacter >= 0 &&
            selectedCharacter <= 3;

        if (!validCharacter) {
            throw new ServerError(400, "Personagem inválido.");
        }

        // --------------------------------------------------
        // VALIDAR COR
        // --------------------------------------------------

        const preferredColor =
            typeof options?.selectedColor === "string" &&
            this.isValidColor(options.selectedColor)
                ? options.selectedColor
                : "red";

        return {
            playerName,
            steamId,
            preferredColor,
            selectedCharacter
        };
    }

    onJoin(client: Client, options: any, auth: any) {

        const player = new Player();

        // Identidade Steam.
        player.name = auth.playerName;
        player.steamId = auth.steamId;

        player.money = INITIAL_PLAYER_MONEY;
        player.character = auth.selectedCharacter ?? 0;

        player.position = 0;
        player.laps = 0;
        player.extraRollAvailable = false;

        player.inPrison = false;
        player.prisonAttempts = 0;
        player.debtsAvailable = 1;

        player.lastCellType = this.board[0].type;
        player.lastCellName = this.board[0].name;

        player.color = this.assignPlayerColor(
            auth.preferredColor ?? "red"
        );

        this.state.players.set(client.sessionId, player);

        this.playerJoinOrder.push(client.sessionId);

        if (this.state.hostSessionId === "") {
            this.state.hostSessionId = client.sessionId;
        }

        console.log("==================================");
        console.log(`${player.name} entrou na sala ${this.roomId}!`);
        console.log("STEAM ID:", player.steamId);
        console.log("SESSION ID:", client.sessionId);
        console.log("Cor atribuída:", player.color);
        console.log("Personagem escolhido:", player.character);
        console.log("Dinheiro inicial:", player.money);
        console.log("Dívidas disponíveis:", player.debtsAvailable);

        console.log(
            "Ordem de entrada:",
            this.playerJoinOrder.join(" -> ")
        );

        console.log("==================================");

        this.resetAutoStartCountdown();
    }

    onLeave(client: Client, code: CloseCode) {
        console.log(client.sessionId, "saiu da sala!", code);

        const wasTheirTurn =
            this.state.currentTurn === client.sessionId;

        const wasHost =
            this.state.hostSessionId === client.sessionId;

        const wasSelectedFirstPlayer =
            this.state.firstPlayerSessionId === client.sessionId;

        const wasExtraRollPlayer =
            this.state.extraRollPlayerSessionId === client.sessionId;

        const wasPrisonMessagePlayer =
            this.state.prisonPlayerSessionId === client.sessionId;

        this.state.players.delete(client.sessionId);

        this.playerJoinOrder = this.playerJoinOrder.filter(
            sessionId => sessionId !== client.sessionId
        );

        // Atualizar a espera de chegada caso alguém saia da sala.
        if (this.pendingMovementAnnouncement) {
            if (
                this.pendingMovementAnnouncement.sessionId ===
                client.sessionId
            ) {
                this.pendingMovementAnnouncement = undefined;
                this.state.movementAnnouncementPending = false;
            }
            else {
                this.pendingMovementAnnouncement.arrivedClients.delete(
                    client.sessionId
                );

                this.finishMovementAnnouncementIfReady(
                    client.sessionId
                );
            }
        }

        if (
            this.state.showFirstPlayerMessage &&
            wasSelectedFirstPlayer
        ) {
            const newFirstPlayer = this.chooseRandomFirstPlayer();

            if (newFirstPlayer !== "") {
                this.updateFirstPlayerAnnouncement(newFirstPlayer);
            } else {
                this.state.firstPlayerSessionId = "";
                this.state.firstPlayerName = "";
            }
        }

        if (
            this.state.showExtraRollMessage &&
            wasExtraRollPlayer
        ) {
            this.extraRollMessageDelay?.clear();
            this.extraRollMessageDelay = undefined;

            this.state.showExtraRollMessage = false;
            this.state.extraRollPlayerName = "";
            this.state.extraRollPlayerSessionId = "";
            this.state.extraRollReason = "";

            if (this.state.status === "playing" && wasTheirTurn) {
                this.nextTurn();
            }
        }

        if (
            this.state.showPrisonMessage &&
            wasPrisonMessagePlayer
        ) {
            this.prisonMessageDelay?.clear();
            this.prisonMessageDelay = undefined;

            this.state.showPrisonMessage = false;
            this.state.prisonPlayerName = "";
            this.state.prisonPlayerSessionId = "";
        }

        if (
            this.state.status === "waiting" &&
            wasHost
        ) {
            const nextHost = this.playerJoinOrder.find(
                sessionId => this.state.players.has(sessionId)
            );

            this.state.hostSessionId = nextHost ?? "";

            if (nextHost) {
                const nextHostPlayer = this.state.players.get(nextHost);

                console.log("==================================");
                console.log("ANFITRIÃO ANTERIOR SAIU:", client.sessionId);
                console.log("NOVA LIDERANÇA TRANSFERIDA!");
                console.log("NOVO ANFITRIÃO:", nextHost);
                console.log(
                    "JOGADOR:",
                    nextHostPlayer?.name ?? "Desconhecido"
                );
                console.log("==================================");
            } else {
                console.log(
                    "O anfitrião saiu e não existem jogadores restantes."
                );
            }
        }

        this.resetAutoStartCountdown();

        if (
            this.state.status === "playing" &&
            wasTheirTurn &&
            !(
                this.state.showExtraRollMessage &&
                wasExtraRollPlayer
            )
        ) {
            this.nextTurn();
        }
    }

    async onDispose() {
        this.turnTimeout?.clear();
        this.autoStartInterval?.clear();
        this.startDelay?.clear();
        this.firstPlayerMessageDelay?.clear();
        this.extraRollMessageDelay?.clear();
        this.prisonMessageDelay?.clear();
        this.prisonAttemptDelay?.clear();

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        await this.presence.srem(
            this.CODE_CHANNEL,
            this.roomId
        );

        console.log("Sala encerrada:", this.roomId);
    }

    // --------------------------------------------------
    // PRÓXIMO TURNO
    // --------------------------------------------------

    nextTurn(firstTurnSessionId?: string) {
        this.turnTimeout?.clear();
        this.prisonAttemptDelay?.clear();
        this.prisonAttemptDelay = undefined;

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        if (this.state.status !== "playing") {
            return;
        }

        const previousPlayer = this.state.players.get(
            this.state.currentTurn
        );

        if (previousPlayer) {
            previousPlayer.extraRollAvailable = false;
            previousPlayer.consecutiveDoubles = 0;
        }

        const sessionIds = [...this.state.players.keys()];

        if (sessionIds.length === 0) {
            this.state.currentTurn = "";
            this.state.turnDeadline = 0;
            return;
        }

        if (
            firstTurnSessionId &&
            sessionIds.includes(firstTurnSessionId)
        ) {
            this.state.currentTurn = firstTurnSessionId;
        } else {
            const previous = sessionIds.indexOf(this.state.currentTurn);

            this.state.currentTurn =
                sessionIds[(previous + 1) % sessionIds.length];
        }

        this.state.turnCount++;

        this.state.hasRolled = false;
        this.state.hasMoved = false;
        this.state.lastRollBy = "";
        this.state.dice1 = 0;
        this.state.dice2 = 0;

        const turnSessionId = this.state.currentTurn;
        const currentPlayer = this.state.players.get(turnSessionId);

        console.log("==================================");
        console.log("NOVO TURNO:", this.state.turnCount);
        console.log("JOGADOR:", currentPlayer?.name ?? "Desconhecido");
        console.log("SESSION ID:", turnSessionId);

        if (currentPlayer?.inPrison) {
            console.log(
                "JOGADOR ESTÁ PRESO: aguardando escolha para tentar escapar."
            );
        }

        console.log("==================================");

        this.startCurrentTurnTimer(turnSessionId);
    }
