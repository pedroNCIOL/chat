    var _numero = string(floor(_valor));
    var _resultado = "";
    var _contador = 0;

    for (var _i = string_length(_numero); _i >= 1; _i--) {
        _resultado = string_char_at(_numero, _i) + _resultado;
        _contador++;

        if (_contador mod 3 == 0 && _i > 1) {
            _resultado = "." + _resultado;
        }
    }

    return _resultado;
};

partida_centro_casa = function(_indice) {
    var _cam_x = camera_get_view_width(view_camera[0]);
    var _cam_y = camera_get_view_height(view_camera[0]);

    var _mesa_x = -45;
    var _mesa_y = 140;
    var _escala = 1.60;

    var _dist_x = 73 * (_escala / 1.75);
    var _dist_y = 37 * (_escala / 1.75);

    var _pos_x = 0;
    var _pos_y = 0;

    if (_indice == 0) {
        _pos_x = _cam_x / 2 + _mesa_x;
        _pos_y = _cam_y / 2 + _mesa_y + 100;
    }
    else if (_indice <= 16) {
        if (_indice < 9) {
            var _xx = _indice;
            var _yy = _indice;

            _pos_x = _cam_x / 2 - _xx * _dist_x + _mesa_x;
            _pos_y = _cam_y / 2 + 100 - _yy * _dist_y + _mesa_y;
        }
        else {
            var _xx = _indice - 16;
            var _yy = _indice;

            _pos_x = _cam_x / 2 + _xx * _dist_x + _mesa_x;
            _pos_y = _cam_y / 2 + 100 - _yy * _dist_y + _mesa_y;
        }
    }
    else if (_indice < 25) {
        var _xx = _indice - 16;
        var _yy = 32 - _indice;

        _pos_x = _cam_x / 2 + _xx * _dist_x + _mesa_x;
        _pos_y = _cam_y / 2 + 100 - _yy * _dist_y + _mesa_y;
    }
    else {
        var _xx = _indice - 32;
        var _yy = 32 - _indice;

        _pos_x = _cam_x / 2 - _xx * _dist_x + _mesa_x;
        _pos_y = _cam_y / 2 + 100 - _yy * _dist_y + _mesa_y;
    }

    var _block_offset = 0;
    var _mapa = instance_find(obj_jogo, 0);

    if (_mapa != noone) {
        if (_indice < array_length(_mapa.block_offset)) {
            _block_offset = _mapa.block_offset[_indice];
        }
    }

    _pos_y -= _block_offset;
    _pos_y += 10;

    return {
        x: _pos_x,
        y: _pos_y
    };
};
