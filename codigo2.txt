menu_amigos_gerar_codigo = function(_steam_id) {

    var _identidade = steam_user_decode_steam_id(_steam_id);

    if (!is_struct(_identidade)) return "------";

    if (!variable_struct_exists(_identidade, "account_id")) {
        return "------";
    }

    if (variable_struct_exists(_identidade, "is_valid")) {
        if (!_identidade.is_valid) return "------";
    }

    var _numero = floor(_identidade.account_id);

    if (_numero < 0 || _numero > 4294967295) {
        return "------";
    }

    var _alfabeto =
        "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

    var _codigo = "";

    repeat (6) {
        var _digito = _numero mod 62;

        _codigo =
            string_char_at(_alfabeto, _digito + 1) + _codigo;

        _numero = floor(_numero / 62);
    }

    return _codigo;
};

// --------------------------------------------------
// CONVERTER IMAGEM STEAM EM SPRITE
// --------------------------------------------------

menu_amigos_criar_sprite = function(_imagem) {

    if (_imagem <= 0) return -1;

    var _dimensoes = steam_utils_get_image_size(_imagem);

    if (!is_struct(_dimensoes)) return -1;

    if (
        !variable_struct_exists(_dimensoes, "width") ||
        !variable_struct_exists(_dimensoes, "height")
    ) return -1;

    var _largura = _dimensoes.width;
    var _altura = _dimensoes.height;

    if (_largura <= 0 || _altura <= 0) return -1;

    var _buffer = buffer_create(
        _largura * _altura * 4,
        buffer_fixed,
        1
    );

    var _ok = steam_utils_get_image_rgba(
        _imagem,
        _buffer
    );

    if (!_ok) {
        buffer_delete(_buffer);
        return -1;
    }

    var _surf = surface_create(_largura, _altura);

    if (!surface_exists(_surf)) {
        buffer_delete(_buffer);
        return -1;
    }

    buffer_set_surface(_buffer, _surf, 0);

    var _sprite = sprite_create_from_surface(
        _surf, 0, 0, _largura, _altura,
        false, false, 0, 0
    );

    surface_free(_surf);
    buffer_delete(_buffer);

    return _sprite;
};

menu_amigos_atualizar = function() {

    menu_amigos_steam_disponivel = steam_api_is_initialized();

    if (!menu_amigos_steam_disponivel) return;

    // App ID do Money Town em execucao.
    var _meu_app_id = steam_utils_get_app_id();

    var _flags = SteamFriendsFriendFlag.Immediate;
    var _total = steam_friends_get_friend_count(_flags);

    if (_total < 0) return;

    var _anteriores = menu_amigos;
    var _atuais = [];

    for (var _i = 0; _i < _total; _i++) {

        var _steam_id = steam_friends_get_friend_by_index(
            _i,
            _flags
        );

        var _nome =
            steam_friends_get_friend_persona_name(_steam_id);

        // ----------------------------------------------
        // IDENTIFICAR STATUS
        //
        // 0 = Jogando Money Town
        // 1 = Online (inclusive jogando outros jogos)
        // 2 = Offline
        // ----------------------------------------------

        var _estado =
            steam_friends_get_friend_persona_state(_steam_id);

        var _grupo = 2;

        if (_estado != SteamFriendsPersonaState.Offline) {

            // Esta online na Steam.
            _grupo = 1;

            var _jogo =
                steam_friends_get_friend_game_played(_steam_id);

            // So entra em "Jogando" quando o jogo
            // do amigo for exatamente Money Town.
            if (
                _meu_app_id > 0 &&
                is_struct(_jogo) &&
                variable_struct_exists(_jogo, "game_id")
            ) {

                if (_jogo.game_id == _meu_app_id) {
                    _grupo = 0;
                }
            }
        }

        // ----------------------------------------------
        // REAPROVEITAR AVATAR
        // ----------------------------------------------

        var _sprite = -1;

        for (var _j = 0; _j < array_length(_anteriores); _j++) {

            if (_anteriores[_j].steam_id == _steam_id) {
                _sprite = _anteriores[_j].sprite;
                break;
            }
        }

        if (_sprite == -1) {

            var _imagem =
                steam_friends_get_large_friend_avatar(_steam_id);

            if (_imagem > 0) {
                _sprite = menu_amigos_criar_sprite(_imagem);
            }
        }

        array_push(_atuais, {
            steam_id: _steam_id,
            nome: _nome,
            sprite: _sprite,
            grupo: _grupo,
            origem: "steam"
        });
    }

    // ----------------------------------------------
    // LIBERAR AVATARES REMOVIDOS
    // ----------------------------------------------

    for (var _j = 0; _j < array_length(_anteriores); _j++) {

        var _antigo = _anteriores[_j];
        var _existe = false;

        for (var _k = 0; _k < array_length(_atuais); _k++) {

            if (_atuais[_k].steam_id == _antigo.steam_id) {
                _existe = true;
                break;
            }
        }

        if (!_existe && _antigo.sprite != -1) {

            if (sprite_exists(_antigo.sprite)) {
                sprite_delete(_antigo.sprite);
            }
        }
    }

    menu_amigos = _atuais;
};

// FILTRAR E MONTAR LISTA COM CATEGORIAS
// --------------------------------------------------

menu_amigos_filtrar = function() {

    menu_amigos_filtrados = [];
    menu_amigos_exibicao = [];

    var _busca = string_lower(
        string_trim(menu_amigos_busca)
    );

    // Juntar Steam e amigos internos do jogo.
    var _todos = [];

    for (var _i = 0; _i < array_length(menu_amigos); _i++) {
        array_push(_todos, menu_amigos[_i]);
    }

    for (var _i = 0; _i < array_length(menu_amigos_jogo); _i++) {

        var _interno = menu_amigos_jogo[_i];
        var _duplicado = false;

        // Nao mostrar duas vezes o mesmo SteamID.
        if (_interno.steam_id != 0) {

            for (var _j = 0; _j < array_length(_todos); _j++) {

                if (_todos[_j].steam_id == _interno.steam_id) {
                    _duplicado = true;
                    break;
                }
            }
        }

        if (!_duplicado) {
            array_push(_todos, _interno);
        }
    }

    for (var _i = 0; _i < array_length(_todos); _i++) {

        var _amigo = _todos[_i];

        if (
            _busca == "" ||
            string_pos(_busca, string_lower(_amigo.nome)) > 0
        ) {

            array_push(menu_amigos_filtrados, _amigo);
        }
    }

    // ----------------------------------------------
    // MONTAR CATEGORIAS NA ORDEM CORRETA
    // ----------------------------------------------

    var _nomes_grupos = [
        "Jogando",
        "Online",
        "Offline"
    ];

    var _pos_y = 0;

    for (var _grupo = 0; _grupo < 3; _grupo++) {

        array_push(menu_amigos_exibicao, {
            tipo: "titulo",
            texto: _nomes_grupos[_grupo],
            pos_y: _pos_y,
            altura: menu_amigos_titulo_altura
        });

        _pos_y += menu_amigos_titulo_altura;

        for (
            var _i = 0;
            _i < array_length(menu_amigos_filtrados);
            _i++
        ) {

            var _amigo = menu_amigos_filtrados[_i];

            if (_amigo.grupo == _grupo) {

                array_push(menu_amigos_exibicao, {
                    tipo: "amigo",
                    amigo: _amigo,
                    pos_y: _pos_y,
                    altura: menu_amigos_item_altura
                });

                _pos_y +=
                    menu_amigos_item_altura +
                    menu_amigos_item_espaco;
            }
        }

        _pos_y += 5;
    }

    menu_amigos_conteudo_altura = _pos_y;
};

net_register_room = function(_room) {

    net_room = _room;
    net_busy = true;
    net_connected = false;

    net_host_session_id = "";
    net_game_status = "waiting";
    net_game_started = false;
    net_auto_start_countdown = 0;

    // Reiniciar dados do tabuleiro ao entrar em outra sala.
    net_board_data = [];
    net_board_data_received = false;
    net_board_data_version = 0;

    colyseus_on_join(_room, function(_joined_room) {

        net_connected = true;
        net_busy = false;
        net_status = "Conectado!";
        net_error = "";

        net_room_id = colyseus_room_get_id(_joined_room);
        net_session_id = colyseus_room_get_session_id(_joined_room);

        net_players = [];

        net_callbacks = colyseus_callbacks_create(_joined_room);

        var _state = colyseus_room_get_state(_joined_room);

        if (_state != undefined) {

            var _host_id = colyseus_schema_get(
                _state,
                "hostSessionId"
            );

            if (is_undefined(_host_id)) {
                net_host_session_id = "";
            }
            else {
                net_host_session_id = _host_id;
            }

            net_game_status = colyseus_schema_get(
                _state,
                "status"
            );

            net_game_started = (net_game_status == "playing");

            var _countdown = colyseus_schema_get(
                _state,
                "autoStartCountdown"
            );

            if (is_undefined(_countdown)) {
                net_auto_start_countdown = 0;
            }
            else {
                net_auto_start_countdown = _countdown;
            }
        }

        colyseus_listen(
            net_callbacks,
            "hostSessionId",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_host_session_id = "";
                }
                else {
                    net_host_session_id = _value;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "status",
            function(_value, _previous) {

                net_game_status = _value;

                if (_value == "playing") {
                    net_game_started = true;
                }
            }
        );

        colyseus_listen(
            net_callbacks,
            "autoStartCountdown",
            function(_value, _previous) {

                if (is_undefined(_value)) {
                    net_auto_start_countdown = 0;
                }
                else {
                    net_auto_start_countdown = _value;
                }
            }
        );

        // Observar jogadores adicionados.
        colyseus_on_add(
            net_callbacks,
            "players",
            function(_instance, _key) {

                var _player_name = colyseus_schema_get(
                    _instance,
                    "name"
                );

                var _player_color = colyseus_schema_get(
                    _instance,
                    "color"
                );

                if (is_undefined(_player_color) || _player_color == "") {
                    _player_color = "red";
                }

                var _player_data = {
                    session_id: _key,
                    name: _player_name,
                    color: _player_color,
                    schema: _instance
                };

                net_players[array_length(net_players)] = _player_data;

                if (_key == net_session_id) {
                    net_player_color = _player_color;
                }

                show_debug_message(
                    "PLAYER ADICIONADO | ID: " + string(_key) +
                    " | NOME: " + string(_player_name) +
                    " | COR: " + string(_player_color)
                );

                show_debug_message(
                    "TOTAL APOS ADICAO: " +
                    string(array_length(net_players))
                );
            }
        );

        // Observar jogadores removidos.
        colyseus_on_remove(
            net_callbacks,
            "players",
            function(_instance, _key) {

                show_debug_message("=== ANTES DA REMOCAO ===");

                show_debug_message(
                    "ID REMOVIDO: " + string(_key)
                );

                show_debug_message(
                    "TOTAL ANTES: " +
                    string(array_length(net_players))
                );

                for (var i = 0; i < array_length(net_players); i++) {

                    show_debug_message(
                        "JOGADOR " + string(i) +
                        " | ID: " + string(net_players[i].session_id) +
                        " | NOME: " + string(net_players[i].name)
                    );
                }

                var _remaining_players = [];

                for (var j = 0; j < array_length(net_players); j++) {

                    if (net_players[j].session_id != _key) {

                        _remaining_players[
                            array_length(_remaining_players)
                        ] = net_players[j];
                    }
                }

                net_players = _remaining_players;

                show_debug_message(
                    "PLAYER REMOVIDO | ID: " + string(_key)
                );

                show_debug_message("=== JOGADORES RESTANTES ===");

                for (var k = 0; k < array_length(net_players); k++) {

                    show_debug_message(
                        string(net_players[k].session_id) + " | " +
                        string(net_players[k].name)
                    );
                }

                show_debug_message(
                    "TOTAL: " + string(array_length(net_players))
                );

                show_debug_message("===========================");
            }
        );

        // Receber mensagens do servidor.
        colyseus_on_message(
            _joined_room,
            function(_room_message, _type, _data) {

                // ------------------------------------------
                // ERRO DE COR
                // ------------------------------------------

                if (_type == "color_error") {

                    net_error = string(_data);
                }

                // ------------------------------------------
                // DADOS DO TABULEIRO
                // ------------------------------------------

                else if (_type == "board_data") {

                    var _parsed_data = undefined;

                    // O servidor envia os dados como JSON string.
                    if (is_string(_data)) {

                        _parsed_data = json_parse(_data);
                    }
                    else {

                        _parsed_data = _data;
                    }

                    if (is_array(_parsed_data)) {

                        net_board_data = _parsed_data;
                        net_board_data_received = true;

                        // Registrar nova versão do tabuleiro.
                        net_board_data_version++;

                        show_debug_message(
                            "DADOS DO TABULEIRO RECEBIDOS | VERSÃO: " +
                            string(net_board_data_version)
                        );

                        show_debug_message(
                            "QUANTIDADE DE CASAS: " +
                            string(array_length(net_board_data))
                        );

                    }
                    else {

                        show_debug_message(
                            "ERRO AO CONVERTER board_data. Tipo recebido: " +
                            typeof(_data)
                        );
                    }
                }
            }
        );

        show_debug_message("=== SALA CONECTADA ===");
        show_debug_message("Sala: " + net_room_id);
        show_debug_message("Sessão: " + net_session_id);
        show_debug_message("Líder: " + string(net_host_session_id));
        show_debug_message("Modalidade: " + net_room_mode);
    });

    colyseus_on_error(_room, function(_code, _message) {

        net_connected = false;
        net_busy = false;
        net_status = "Erro ao entrar na sala";
        net_error = _message;

        show_debug_message("ERRO COLYSEUS [" + string(_code) + "]");
        show_debug_message(_message);
    });
};

            if (_type == "rooms_begin") {

                net_lobby_rooms = [];

            else if (_type == "public_room") {

                if (is_struct(_data)) {

                    var _found = false;

                    for (var i = 0; i < array_length(net_lobby_rooms); i++) {

                        if (net_lobby_rooms[i].roomId == _data.roomId) {

                            net_lobby_rooms[i] = _data;
                            _found = true;
                            break;
                        }
                    }

                    if (!_found) {

                        net_lobby_rooms[
                            array_length(net_lobby_rooms)
                        ] = _data;
                    }

                    show_debug_message(
                        "PARTIDA RECEBIDA: " + _data.roomId
                    );

                } else {

                    show_debug_message(
                        "ERRO: Dados da partida não são um struct."
                    );
                }
            }
            else if (_type == "rooms_end") {

                show_debug_message(
                    "TOTAL DE PARTIDAS: " +
                    string(array_length(net_lobby_rooms))
                );
            }
        }
    );

    // --------------------------------------------------
    // ALEATORIZAR POSIÇÃO DAS CIDADES ESPECIAIS
    // --------------------------------------------------

    private randomizeSpecialCities() {
        const specialGroups = [
            { name: "Bali", positions: [3, 4, 5] },
            { name: "Chipre", positions: [13, 14, 15] },
            { name: "Rio de Janeiro", positions: [17, 18, 19] },
            { name: "Egito", positions: [25, 26, 27] }
        ];

        console.log("==================================");
        console.log("SORTEANDO POSIÇÕES DAS CIDADES ESPECIAIS");

        for (const group of specialGroups) {
            const currentIndex = this.board.findIndex(
                cell => cell.name === group.name
            );

            if (currentIndex === -1) {
                console.log(
                    `ERRO: cidade especial ${group.name} não encontrada.`
                );
                continue;
            }

            const randomIndex = Math.floor(
                Math.random() * group.positions.length
            );

            const targetPosition = group.positions[randomIndex];

            const previousCell = this.board[targetPosition];
            const specialCell = this.board[currentIndex];

            this.board[currentIndex] = previousCell;
            this.board[targetPosition] = specialCell;

            console.log(
                `${group.name}: posição ${currentIndex} -> ${targetPosition}`
            );

            console.log(
                `Casa deslocada: ${previousCell.name} (${targetPosition} -> ${currentIndex})`
            );
        }

        console.log("==================================");
        console.log("TABULEIRO FINAL:");

        this.board.forEach((cell, index) => {
            console.log(
                `Posição ${index}: ${cell.name} | Sprite: ${cell.spriteIndex}`
            );
        });

        console.log("==================================");
    }

    // --------------------------------------------------

    private broadcastBoard() {
        const boardData = this.board.map((cell, index) => ({
            position: index,
            name: cell.name,
            type: cell.type,
            spriteIndex: cell.spriteIndex
        }));

        this.broadcast("board_data", JSON.stringify(boardData));

        console.log(

        );
    }


    // --------------------------------------------------
    // AGENDAR ANÚNCIO APÓS A CHEGADA VISUAL
    // --------------------------------------------------

    private queueMovementAnnouncement(
        sessionId: string,
        type: "prison" | "extra" | "none" | "triple_prison",
        reason?: ExtraRollReason
    ) {

        this.pendingMovementAnnouncement = {
            sessionId,
            type,
            reason,
            arrivedClients: new Set<string>()
        };

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;
        this.state.movementAnnouncementPending = true;

        console.log(
            `Aguardando todos os jogadores confirmarem a chegada de ${sessionId} antes de liberar a próxima etapa.`
        );
    }




    // --------------------------------------------------
    // VERIFICAR SE TODOS OS CLIENTES CHEGARAM
    // --------------------------------------------------

    private finishMovementAnnouncementIfReady(
        departedSessionId?: string
    ) {

        const pending = this.pendingMovementAnnouncement;

        if (!pending) {
            return;
        }

        const connectedSessionIds = this.clients
            .map(client => client.sessionId)
            .filter(sessionId => sessionId !== departedSessionId);

        const everyoneArrived = connectedSessionIds.every(
            sessionId => pending.arrivedClients.has(sessionId)
        );

        if (!everyoneArrived) {
            return;
        }

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        if (!this.state.players.has(pending.sessionId)) {
            return;
        }

        console.log(
            `Todos os jogadores confirmaram a chegada de ${pending.sessionId}.`
        );

        // --------------------------------------------------
        // PRISÃO NORMAL
        // --------------------------------------------------

        if (pending.type === "prison") {

            this.showPrisonAnnouncement(pending.sessionId);
        }

        // --------------------------------------------------
        // TERCEIRA DUPLA CONSECUTIVA
        // --------------------------------------------------

        else if (pending.type === "triple_prison") {

            if (this.state.currentTurn === pending.sessionId) {

                // Exibir o popup somente depois da chegada.
                this.showTripleDoublesAnnouncement(
                    pending.sessionId
                );
            }
        }

        // --------------------------------------------------
        // ROLAGEM EXTRA
        // --------------------------------------------------

        else if (pending.type === "extra" && pending.reason) {

            this.showExtraRollAnnouncement(
                pending.sessionId,
                pending.reason
            );
        }
    }



    // --------------------------------------------------
    // VERIFICAR SE EXISTE ALGUM POPUP ATIVO
    // --------------------------------------------------

    private isPopupActive(): boolean {
        return (
            this.state.movementAnnouncementPending ||
            this.state.showFirstPlayerMessage ||
            this.state.showExtraRollMessage ||
            this.state.showPrisonMessage
        );
    }

    // --------------------------------------------------
    // INICIAR OU REINICIAR O TEMPO DO TURNO ATUAL
    // --------------------------------------------------

    private startCurrentTurnTimer(sessionId: string) {
        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.state.currentTurn !== sessionId) {
                return;
            }

            const currentPlayer = this.state.players.get(sessionId);

            if (!currentPlayer) {
                return;
            }

            if (!this.state.hasRolled) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a vez.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);
    }

    // --------------------------------------------------
    // LIBERAR JOGADOR DA PRISÃO E PERMITIR UMA ROLAGEM NORMAL
    // --------------------------------------------------

    private releasePlayerFromPrison(sessionId: string) {
        const player = this.state.players.get(sessionId);

        if (!player) {
            return;
        }

        this.prisonAttemptDelay?.clear();
        this.prisonAttemptDelay = undefined;

        player.inPrison = false;
        player.prisonAttempts = 0;

        player.extraRollAvailable = false;

        this.state.dice1 = 0;
        this.state.dice2 = 0;

        this.state.lastRollBy = "";

        this.state.hasRolled = false;
        this.state.hasMoved = false;

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.startCurrentTurnTimer(sessionId);

        console.log("==================================");
        console.log("JOGADOR LIBERADO DA PRISÃO!");
        console.log("JOGADOR:", player.name);
        console.log("SESSION ID:", sessionId);

    // --------------------------------------------------
    // VERIFICAR SE UMA COR É VÁLIDA
    // --------------------------------------------------

    private isValidColor(color: string): color is PlayerColor {
        return this.PLAYER_COLORS.includes(color as PlayerColor);
    }

    // --------------------------------------------------
    // ATRIBUIR COR AO JOGADOR
    // --------------------------------------------------

    private assignPlayerColor(preferredColor: string): PlayerColor {
        const occupiedColors = new Set<string>();

        this.state.players.forEach((existingPlayer) => {
            occupiedColors.add(existingPlayer.color);
        });

        if (
            this.isValidColor(preferredColor) &&
            !occupiedColors.has(preferredColor)
        ) {
            return preferredColor;
        }

        for (const color of this.PLAYER_COLORS) {
            if (!occupiedColors.has(color)) {
                return color;
            }
        }

        return "red";
    }

    // --------------------------------------------------
    // ESCOLHER PRIMEIRO JOGADOR ALEATORIAMENTE
    // --------------------------------------------------

    private chooseRandomFirstPlayer(): string {
        const sessionIds = [...this.state.players.keys()];

        if (sessionIds.length === 0) {
            return "";
        }

        const randomIndex = Math.floor(
            Math.random() * sessionIds.length
        );

        return sessionIds[randomIndex];
    }

    // --------------------------------------------------
    // ATUALIZAR JOGADOR SORTEADO
    // --------------------------------------------------

    private updateFirstPlayerAnnouncement(sessionId: string) {
        const player = this.state.players.get(sessionId);

        if (!player) {
            this.state.firstPlayerSessionId = "";
            this.state.firstPlayerName = "";
            return;
        }

        this.state.firstPlayerSessionId = sessionId;
        this.state.firstPlayerName = player.name;

        console.log("==================================");
        console.log("PRIMEIRO JOGADOR SORTEADO!");
        console.log("JOGADOR:", player.name);
        console.log("SESSION ID:", sessionId);
        console.log("==================================");
    }

    // --------------------------------------------------
    // EXIBIR ANÚNCIO DO PRIMEIRO JOGADOR
    // --------------------------------------------------

    private showFirstPlayerAnnouncement() {
        if (this.state.status !== "playing") {
            return;
        }

        const firstPlayerSessionId = this.chooseRandomFirstPlayer();

        if (firstPlayerSessionId === "") {
            return;
        }

        this.updateFirstPlayerAnnouncement(firstPlayerSessionId);

        this.state.turnDeadline = 0;
        this.state.showFirstPlayerMessage = true;

        console.log(
            "Exibindo anúncio do primeiro jogador por 3 segundos."
        );

        this.firstPlayerMessageDelay?.clear();

        this.firstPlayerMessageDelay = this.clock.setTimeout(() => {
            this.firstPlayerMessageDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            let selectedSessionId = this.state.firstPlayerSessionId;

            if (!this.state.players.has(selectedSessionId)) {
                selectedSessionId = this.chooseRandomFirstPlayer();

                if (selectedSessionId === "") {
                    this.state.showFirstPlayerMessage = false;
                    this.state.firstPlayerSessionId = "";
                    this.state.firstPlayerName = "";
                    return;
                }

                this.updateFirstPlayerAnnouncement(selectedSessionId);
            }

            this.state.showFirstPlayerMessage = false;

            console.log(
                "Anúncio encerrado! Liberando o primeiro turno."
            );

            this.nextTurn(selectedSessionId);
        }, FIRST_PLAYER_MESSAGE_DURATION);
    }

 
    // --------------------------------------------------
    // POPUP: TRÊS DUPLAS CONSECUTIVAS
    // --------------------------------------------------

    private showTripleDoublesAnnouncement(sessionId: string) {

        const player = this.state.players.get(sessionId);

        if (!player) {
            return;
        }

        this.prisonMessageDelay?.clear();

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;

        // A prisão passa a valer após o personagem chegar.
        player.inPrison = true;
        player.prisonAttempts = 0;
        player.extraRollAvailable = false;

        this.state.prisonPlayerSessionId = sessionId;
        this.state.prisonPlayerName = player.name;
        this.state.prisonMessageReason = "triple_doubles";

        this.state.showPrisonMessage = true;

        console.log(
            `Exibindo popup de três duplas para ${player.name} após a chegada.`
        );

        this.prisonMessageDelay = this.clock.setTimeout(() => {

            this.prisonMessageDelay = undefined;

            this.state.showPrisonMessage = false;
            this.state.prisonPlayerName = "";
            this.state.prisonPlayerSessionId = "";
            this.state.prisonMessageReason = "";

            if (this.state.status !== "playing") {
                return;
            }

            // Encerrar o turno somente após o popup.
            if (
                this.state.currentTurn === sessionId &&
                this.state.players.has(sessionId)
            ) {

                console.log(
                    `Popup de três duplas encerrado. Passando o turno de ${player.name}.`
                );

                this.nextTurn();
            }

        }, TRIPLE_DOUBLES_MESSAGE_DURATION);
    }


 
    // --------------------------------------------------
    // EXIBIR ANÚNCIO DE PRISÃO
    // --------------------------------------------------

    private showPrisonAnnouncement(sessionId: string) {

        const player = this.state.players.get(sessionId);

        if (!player) {
            return;
        }

        this.prisonMessageDelay?.clear();

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;

        // --------------------------------------------------
        // FINALIZAR RESULTADO E ATIVAR PRISÃO
        // --------------------------------------------------

        player.inPrison = true;

        this.state.hasRolled = false;
        this.state.lastRollBy = "";

        // Os valores dos dados não são apagados aqui.
        // O próximo turno já possui sua própria limpeza.

        this.state.prisonPlayerSessionId = sessionId;
        this.state.prisonPlayerName = player.name;
        this.state.showPrisonMessage = true;

        console.log("==================================");
        console.log("JOGADOR CAIU NA PRISÃO!");
        console.log("JOGADOR:", player.name);
        console.log("SESSION ID:", sessionId);
        console.log("==================================");

        this.prisonMessageDelay = this.clock.setTimeout(() => {

            this.prisonMessageDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            this.state.showPrisonMessage = false;
            this.state.prisonPlayerName = "";
            this.state.prisonPlayerSessionId = "";

            console.log(
                "Anúncio de prisão encerrado. Encerrando o turno do jogador."
            );

            if (
                this.state.currentTurn === sessionId &&
                this.state.players.has(sessionId)

                this.nextTurn();
            }

        }, PRISON_MESSAGE_DURATION);

    // --------------------------------------------------
    // INICIAR PRAZO DA ROLAGEM EXTRA
    // --------------------------------------------------

    private startExtraRollTimer(sessionId: string) {
        if (this.state.status !== "playing") {
            return;
        }

        if (this.state.currentTurn !== sessionId) {
            return;
        }

        const player = this.state.players.get(sessionId);

        if (!player || !player.extraRollAvailable) {
            return;
        }

        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.state.currentTurn !== sessionId) {
                return;
            }

            const currentPlayer = this.state.players.get(sessionId);

            if (
                currentPlayer?.extraRollAvailable &&
                !this.state.hasRolled
            ) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a rolagem extra.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);
    }


 

    // --------------------------------------------------
    // EXIBIR ANÚNCIO DA ROLAGEM EXTRA
    // --------------------------------------------------

    private showExtraRollAnnouncement(
        sessionId: string,
        reason: ExtraRollReason
    ) {

        const player = this.state.players.get(sessionId);

        if (!player) {
            return;
        }

        this.extraRollMessageDelay?.clear();

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;

        // --------------------------------------------------
        // FINALIZAR ROLAGEM ANTERIOR APÓS A CHEGADA
        // --------------------------------------------------

        this.state.hasRolled = false;

            this.extraRollMessageDelay = undefined;

                return;

            this.state.showExtraRollMessage = false;
            this.state.extraRollPlayerName = "";
            this.state.extraRollPlayerSessionId = "";
            this.state.extraRollReason = "";

            );

    // --------------------------------------------------
    // GERAR CÓDIGO DE 6 CARACTERES
    // --------------------------------------------------

    private generateCode(): string {
        let code = "";

        for (let i = 0; i < 6; i++) {
            const index = Math.floor(
                Math.random() * this.CODE_CHARACTERS.length
            );

            code += this.CODE_CHARACTERS[index];
        }

        return code;
    }

    // --------------------------------------------------
    // GARANTIR CÓDIGO ÚNICO
    // --------------------------------------------------

    private async generateRoomCode(): Promise<string> {
        const existingCodes = await this.presence.smembers(
            this.CODE_CHANNEL
        );

        let code = "";

        do {
            code = this.generateCode();
        } while (existingCodes.includes(code));

        await this.presence.sadd(this.CODE_CHANNEL, code);

        return code;
    }

    // --------------------------------------------------
    // CANCELAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private clearAutoStartCountdown() {
        this.autoStartInterval?.clear();
        this.autoStartInterval = undefined;

        this.state.autoStartCountdown = 0;
    }

    // --------------------------------------------------
    // REINICIAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private resetAutoStartCountdown() {
        if (this.mode !== "public") {
            this.clearAutoStartCountdown();
            return;
        }

        if (this.state.status !== "waiting") {
            this.clearAutoStartCountdown();
            return;
        }

        this.clearAutoStartCountdown();

        if (this.state.players.size < 2) {
            return;
        }

        this.state.autoStartCountdown = AUTO_START_DURATION;

        console.log(
            `Contagem automática iniciada: ${this.roomId} (${AUTO_START_DURATION}s)`
        );

        this.autoStartInterval = this.clock.setInterval(() => {
            if (this.state.status !== "waiting") {
                this.clearAutoStartCountdown();
                return;
            }

            if (this.state.players.size < 2) {
                this.clearAutoStartCountdown();
                return;
            }

            this.state.autoStartCountdown--;

            console.log(
                `Sala ${this.roomId}: ${this.state.autoStartCountdown}s`
            );

            if (this.state.autoStartCountdown <= 0) {
                this.clearAutoStartCountdown();
                this.startGame();
            }
        }, 1000);
    }

    // --------------------------------------------------
    // INICIAR PARTIDA
    // --------------------------------------------------

    private startGame() {
        if (this.state.status !== "waiting") {
            return;
        }

        if (this.state.players.size < 2) {
            return;
        }

        this.clearAutoStartCountdown();

        this.randomizeSpecialCities();
        this.broadcastBoard();

        this.state.status = "playing";

        this.state.currentTurn = "";
        this.state.turnDeadline = 0;
        this.state.turnCount = 0;

        this.state.hasRolled = false;
        this.state.hasMoved = false;

        this.state.dice1 = 0;
        this.state.dice2 = 0;
        this.state.lastRollBy = "";

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        this.playersResetForNewGame();

        void this.setMetadata({
            mode: this.mode,
            status: this.state.status
        });

        this.lock();

        console.log("Partida iniciada:", this.roomId);
        console.log("Aguardando 3 segundos para iniciar o anúncio...");

        this.startDelay?.clear();

        this.startDelay = this.clock.setTimeout(() => {
            this.startDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            console.log("Animação concluída! Sorteando primeiro jogador.");

            this.showFirstPlayerAnnouncement();
        }, GAME_START_DELAY);
    }

    // --------------------------------------------------
    // REINICIAR ESTADOS DOS JOGADORES
    // --------------------------------------------------

    private playersResetForNewGame() {

        this.state.players.forEach((player) => {

            player.extraRollAvailable = false;

            player.consecutiveDoubles = 0;
            player.prisonReturnBackward = false;

            player.inPrison = false;
            player.prisonAttempts = 0;

            player.debtsAvailable = 1;
        });
    }


    messages = {
        select_color: (client: Client, data: any) => {
            if (this.state.status !== "waiting") {
                client.send(
                    "color_error",
                    "Não é possível trocar de cor depois que a partida começou."
                );
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            const requestedColor = String(data?.color ?? "");

            if (!this.isValidColor(requestedColor)) {
                client.send("color_error", "Cor inválida.");
                return;
            }

            const occupied = [...this.state.players.keys()].some(
                sessionId => {
                    if (sessionId === client.sessionId) {
                        return false;
                    }

                    const otherPlayer = this.state.players.get(sessionId);

                    return otherPlayer?.color === requestedColor;
                }
            );

            if (occupied) {
                client.send(
                    "color_error",
                    "Essa cor já está sendo utilizada por outro jogador."
                );
                return;
            }

            player.color = requestedColor;

            console.log(
                `${player.name} escolheu a cor: ${requestedColor}`
            );
        },

        start: (client: Client) => {
            if (this.state.status !== "waiting") {
                client.send("start_error", "A partida já começou.");
                return;
            }

            if (this.mode !== "private") {
                client.send(
                    "start_error",
                    "Partidas públicas começam automaticamente."
                );
                return;
            }

            if (client.sessionId !== this.state.hostSessionId) {
                client.send(
                    "start_error",
                    "Somente o anfitrião pode iniciar."
                );
                return;
            }

            if (this.state.players.size < 2) {
                client.send(
                    "start_error",
                    "São necessários pelo menos 2 jogadores."
                );
                return;
            }

            this.startGame();
        },

        // --------------------------------------------------
        // ENVIAR TABULEIRO
        // --------------------------------------------------

        get_board: (client: Client) => {
            const boardData = this.board.map((cell, index) => ({
                position: index,
                name: cell.name,
                type: cell.type,
                spriteIndex: cell.spriteIndex
            }));

            client.send("board_data", JSON.stringify(boardData));

            console.log(
                `Tabuleiro enviado para ${client.sessionId}: ${boardData.length} casas.`
            );
        },

        // --------------------------------------------------
        // CONFIRMAR CHEGADA DA ANIMAÇÃO DE MOVIMENTO
        // --------------------------------------------------

        movement_arrived: (client: Client) => {
            const pending = this.pendingMovementAnnouncement;

            if (!pending) {
                return;
            }

            if (!this.state.players.has(client.sessionId)) {
                return;
            }

            pending.arrivedClients.add(client.sessionId);

            console.log(
                `Chegada confirmada: ${client.sessionId} (${pending.arrivedClients.size}/${this.clients.length})`
            );

            this.finishMovementAnnouncementIfReady();
        },



        // --------------------------------------------------
        // ROLAR DADOS NORMALMENTE
        // --------------------------------------------------

        roll_dice: (client: Client) => {

            if (this.state.status !== "playing") {

                client.send(
                    "action_error",
                    "A partida ainda não começou."
                );

                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {

                client.send(
                    "action_error",
                    "Não é seu turno."
                );

                return;
            }

            const player = this.state.players.get(
                client.sessionId
            );

            if (!player) {
                return;
            }

            if (player.inPrison) {

                client.send(
                    "action_error",
                    "Você precisa resolver sua situação na prisão primeiro."
                );

                return;
            }

            if (this.state.hasRolled) {

                client.send(
                    "action_error",
                    "Você já rolou os dados nesta rodada."
                );

                return;
            }

            if (
                this.state.hasMoved &&
                !player.extraRollAvailable
            ) {

                client.send(
                    "action_error",
                    "Você já movimentou nesta rodada."
                );

                return;
            }

            // --------------------------------------------------
            // ROLAR DADOS
            // --------------------------------------------------

            const dice1 = Math.floor(Math.random() * 6) + 1;
            const dice2 = Math.floor(Math.random() * 6) + 1;

            const rolledDoubles = dice1 === dice2;

            this.state.dice1 = dice1;
            this.state.dice2 = dice2;

            this.state.lastRollBy = client.sessionId;
            this.state.hasRolled = true;

            player.extraRollAvailable = false;
            player.prisonReturnBackward = false;

            this.state.hasMoved = false;

            // --------------------------------------------------
            // CONTAR DUPLAS CONSECUTIVAS
            // --------------------------------------------------

            if (rolledDoubles) {

                player.consecutiveDoubles++;

            } else {

                player.consecutiveDoubles = 0;
            }

            console.log(
                `${player.name}: ${dice1} + ${dice2}`
            );

            console.log(
                `Duplas consecutivas: ${player.consecutiveDoubles}`
            );

            // --------------------------------------------------
            // TERCEIRA DUPLA: PRISÃO
            // --------------------------------------------------

            if (player.consecutiveDoubles >= 3) {

                const previousPosition = player.position;

                player.consecutiveDoubles = 0;

                player.prisonReturnBackward =
                    previousPosition > PRISON_CELL;

                player.position = PRISON_CELL;

                player.lastCellType =
                    this.board[PRISON_CELL].type;

                player.lastCellName =
                    this.board[PRISON_CELL].name;

                // A prisão será ativada quando chegar.
                // Manter inPrison como false durante a
                // animação para exibir os dados a todos.
                player.inPrison = false;

                player.prisonAttempts = 0;
                player.extraRollAvailable = false;

                this.state.hasMoved = true;

                // IMPORTANTE:
                // Não apagar hasRolled nem lastRollBy.
                // O resultado deve continuar visível
                // durante toda a caminhada.

                console.log(
                    `${player.name} fez três duplas e foi enviado à prisão.`
                );

                console.log(
                    `Movimento: ${previousPosition} -> ${PRISON_CELL}`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "triple_prison"
                );

                // Não exibir o popup aqui.
                // O anúncio será iniciado após todos
                // confirmarem a chegada à casa 8.

                return;
            }
        },

        // --------------------------------------------------
        // TENTAR SAIR DA PRISÃO COM DUPLAS
        // --------------------------------------------------

        prison_try_doubles: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive() || this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (this.state.hasRolled || this.state.hasMoved) {
                client.send(
                    "action_error",
                    "Não é possível tentar sair da prisão agora."
                );
                return;
            }

            const dice1 = Math.floor(Math.random() * 6) + 1;
            const dice2 = Math.floor(Math.random() * 6) + 1;

            const rolledDoubles = dice1 === dice2;

            player.prisonAttempts++;

            // Os dados são compartilhados com o cliente para exibição.
            // O jogador continua preso durante esse período.
            this.state.dice1 = dice1;
            this.state.dice2 = dice2;
            this.state.lastRollBy = client.sessionId;
            this.state.hasRolled = true;
            this.state.hasMoved = false;

            // Pausar o limite do turno enquanto o resultado aparece.
            this.turnTimeout?.clear();
            this.turnTimeout = undefined;
            this.state.turnDeadline = 0;

            console.log("==================================");
            console.log("TENTATIVA DE SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log(`DADOS: ${dice1} + ${dice2}`);
            console.log("DUPLAS:", rolledDoubles);
            console.log(
                `TENTATIVA: ${player.prisonAttempts}/${MAX_PRISON_ATTEMPTS}`
            );
            console.log("==================================");

            // Aguardar a HUD exibir os dados e o resultado.
            this.prisonAttemptDelay?.clear();

            this.prisonAttemptDelay = this.clock.setTimeout(() => {
                this.prisonAttemptDelay = undefined;

                if (this.state.status !== "playing") {
                    return;
                }

                if (this.state.currentTurn !== client.sessionId) {
                    return;
                }

                const currentPlayer = this.state.players.get(client.sessionId);

                if (!currentPlayer || !currentPlayer.inPrison) {
                    return;
                }

                // Conseguiu duplas: liberar e permitir uma rolagem normal.
                if (rolledDoubles) {
                    console.log(
                        `${currentPlayer.name} conseguiu duplas e saiu da prisão.`
                    );

                    this.releasePlayerFromPrison(client.sessionId);
                    return;
                }

                // Terceira tentativa: libertar automaticamente.
                if (currentPlayer.prisonAttempts >= MAX_PRISON_ATTEMPTS) {
                    console.log(
                        `${currentPlayer.name} atingiu o limite de tentativas e foi liberado automaticamente.`
                    );

                    this.releasePlayerFromPrison(client.sessionId);
                    return;
                }

                // Falhou: informar e encerrar o turno.
                client.send(
                    "prison_attempt_failed",
                    {
                        attempts: currentPlayer.prisonAttempts,
                        maxAttempts: MAX_PRISON_ATTEMPTS
                    }
                );

                console.log(
                    `${currentPlayer.name} não conseguiu duplas. A próxima tentativa será no próximo turno.`
                );

                this.nextTurn();
            }, PRISON_ATTEMPT_RESULT_DURATION);
        },

        // --------------------------------------------------
        // PAGAR PARA SAIR DA PRISÃO
        // --------------------------------------------------

        prison_pay: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (player.money < PRISON_ESCAPE_COST) {
                client.send(
                    "action_error",
                    "Você não possui dinheiro suficiente para pagar a saída da prisão."
                );
                return;
            }

            player.money -= PRISON_ESCAPE_COST;

            console.log("==================================");
            console.log("JOGADOR PAGOU PARA SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log("VALOR PAGO:", PRISON_ESCAPE_COST);
            console.log("DINHEIRO RESTANTE:", player.money);
            console.log("==================================");

            this.releasePlayerFromPrison(client.sessionId);
        },

        // --------------------------------------------------
        // USAR DÍVIDA PARA SAIR DA PRISÃO
        // --------------------------------------------------

        prison_use_debt: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (player.debtsAvailable <= 0) {
                client.send(
                    "action_error",
                    "Você não possui dívidas disponíveis."
                );
                return;
            }

            player.debtsAvailable--;

            console.log("==================================");
            console.log("JOGADOR USOU UMA DÍVIDA PARA SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log(
                "DÍVIDAS RESTANTES:",
                player.debtsAvailable
            );
            console.log("==================================");

            this.releasePlayerFromPrison(client.sessionId);
        },


        // --------------------------------------------------
        // MOVIMENTAR JOGADOR
        // --------------------------------------------------

        move: (client: Client) => {

            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(
                client.sessionId
            );

            if (!player) {
                return;
            }

            if (player.inPrison) {

                client.send(
                    "action_error",
                    "Você precisa sair da prisão antes de movimentar."
                );

                return;
            }

            if (
                !this.state.hasRolled ||
                this.state.lastRollBy !== client.sessionId
            ) {

                client.send(
                    "action_error",
                    "Você precisa rolar os dados primeiro."
                );

                return;
            }

            if (this.state.hasMoved) {

                client.send(
                    "action_error",
                    "Você já movimentou nesta rodada."
                );

                return;
            }

            const dice1 = this.state.dice1;
            const dice2 = this.state.dice2;

            const rolledDoubles = dice1 === dice2;

            const steps = dice1 + dice2;

            const startingPosition = player.position;
            const newPosition = startingPosition + steps;

            const completedLaps = Math.floor(
                newPosition / BOARD_SIZE
            );

            const landedPosition = newPosition % BOARD_SIZE;

            player.position = landedPosition;
            player.laps += completedLaps;

            const landedCell = this.board[landedPosition];

            player.lastCellType = landedCell.type;
            player.lastCellName = landedCell.name;

            this.state.hasMoved = true;

            console.log(
                `${player.name} avançou ${steps} casas. Posição: ${player.position}`
            );

            console.log(
                `${player.name} caiu na casa ${player.position}: ${landedCell.name} (${landedCell.type})`
            );

            console.log(
                `Dados: ${dice1} + ${dice2} | Duplas: ${rolledDoubles}`
            );

            // --------------------------------------------------
            // RECOMPENSA POR COMPLETAR VOLTAS
            // --------------------------------------------------

            if (completedLaps > 0) {

                console.log(
                    `${player.name} completou ${completedLaps} volta(s). Total: ${player.laps}`
                );

                const totalMoneyReward =
                    START_CELL_MONEY_REWARD * completedLaps;

                player.money += totalMoneyReward;

                console.log(
                    `${player.name} recebeu $${totalMoneyReward.toLocaleString("pt-BR")} por completar ${completedLaps} volta(s).`
                );
            }

            // --------------------------------------------------
            // VERIFICAR SE CAIU NA PRISÃO
            // --------------------------------------------------

            const landedInPrison =
                landedCell.name === "Prisão" &&
                landedCell.type === "island";

            if (landedInPrison) {

                // A prisão será ativada após a chegada.
                // Enquanto caminha, manter a exibição
                // do resultado dos dados.

                player.prisonAttempts = 0;
                player.extraRollAvailable = false;

                this.state.hasMoved = true;

                // Não zerar hasRolled nem lastRollBy aqui.

                console.log(
                    `${player.name} foi enviado para a prisão. O turno será encerrado após o anúncio.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "prison"
                );

                return;
            }

            // --------------------------------------------------
            // CASA INICIAL
            // --------------------------------------------------

            const landedExactlyOnStart =
                landedPosition === 0 &&
                landedCell.type === "start" &&
                landedCell.name === "Início";

            const receivesStartExtraRoll =
                landedExactlyOnStart &&
                completedLaps > 0 &&
                player.laps >= 1;

            let extraRollReason: ExtraRollReason | undefined;

            if (receivesStartExtraRoll) {

                extraRollReason = "start";

            } else if (rolledDoubles) {

                extraRollReason = "doubles";
            }

            const canReceiveExtraRoll =
                extraRollReason !== undefined;

            console.log(
                `[VERIFICAÇÃO ROLAGEM EXTRA] ${player.name} | ` +
                `posição inicial: ${startingPosition} | ` +
                `dados: ${dice1} + ${dice2} | ` +
                `duplas: ${rolledDoubles} | ` +
                `posição final: ${landedPosition} | ` +
                `voltas: ${player.laps} | ` +
                `terminou no início: ${receivesStartExtraRoll} | ` +
                `motivo: ${extraRollReason ?? "nenhum"} | ` +
                `recebe extra: ${canReceiveExtraRoll}`
            );

            // --------------------------------------------------
            // ROLAGEM EXTRA
            // --------------------------------------------------

            if (canReceiveExtraRoll && extraRollReason) {

                player.extraRollAvailable = true;

                this.state.hasMoved = true;

                // Preservar hasRolled e lastRollBy
                // até a chegada do personagem.
                //
                // Isso mantém o resultado visível
                // para todos os clientes.

                console.log(
                    `${player.name} recebeu uma rolagem extra pelo motivo: ${extraRollReason}.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "extra",
                    extraRollReason
                );

            } else {

                // --------------------------------------------------
                // MOVIMENTO NORMAL
                // --------------------------------------------------

                // Mesmo sem popup, aguardar a animação
                // antes de liberar as ações da HUD.

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "none"
                );
            }
        },


        // --------------------------------------------------
        // ENCERRAR TURNO
        // --------------------------------------------------

        end_turn: (client: Client) => {

            const player = this.state.players.get(client.sessionId);

            console.log("==================================");
            console.log("[END_TURN] SOLICITACAO RECEBIDA");
            console.log("[END_TURN] Cliente:", client.sessionId);
            console.log("[END_TURN] Jogador:", player?.name ?? "Nao encontrado");
            console.log("[END_TURN] Status:", this.state.status);
            console.log("[END_TURN] Turno atual:", this.state.currentTurn);
            console.log("[END_TURN] HasRolled:", this.state.hasRolled);
            console.log("[END_TURN] HasMoved:", this.state.hasMoved);
            console.log("[END_TURN] Primeiro jogador popup:", this.state.showFirstPlayerMessage);
            console.log("[END_TURN] Rolagem extra popup:", this.state.showExtraRollMessage);
            console.log("[END_TURN] Prisao popup:", this.state.showPrisonMessage);
            console.log("[END_TURN] Preso:", player?.inPrison);
            console.log("[END_TURN] Rolagem extra:", player?.extraRollAvailable);
            console.log("[END_TURN] Jogadores na sala:", this.state.players.size);

            // --------------------------------------------------
            // VERIFICAR PARTIDA
            // --------------------------------------------------

            if (this.state.status !== "playing") {

                console.log("[END_TURN] BLOQUEADO: partida nao esta em andamento.");

                client.send(
                    "action_error",
                    "A partida nao esta em andamento."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR POPUPS
            // --------------------------------------------------

            if (this.isPopupActive()) {

                console.log("[END_TURN] BLOQUEADO: popup ativo no servidor.");

                client.send(
                    "action_error",
                    "Aguarde o encerramento do popup."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR JOGADOR
            // --------------------------------------------------

            if (!player) {

                console.log("[END_TURN] BLOQUEADO: jogador nao encontrado.");

                client.send(
                    "action_error",
                    "Jogador nao encontrado."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR TURNO
            // --------------------------------------------------

            if (this.state.currentTurn !== client.sessionId) {

                console.log("[END_TURN] BLOQUEADO: nao e o turno deste cliente.");

                client.send(
                    "action_error",
                    "Nao e seu turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR MOVIMENTO
            // --------------------------------------------------

            if (!this.state.hasMoved) {

                console.log("[END_TURN] BLOQUEADO: movimento nao concluido.");

                client.send(
                    "action_error",
                    "Voce precisa movimentar antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR PRISAO
            // --------------------------------------------------

            if (player.inPrison) {

                console.log("[END_TURN] BLOQUEADO: jogador esta preso.");

                client.send(
                    "action_error",
                    "Voce ainda esta na prisao."
                );

                return;
            }

            // --------------------------------------------------
            // VERIFICAR ROLAGEM EXTRA
            // --------------------------------------------------

            if (player.extraRollAvailable) {

                console.log("[END_TURN] BLOQUEADO: rolagem extra pendente.");

                client.send(
                    "action_error",
                    "Utilize sua rolagem extra antes de encerrar o turno."
                );

                return;
            }

            // --------------------------------------------------
            // ENCERRAR TURNO
            // --------------------------------------------------

            console.log("[END_TURN] AUTORIZADO!");
            console.log("[END_TURN] Encerrando turno de:", player.name);

            const previousTurn = this.state.currentTurn;

            this.nextTurn();

            console.log("[END_TURN] Turno anterior:", previousTurn);
            console.log("[END_TURN] Novo turno:", this.state.currentTurn);
            console.log("==================================");
        },


        play: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                return;
            }

            player.score++;
            this.nextTurn();
        },
    };

        this.roomId = await this.generateRoomCode();

        this.mode = options.mode === "private" ? "private" : "public";

        this.state.status = "waiting";
        this.state.autoStartCountdown = 0;

        this.state.movementAnnouncementPending = false;
        this.pendingMovementAnnouncement = undefined;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        await this.setMatchmaking({
            metadata: {
                mode: this.mode,
                status: this.state.status
            },
            private: this.mode === "private"
        });

        console.log("==================================");

        console.log("MODO:", this.mode);
        console.log("PRIVADA:", this.mode === "private");
        console.log(

            this.mode === "public" ? "PUBLICA" : "OCULTA"
        );

        if (this.board.length !== BOARD_SIZE) {
            throw new Error(
                `Tabuleiro inválido: esperado ${BOARD_SIZE}, encontrado ${this.board.length}.`
            );
        }
    }

    // AUTENTICAÇÃO
    // --------------------------------------------------

    onAuth(client: Client, options: any) {

        if (this.state.status !== "waiting") {
            throw new ServerError(409, "Esta partida já começou.");
        }

        if (options?.mode !== this.mode) {
            throw new ServerError(
                403,
                "Esta sala não pertence a essa modalidade de partida."
            );
        }

        // --------------------------------------------------
        // VALIDAR NOME STEAM
        // --------------------------------------------------

        if (
            typeof options?.playerName !== "string"
        ) {
            throw new ServerError(400, "Nome Steam inválido.");
        }

        const playerName = String(options.playerName).trim();

        if (playerName.length === 0 || playerName.length > 32) {
            throw new ServerError(
                400,
                "O nome Steam deve possuir entre 1 e 32 caracteres."
            );
        }

        // --------------------------------------------------
        // VALIDAR STEAM ID
        // --------------------------------------------------

        const steamId = String(options?.steamId ?? "").trim();

        if (!/^\d{17}$/.test(steamId)) {
            throw new ServerError(
                400,
                "Steam ID inválido ou não recebido."
            );
        }

        // --------------------------------------------------
        // VALIDAR PERSONAGEM
        // --------------------------------------------------

        const selectedCharacter = Number(
            options?.selectedCharacter ?? 0
        );

        const validCharacter =
            Number.isInteger(selectedCharacter) &&
            selectedCharacter >= 0 &&
            selectedCharacter <= 3;

        if (!validCharacter) {
            throw new ServerError(400, "Personagem inválido.");
        }

        // --------------------------------------------------
        // VALIDAR COR
        // --------------------------------------------------

        const preferredColor =
            typeof options?.selectedColor === "string" &&
            this.isValidColor(options.selectedColor)
                ? options.selectedColor
                : "red";

        return {
            playerName,
            steamId,
            preferredColor,
            selectedCharacter
        };
    }

    onJoin(client: Client, options: any, auth: any) {

        const player = new Player();

        // Identidade Steam.
        player.name = auth.playerName;
        player.steamId = auth.steamId;

        player.money = INITIAL_PLAYER_MONEY;
        player.character = auth.selectedCharacter ?? 0;

        player.position = 0;
        player.laps = 0;

function friendCodeFromSteamId(steamId: string): string {

    if (!STEAM_PATTERN.test(steamId)) {
        throw new Error("SteamID invalido.");
    }

    let accountId = BigInt(steamId) & 0xFFFFFFFFn;
    let code = "";

    for (let i = 0; i < 6; i++) {
        const digit = Number(accountId % 62n);

        code =
            FRIEND_CODE_ALPHABET[digit] + code;

        accountId /= 62n;
    }

    return code;
}

                    if (!ownerId) {
                        throw new Error(
                            "Conta nao identificada."
                        );
                    }

                    // Remover efetivamente do banco.
                    database.removeFriend(
                        ownerId,
                        targetId
                    );

                    // Confirmar a operacao ao cliente.
                    client.send(
                        "friend_remove_result",
                        {
                            ok: true,
                            steamId: targetId,
                            message: "Amizade removida."
                        }
                    );

                    // Enviar a nova lista para TODAS
                    // as instancias conectadas dos dois.
                    this.broadcastFriends(ownerId);
                    this.broadcastFriends(targetId);

                    console.log(
                        "[FRIENDS] Lista atualizada apos remocao."
                    );

                } catch (error) {

                    console.error(
                        "[FRIENDS] FALHA AO REMOVER:",
                        error
                    );

                    client.send(
                        "friend_remove_result",
                        {
                            ok: false,
                            steamId: targetId,
                            message: this.errorText(error)
                        }
                    );
                }
            }
        );

        // ==============================================
        // CONSULTAR LISTA
        // ==============================================

        this.onMessage(
            "friend_list",
            client => {

                const steamId =
                    this.sessions.get(
                        client.sessionId
                    );

                if (steamId) {
                    this.sendFriends(
                        client,
                        steamId
                    );
                }
            }
        );
    }

    // --------------------------------------------------
    // ENTRAR NA SALA
    // --------------------------------------------------

    onJoin(client: Client, options: JoinOptions) {

        try {

            const steamId = String(
                options?.steamId ?? ""
            ).trim();

            const playerName = String(
                options?.playerName ?? "Jogador"
            ).slice(0, 64);

            const suppliedFriendCode = String(
                options?.friendCode ?? ""
            ).trim();

            if (!STEAM_PATTERN.test(steamId)) {
                throw new Error(
                    "SteamID invalido."
                );
            }

            const friendCode =
                friendCodeFromSteamId(steamId);

            if (suppliedFriendCode !== friendCode) {

                console.log(
                    "[FRIENDS] Codigo informado difere " +
                    "do calculado:",
                    suppliedFriendCode || "(vazio)",
                    "->",
                    friendCode
                );
            }

            const account = database.register(
                steamId,
                playerName,
                friendCode
            );

            const explicitInstanceId = String(
                options?.instanceId ?? ""
            ).trim();

            const instanceId =
                explicitInstanceId ||
                (
                    suppliedFriendCode &&
                    suppliedFriendCode !== friendCode
                        ? suppliedFriendCode
                        : client.sessionId
                );

            this.sessions.set(
                client.sessionId,
                steamId
            );

            this.sessionInstances.set(
                client.sessionId,
                instanceId
            );

            console.log(
                "[FRIENDS] Sessao registrada:",
                client.sessionId,
                "| Instancia:",
                instanceId
            );

            client.send(
                "friend_registered",
                {
                    ok: true,
                    code: account.code,
                    steamId: account.steamId
                }
            );

            this.sendFriends(
                client,
                steamId
            );

            const pending =
                database.getPending(steamId);

            for (const request of pending) {

                if (
                    database.isTestAccount(
                        request.from
                    ) &&
                    !request.originInstanceId
                ) {
                    continue;
                }

                if (
                    request.originInstanceId === instanceId ||
                    this.requestOriginSessions.get(
                        request.id
                    ) === client.sessionId
                ) {
                    continue;
                }

                client.send(
                    "friend_request_received",
                    this.serializeRequest(request)
                );
            }

            this.broadcastFriends(steamId);

            console.log(
                "[FRIENDS] Conta conectada:",
                account.name,
                "| Codigo:",
                account.code
            );

        } catch (error) {

            console.error(
                "[FRIENDS] Erro no cadastro:",
                error
            );

            client.send(
                "friend_registered",
                {
                    ok: false,
                    message: this.errorText(error)
                }
            );
        }
    }

    // --------------------------------------------------
    // SAIR DA SALA
    // --------------------------------------------------

    onLeave(client: Client) {

        const steamId =
            this.sessions.get(
                client.sessionId
            );

        this.sessions.delete(
            client.sessionId
        );

        this.sessionInstances.delete(
            client.sessionId
        );

        if (steamId) {
            this.broadcastFriends(steamId);
        }
    }

    // --------------------------------------------------
    // VERIFICAR STATUS ONLINE
    // --------------------------------------------------

    private isOnline(steamId: string) {

        for (const id of this.sessions.values()) {

            if (id === steamId) {
                return true;
            }
        }

        return false;
    }

    // --------------------------------------------------
    // SERIALIZAR PEDIDO
    // --------------------------------------------------

    private serializeRequest(
        request: FriendRequest
    ) {

        const sender =
            database.getAccount(
                request.from
            );

        const simulated =
            database.isTestAccount(
                request.from
            );

        if (simulated) {

            const realAccount =
                database.getAccount(
                    request.to
                );

            return {
                requestId: request.id,
                steamId: request.to,
                name: sender?.name ??
                    (
                        realAccount
                            ? realAccount.name + " (Teste)"
                            : "Jogador (Teste)"
                    ),
                code: sender?.code ?? "__DEV__",
                simulated: true
            };
        }

        return {
            requestId: request.id,
            steamId: request.from,
            name: sender?.name ?? "Jogador",
            code: sender?.code ?? "------",
            simulated: false
        };
    }

    // --------------------------------------------------
    // ENVIAR LISTA DE AMIGOS
    // --------------------------------------------------

    private sendFriends(
        client: Client,
        steamId: string
    ) {

        const friends = database.getFriends(
            steamId
        ).map(account => {

            const simulated =
                database.isTestAccount(
                    account.steamId
                );

            return {
                steamId: account.steamId,
                name: account.name,
                code: account.code,
                online: simulated
                    ? false
                    : this.isOnline(account.steamId),
                simulated
            };
        });

        // Enviar nos dois formatos para manter
        // compatibilidade com o GameMaker.
        client.send(
            "friend_list_result",
            {
                friends,
                friendsJson: JSON.stringify(friends)
            }
        );
    }

    // --------------------------------------------------
    // ENVIAR MENSAGEM A UMA CONTA
    // --------------------------------------------------

    private sendToAccount(
        steamId: string,
        type: string,
        data: unknown,
        excludeSessionId?: string,
        excludeInstanceId?: string
    ) {

        for (const client of this.clients) {

            if (
                client.sessionId !== excludeSessionId &&
                (
                    !excludeInstanceId ||
                    this.sessionInstances.get(
                        client.sessionId
                    ) !== excludeInstanceId
                ) &&
                this.sessions.get(
                    client.sessionId
                ) === steamId
            ) {

                client.send(type, data);
            }
        }
    }

    // --------------------------------------------------
    // ATUALIZAR AMIGOS CONECTADOS
    // --------------------------------------------------

    private broadcastFriends(steamId: string) {

        const affected = new Set<string>();

        affected.add(steamId);

        for (
            const account of database.getFriends(
                steamId
            )
        ) {
            affected.add(account.steamId);
        }

        for (const client of this.clients) {

            const connectedId =
                this.sessions.get(
                    client.sessionId
                );

            if (
                connectedId &&
                affected.has(connectedId)
            ) {

                this.sendFriends(
                    client,
                    connectedId
                );
            }
        }
    }

    // --------------------------------------------------
    // TEXTO DE ERRO
    // --------------------------------------------------

    private errorText(error: unknown) {

        return error instanceof Error
            ? error.message
            : "Erro desconhecido.";
    }
