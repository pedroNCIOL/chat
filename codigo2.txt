obj_partida draw gui:"/// obj_partida - Draw GUI

var _gui_w = display_get_gui_width();
var _gui_h = display_get_gui_height();

// --------------------------------------------------
// IDENTIFICAR JOGADOR LOCAL
// --------------------------------------------------

var _net = instance_find(obj_network_manager, 0);
var _meu_session_id = "";

if (_net != noone) {
    _meu_session_id = _net.net_session_id;
}

// --------------------------------------------------
// INICIALIZAR ANIMAÇÕES
// --------------------------------------------------

if (!variable_instance_exists(id, "partida_icone_multiplicadores")) {
    partida_icone_multiplicadores = [1, 1, 1, 1];
}

if (!variable_instance_exists(id, "partida_barra_back_desloc_x")) {
    partida_barra_back_desloc_x = [0, 0];
    partida_barra_back_desloc_y = [0, 0];
}

// --------------------------------------------------
// HUD DOS JOGADORES
// --------------------------------------------------

var _barra_escala = 3.5;

var _barra_largura = sprite_get_width(spr_barra) * _barra_escala;
var _barra_altura = sprite_get_height(spr_barra) * _barra_escala;

for (var _barra_index = 0; _barra_index < 4; _barra_index++) {

    var _barra_cor = "";
    var _barra_x = 0;
    var _barra_y = 0;

    switch (_barra_index) {

        case 0:
            _barra_cor = "red";
            _barra_x = 0;
            _barra_y = -135;
        break;

        case 1:
            _barra_cor = "green";
            _barra_x = _gui_w - _barra_largura;
            _barra_y = -135;
        break;

        case 2:
            _barra_cor = "yellow";
            _barra_x = 0;
            _barra_y = _gui_h - _barra_altura + 137;
        break;

        case 3:
            _barra_cor = "blue";
            _barra_x = _gui_w - _barra_largura;
            _barra_y = _gui_h - _barra_altura + 137;
        break;
    }

    var _nome_jogador = "";
    var _dinheiro_jogador = 0;
    var _icone_jogador = -1;
    var _session_id_jogador = "";

    for (var _player_index = 0; _player_index < array_length(partida_jogadores); _player_index++) {

        if (partida_jogadores[_player_index].color == _barra_cor) {

            _nome_jogador = partida_jogadores[_player_index].name;
            _dinheiro_jogador = partida_jogadores[_player_index].money;
            _icone_jogador = partida_jogadores[_player_index].icon;
            _session_id_jogador = partida_jogadores[_player_index].session_id;

            break;
        }
    }

    if (_nome_jogador == "") continue;

    var _icone_e_turno = (
        _session_id_jogador == partida_turno_atual &&
        partida_turno_atual != ""
    );

    var _icone_largura = sprite_get_width(_icone_jogador);
    var _icone_altura = sprite_get_height(_icone_jogador);

    var _icone_escala = min(
        1,
        (_barra_altura * 0.6) / _icone_altura
    ) * 2.5;

    var _icone_multiplicador_alvo = 1;

    if (_icone_e_turno) {
        _icone_multiplicador_alvo = 1.2;
    }

    var _icone_multiplicador_atual =
        partida_icone_multiplicadores[_barra_index];

    _icone_multiplicador_atual = lerp(
        _icone_multiplicador_atual,
        _icone_multiplicador_alvo,
        0.1
    );

    if (abs(_icone_multiplicador_atual - _icone_multiplicador_alvo) < 0.001) {
        _icone_multiplicador_atual = _icone_multiplicador_alvo;
    }

    partida_icone_multiplicadores[_barra_index] =
        _icone_multiplicador_atual;

    var _icone_escala_final =
        _icone_escala * _icone_multiplicador_atual;

    var _icone_largura_final =
        _icone_largura * _icone_escala_final;

    var _icone_altura_final =
        _icone_altura * _icone_escala_final;

    var _icone_escala_x = _icone_escala_final;

    if (_barra_index == 1 || _barra_index == 3) {
        _icone_escala_x = -_icone_escala_final;
    }

    var _icone_margem = 12;

    var _icone_x;
    var _icone_y;

    switch (_barra_index) {

        case 0:
            _icone_x = _barra_x - 200 + _barra_largura * 0.55;
            _icone_y = _barra_y + 100;
        break;

        case 1:
            _icone_x = _barra_x + 200 + _barra_largura * 0.45;
            _icone_y = _barra_y + 100;
        break;

        case 2:
            _icone_x = _barra_x - 200 + _barra_largura * 0.55;
            _icone_y = _barra_y + _barra_altura - _icone_altura_final - 134;
        break;

        case 3:
            _icone_x = _barra_x + 200 + _barra_largura * 0.45;
            _icone_y = _barra_y + _barra_altura - _icone_altura_final - 134;
        break;
    }

    // --------------------------------------------------
    // ANIMAÇÃO DO SPR_BARRA_BACK
    // SOMENTE JOGADORES VERMELHO E VERDE
    // --------------------------------------------------

    if (_barra_index == 0 || _barra_index == 1) {

        var _back_index = _barra_index;

        var _back_alvo_x = 0;
        var _back_alvo_y = 0;

        var _back_recuo_x = _barra_largura * 0.2;
        var _back_recuo_y = _barra_altura * 0.2;

        if (_icone_e_turno) {

            var _icone_esquerda = _icone_x;

            if (_barra_index == 1) {
                _icone_esquerda -= _icone_largura_final;
            }

            var _back_distancia_x = abs(_icone_esquerda - _barra_x);

            if (_barra_index == 0) {
                _back_alvo_x = _back_distancia_x * 0.002;
            }
            else {
                _back_alvo_x = -_back_distancia_x * 0.002;
            }

            _back_alvo_y = (_icone_y - _barra_y) * 0.01;
        }
        else {

            if (_barra_index == 0) {
                _back_alvo_x = -_back_recuo_x;
            }
            else {
                _back_alvo_x = _back_recuo_x;
            }

            _back_alvo_y = -_back_recuo_y;
        }

        partida_barra_back_desloc_x[_back_index] = lerp(
            partida_barra_back_desloc_x[_back_index],
            _back_alvo_x,
            0.2
        );

        partida_barra_back_desloc_y[_back_index] = lerp(
            partida_barra_back_desloc_y[_back_index],
            _back_alvo_y,
            0.2
        );

        if (abs(partida_barra_back_desloc_x[_back_index] - _back_alvo_x) < 0.1) {
            partida_barra_back_desloc_x[_back_index] = _back_alvo_x;
        }

        if (abs(partida_barra_back_desloc_y[_back_index] - _back_alvo_y) < 0.1) {
            partida_barra_back_desloc_y[_back_index] = _back_alvo_y;
        }

        draw_sprite_ext(
            spr_barra_back,
            _barra_index,
            _barra_x + partida_barra_back_desloc_x[_back_index],
            _barra_y + partida_barra_back_desloc_y[_back_index],
            _barra_escala,
            _barra_escala,
            0,
            c_white,
            1
        );
    }


    // --------------------------------------------------
    // DESENHAR FOTO DA STEAM
    // DEPOIS DO SPR_BARRA_BACK
    // ANTES DO SPR_BARRA
    // --------------------------------------------------

    var _avatar_hud = instance_find(
        obj_partida_steam_avatars,
        0
    );
	
    if (_avatar_hud != noone) {
	gpu_set_texfilter(true);
        _avatar_hud.avatar_desenhar(
            _session_id_jogador,
            _barra_index,
            _barra_x,
            _barra_y,
            _barra_largura,
            _barra_altura
        );
    
	gpu_set_texfilter(false);
	}

    // --------------------------------------------------
    // DESENHAR BARRA NORMAL
    // POR CIMA DA FOTO DA STEAM
    // --------------------------------------------------

    draw_sprite_ext(
        spr_barra,
        _barra_index,
        _barra_x,
        _barra_y,
        _barra_escala,
        _barra_escala,
        0,
        c_white,
        1
    );


    // --------------------------------------------------
    // DESENHAR ÍCONE ORIGINAL
    // --------------------------------------------------

    draw_sprite_ext(
        _icone_jogador,
        0,
        _icone_x,
        _icone_y,
        _icone_escala_x,
        _icone_escala_final,
        0,
        c_white,
        1
    );

    // --------------------------------------------------
    // CÓPIA ESCURECIDA DO ÍCONE
    // --------------------------------------------------

    var _icone_alpha_escuro = 0.5;

    if (_icone_e_turno) {
        _icone_alpha_escuro = 0;
    }

    if (_icone_alpha_escuro > 0) {

        draw_sprite_ext(
            _icone_jogador,
            0,
            _icone_x,
            _icone_y,
            _icone_escala_x,
            _icone_escala_final,
            0,
            c_black,
            _icone_alpha_escuro
        );
    }

    // --------------------------------------------------
    // POSICIONAMENTO DO NOME
    // --------------------------------------------------

    var tx;
    var ty;

    if (_barra_index == 0 || _barra_index == 2) {
        tx = 85;
    }
    else {
        tx = -85;
    }

    if (_barra_index <= 1) {
        ty = -30;
    }
    else {
        ty = -15;
    }

    var _texto_x = tx + _barra_x + _barra_largura / 2;
    var _texto_y = ty + _barra_y + _barra_altura / 2;

    draw_set_halign(fa_center);
    draw_set_valign(fa_middle);

    var color_text_bar = [
        make_colour_rgb(36,21,39),
        make_colour_rgb(25,51,45),
        make_colour_rgb(52,28,39),
        make_colour_rgb(23,32,56)
    ];

    draw_set_color(c_white);
    draw_set_font(fnt_pixelari);

    draw_text_transformed_contour(
        _texto_x,
        _texto_y,
        _nome_jogador,
        .2,
        .2,
        0,
        4,
        c_white,
        color_text_bar[_barra_index]
    );

    var _mx;
    var _my;

    if (_barra_index == 0) {
        _mx = 10;
        _my = 50;
    }
    else if (_barra_index == 1) {
        _mx = -10;
        _my = 50;
    }
    else if (_barra_index == 2) {
        _mx = 10;
        _my = 55;
    }
    else if (_barra_index == 3) {
        _mx = 0;
        _my = 55;
    }

    // Dinheiro do jogador.
    var _texto_dinheiro = partida_formatar_dinheiro(_dinheiro_jogador);

    draw_text_transformed_contour(
        _texto_x + _mx,
        _texto_y + _my,
        _texto_dinheiro,
        .16,
        .16,
        0,
        3,
        c_white,
        color_text_bar[_barra_index]
    );

    // --------------------------------------------------
    // ANIMAÇÃO DE GANHO/PERDA DE DINHEIRO
    // --------------------------------------------------

    for (
        var _ganho_index = 0;
        _ganho_index < array_length(partida_dinheiro_animacoes);
        _ganho_index++
    ) {

        var _ganho = partida_dinheiro_animacoes[_ganho_index];

        if (
            _ganho.session_id == _session_id_jogador &&
            _ganho.start_ms >= 0 &&
            !partida_popup_ativo
        ) {

            var _tempo_ganho = current_time - _ganho.start_ms;

            var _alpha_ganho = 1;

            if (_tempo_ganho > 1000) {
                _alpha_ganho = 1 - ((_tempo_ganho - 1000) / 1000);
            }

            _alpha_ganho = clamp(_alpha_ganho, 0, 1);

            var _direcao_ganho = 1;

            if (_barra_index >= 2) {
                _direcao_ganho = -1;
            }

            var _movimento_ganho = (_tempo_ganho / 2000) * 35;

            var _offset_ganho_y = -30;

            if (_barra_index <= 1) {
                _offset_ganho_y = 5;
            }

            var _ganho_x = _texto_x + _mx;

            var _ganho_y =
                _texto_y + _my + _offset_ganho_y +
                (_movimento_ganho * _direcao_ganho);

            var _sinal_dinheiro = "+";

            if (_ganho.negative) {
                _sinal_dinheiro = "-";
            }

            var _texto_ganho =
                _sinal_dinheiro +
                partida_formatar_dinheiro(_ganho.amount);

            draw_set_alpha(_alpha_ganho);
            draw_set_font(fnt_pixelari);
            draw_set_halign(fa_center);
            draw_set_valign(fa_middle);

            draw_text_transformed_contour(
                _ganho_x,
                _ganho_y,
                _texto_ganho,
                .16,
                .16,
                0,
                3,
                c_white,
                color_text_bar[_barra_index]
            );

            draw_set_alpha(1);
        }
    }
}

// --------------------------------------------------
// INFORMAÇÃO DO TURNO
// --------------------------------------------------

draw_set_halign(fa_center);
draw_set_valign(fa_middle);

var _turn_text = "Aguardando jogadores";

for (var _i = 0; _i < array_length(partida_jogadores); _i++) {

    if (
        partida_jogadores[_i].session_id ==
        partida_turno_atual
    ) {

        _turn_text = "Turno de: " + partida_jogadores[_i].name;
        break;
    }
}

draw_set_font(fnt_pixelari);

draw_text_transformed_contour(
    _gui_w / 2,
    42,
    _turn_text,
    .2,.2,
    0,
    4,
    c_white,
    make_colour_rgb(16,20,31)
);

draw_set_font(-1);

// --------------------------------------------------
// ELEMENTOS DA PARTIDA
// OCULTAR DURANTE OS POPUPS E MENU DA PRISÃO
// --------------------------------------------------

if (!partida_popup_ativo && !partida_prisao_menu_ativo) {

    // --------------------------------------------------
    // RESULTADO DOS DADOS
    // --------------------------------------------------

    if (
        partida_tem_rolagem &&
        (
            !partida_jogador_preso ||
            partida_prisao_tentativa_em_andamento
        )
    ) {

        var _contador_escala = partida_escala_dados;

        var _contador_largura =
            sprite_get_width(spr_dado_contador) * _contador_escala;

        var _contador_altura =
            sprite_get_height(spr_dado_contador) * _contador_escala;

        var _contador_espacamento = partida_espacamento_dados;

        var _contador_grupo_largura =
            (_contador_largura * 2) +
            _contador_espacamento;

        var _contador_x =
            (_gui_w - _contador_grupo_largura) / 2;

        var _contador_y =
            _gui_h / 2 - _contador_altura / 2 + 35;

        draw_sprite_ext(
            spr_dado_contador,
            partida_dado1 - 1,
            _contador_x + sprite_get_xoffset(spr_dado_contador) * _contador_escala,
            _contador_y + sprite_get_yoffset(spr_dado_contador) * _contador_escala,
            _contador_escala,
            _contador_escala,
            0,
            c_white,
            1
        );

        var _contador_direita_x =
            _contador_x +
            _contador_largura +
            _contador_espacamento;

        draw_sprite_ext(
            spr_dado_contador,
            partida_dado2 - 1,
            _contador_direita_x + sprite_get_xoffset(spr_dado_contador) * _contador_escala,
            _contador_y + sprite_get_yoffset(spr_dado_contador) * _contador_escala,
            _contador_escala,
            _contador_escala,
            0,
            c_white,
            1
        );

        // --------------------------------------------------
        // NÚMERO ANIMADO DO RESULTADO
        // --------------------------------------------------

        var _valor_contador = partida_contador_resultado;

        if (partida_contador_ativo) {
            _valor_contador = partida_contador_valor;
        }

        draw_set_font(fnt_pixelari);
        draw_set_halign(fa_center);
        draw_set_valign(fa_middle);

        draw_text_transformed_contour(
            _gui_w / 2,
            _contador_y - 35,
            string(_valor_contador),
            0.35,
            0.35,
            0,
            4,
            c_white,
            make_colour_rgb(16,20,31)
        );

        draw_set_font(-1);
    }

    // --------------------------------------------------
    // DESENHAR OS DOIS DADOS ANIMADOS
    // --------------------------------------------------

    if (partida_dados_visiveis) {

        var _escala = partida_escala_dados;

        var _dado_largura = sprite_get_width(spr_dado1) * _escala;
        var _dado_altura = sprite_get_height(spr_dado1) * _escala;

        var _grupo_largura =
            (_dado_largura * 2) +
            partida_espacamento_dados;

        var _dados_x = (_gui_w - _grupo_largura) / 2;
        var _dados_y = (_gui_h - _dado_altura) / 2;

        var _indice_dados = partida_dados_indice_animacao;
        var _angulo_dados = partida_dados_angulo;

        draw_sprite_ext(
            spr_dado1,
            _indice_dados,
            _dados_x + sprite_get_xoffset(spr_dado1) * _escala,
            _dados_y + sprite_get_yoffset(spr_dado1) * _escala,
            _escala,
            _escala,
            _angulo_dados,
            c_white,
            1
        );

        var _direita_x =
            _dados_x +
            _dado_largura +
            partida_espacamento_dados;

        draw_sprite_ext(
            spr_dado2,
            _indice_dados,
            _direita_x + sprite_get_xoffset(spr_dado2) * _escala,
            _dados_y + sprite_get_yoffset(spr_dado2) * _escala,
            _escala,
            _escala,
            _angulo_dados,
            c_white,
            1
        );
    }

    // --------------------------------------------------
    // BOTÃO PRINCIPAL — ENCERRAR TURNO
    // --------------------------------------------------

    var _button_x1 = _gui_w / 2 - 150;
    var _button_y1 = _gui_h - 115;

    var _button_x2 = _gui_w / 2 + 150;
    var _button_y2 = _gui_h - 45;

    if (partida_botao_ativo) {
        draw_set_color(make_color_rgb(45, 120, 65));
    }
    else {
        draw_set_color(make_color_rgb(65, 65, 65));
    }

    draw_roundrect(
        _button_x1,
        _button_y1,
        _button_x2,
        _button_y2,
        false
    );

    draw_set_color(c_white);

    if (
        partida_contagem_rolagem >= 0 &&
        partida_e_meu_turno &&
        !partida_tem_rolagem &&
        !partida_rolagem_solicitada
    ) {

        draw_text(
            _gui_w / 2,
            (_button_y1 + _button_y2) / 2 - 12,
            partida_botao_texto
        );

        draw_set_color(c_yellow);

        draw_text(
            _gui_w / 2,
            (_button_y1 + _button_y2) / 2 + 13,
            string(partida_contagem_rolagem)
        );

    }
    else {

        draw_text(
            _gui_w / 2,
            (_button_y1 + _button_y2) / 2,
            partida_botao_texto
        );
    }
}

// --------------------------------------------------
// POPUP DO PRIMEIRO JOGADOR
// --------------------------------------------------

if (partida_mostrar_primeiro_jogador) {

    var _popup_centro_x = _gui_w / 2;
    var _popup_centro_y = _gui_h / 2;

    var _popup_escala = 4;

    var _popup_largura =
        sprite_get_width(spr_msg) * _popup_escala;

    var _popup_altura =
        sprite_get_height(spr_msg) * _popup_escala;

    draw_sprite_ext(
        spr_msg,
        0,
        _popup_centro_x,
        _popup_centro_y,
        _popup_escala,
        _popup_escala,
        0,
        c_white,
        1
    );

    var _popup_icone = -1;

    for (
        var _popup_index = 0;
        _popup_index < array_length(partida_jogadores);
        _popup_index++
    ) {

        if (
            partida_jogadores[_popup_index].session_id ==
            partida_primeiro_jogador_session_id
        ) {

            _popup_icone = partida_jogadores[_popup_index].icon;
            break;
        }
    }

    var _texto_primeiro_jogador = "";

    if (
        partida_primeiro_jogador_session_id ==
        _meu_session_id
    ) {

        _texto_primeiro_jogador =
            "Voce comecara a partida!";

    }
    else {

        _texto_primeiro_jogador =
            partida_primeiro_jogador_nome +
            "\ncomecara a partida!";

    }

    draw_set_font(fnt_pixelari);
    draw_set_halign(fa_center);
    draw_set_valign(fa_middle);

    draw_text_transformed_contour(
        _popup_centro_x,
        _popup_centro_y - _popup_altura * 0.3,
        _texto_primeiro_jogador,
        0.25,
        0.25,
        0,
        4,
        c_white,
        make_colour_rgb(16,20,31)
    );

    if (_popup_icone != -1) {

        var _popup_icone_largura = sprite_get_width(_popup_icone);
        var _popup_icone_altura = sprite_get_height(_popup_icone);

        var _popup_icone_escala = min(
            (_popup_largura * 0.25) / _popup_icone_largura,
            (_popup_altura * 0.28) / _popup_icone_altura
        ) * 2.4;

        var _popup_icone_largura_final =
            _popup_icone_largura * _popup_icone_escala;

        var _popup_icone_altura_final =
            _popup_icone_altura * _popup_icone_escala;

        var _popup_icone_x =
            _popup_centro_x - _popup_icone_largura_final / 2;

        var _popup_icone_y =
            _popup_centro_y + _popup_altura * 0.06;

        draw_sprite_ext(
            _popup_icone,
            0,
            _popup_icone_x,
            _popup_icone_y-100,
            _popup_icone_escala,
            _popup_icone_escala,
            0,
            c_white,
            1
        );
    }
}

// --------------------------------------------------
// POPUP DA ROLAGEM EXTRA
// --------------------------------------------------

if (partida_mostrar_rolagem_extra) {

    var _extra_popup_centro_x = _gui_w / 2;
    var _extra_popup_centro_y = _gui_h / 2;

    var _extra_popup_escala = 4;

    var _extra_popup_largura =
        sprite_get_width(spr_msg) * _extra_popup_escala;

    var _extra_popup_altura =
        sprite_get_height(spr_msg) * _extra_popup_escala;

    draw_sprite_ext(
        spr_msg,
        0,
        _extra_popup_centro_x,
