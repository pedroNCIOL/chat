                Math.random() * group.positions.length

            const specialCell = this.board[currentIndex];

            this.board[currentIndex] = previousCell;
            this.board[targetPosition] = specialCell;

            console.log(
                `${group.name}: posição ${currentIndex} -> ${targetPosition}`

            console.log(
                `Casa deslocada: ${previousCell.name} (${targetPosition} -> ${currentIndex})`

        }

    private queueMovementAnnouncement(
        sessionId: string,
        type: "prison" | "extra" | "none",
        reason?: ExtraRollReason
    ) {
        this.pendingMovementAnnouncement = {
            sessionId,
            type,

        this.turnTimeout?.clear();
        this.turnTimeout = undefined;

        this.state.turnDeadline = 0;
        this.state.movementAnnouncementPending = true;

        console.log(
            `Aguardando todos os jogadores confirmarem a chegada de ${sessionId} antes de liberar a próxima etapa.`

    // --------------------------------------------------
    // VERIFICAR SE TODOS OS CLIENTES CHEGARAM
    // --------------------------------------------------

    private finishMovementAnnouncementIfReady(
        departedSessionId?: string
    ) {
        const pending = this.pendingMovementAnnouncement;

        if (!pending) {
            return;
        }

        const connectedSessionIds = this.clients
            .map(client => client.sessionId)
            .filter(sessionId => sessionId !== departedSessionId);

        const everyoneArrived = connectedSessionIds.every(
            sessionId => pending.arrivedClients.has(sessionId)
        );

        if (!everyoneArrived) {
            return;
        }

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        if (!this.state.players.has(pending.sessionId)) {
            return;
        }

        console.log(
            `Todos os jogadores confirmaram a chegada de ${pending.sessionId}.`
        );

        if (pending.type === "prison") {
            this.showPrisonAnnouncement(pending.sessionId);
        }
        else if (pending.type === "extra" && pending.reason) {
            this.showExtraRollAnnouncement(
                pending.sessionId,
                pending.reason
            );
        }
    }

            this.state.movementAnnouncementPending ||
            this.state.showFirstPlayerMessage ||
            this.state.showExtraRollMessage ||

    private startCurrentTurnTimer(sessionId: string) {
        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;

            if (this.state.currentTurn !== sessionId) {
                return;

            if (!currentPlayer) {
                return;

            if (!this.state.hasRolled) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a vez.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);

    // --------------------------------------------------

    private isValidColor(color: string): color is PlayerColor {
        return this.PLAYER_COLORS.includes(color as PlayerColor);
    }

    // --------------------------------------------------
    // ATRIBUIR COR AO JOGADOR
    // --------------------------------------------------

    private assignPlayerColor(preferredColor: string): PlayerColor {
        const occupiedColors = new Set<string>();

        this.state.players.forEach((existingPlayer) => {
            occupiedColors.add(existingPlayer.color);
        });

        if (
            this.isValidColor(preferredColor) &&
            !occupiedColors.has(preferredColor)
        ) {
            return preferredColor;
        }

        for (const color of this.PLAYER_COLORS) {
            if (!occupiedColors.has(color)) {
                return color;
            }
        }

        return "red";
    }

    // --------------------------------------------------

            return;
        }

        if (this.state.currentTurn !== sessionId) {
            return;

        const player = this.state.players.get(sessionId);

        if (!player || !player.extraRollAvailable) {
            return;
        }

        this.turnTimeout?.clear();

        this.state.turnDeadline =
            this.clock.currentTime + TURN_DURATION;

        this.turnTimeout = this.clock.setTimeout(() => {
            if (this.state.status !== "playing") {
                return;

            if (this.state.currentTurn !== sessionId) {
                return;
            }

            const currentPlayer = this.state.players.get(sessionId);

            if (
                currentPlayer?.extraRollAvailable &&
                !this.state.hasRolled
            ) {
                console.log(
                    `Tempo esgotado! Jogador ${sessionId} perdeu a rolagem extra.`
                );

                this.nextTurn();
            }
        }, TURN_DURATION);

    // --------------------------------------------------

    // --------------------------------------------------
    // GERAR CÓDIGO DE 6 CARACTERES
    // --------------------------------------------------

    private generateCode(): string {
        let code = "";

        for (let i = 0; i < 6; i++) {
            const index = Math.floor(
                Math.random() * this.CODE_CHARACTERS.length
            );

            code += this.CODE_CHARACTERS[index];
        }

        return code;
    }

    // --------------------------------------------------
    // GARANTIR CÓDIGO ÚNICO
    // --------------------------------------------------

    private async generateRoomCode(): Promise<string> {
        const existingCodes = await this.presence.smembers(
            this.CODE_CHANNEL
        );

        let code = "";

        do {
            code = this.generateCode();
        } while (existingCodes.includes(code));

        await this.presence.sadd(this.CODE_CHANNEL, code);

        return code;
    }

    // --------------------------------------------------
    // CANCELAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private clearAutoStartCountdown() {
        this.autoStartInterval?.clear();
        this.autoStartInterval = undefined;

        this.state.autoStartCountdown = 0;
    }

    // --------------------------------------------------
    // REINICIAR CONTAGEM AUTOMÁTICA
    // --------------------------------------------------

    private resetAutoStartCountdown() {
        if (this.mode !== "public") {
            this.clearAutoStartCountdown();
            return;
        }

        if (this.state.status !== "waiting") {
            this.clearAutoStartCountdown();
            return;
        }

        this.clearAutoStartCountdown();

        if (this.state.players.size < 2) {
            return;
        }

        this.state.autoStartCountdown = AUTO_START_DURATION;

        console.log(
            `Contagem automática iniciada: ${this.roomId} (${AUTO_START_DURATION}s)`
        );

        this.autoStartInterval = this.clock.setInterval(() => {
            if (this.state.status !== "waiting") {
                this.clearAutoStartCountdown();
                return;
            }

            if (this.state.players.size < 2) {
                this.clearAutoStartCountdown();
                return;
            }

            this.state.autoStartCountdown--;

            console.log(
                `Sala ${this.roomId}: ${this.state.autoStartCountdown}s`
            );

            if (this.state.autoStartCountdown <= 0) {
                this.clearAutoStartCountdown();
                this.startGame();
            }
        }, 1000);
    }

    // --------------------------------------------------
    // INICIAR PARTIDA
    // --------------------------------------------------

    private startGame() {
        if (this.state.status !== "waiting") {
            return;
        }

        if (this.state.players.size < 2) {
            return;
        }

        this.clearAutoStartCountdown();

        this.randomizeSpecialCities();
        this.broadcastBoard();

        this.state.status = "playing";

        this.state.currentTurn = "";
        this.state.turnDeadline = 0;
        this.state.turnCount = 0;

        this.state.hasRolled = false;
        this.state.hasMoved = false;

        this.state.dice1 = 0;
        this.state.dice2 = 0;
        this.state.lastRollBy = "";

        this.pendingMovementAnnouncement = undefined;
        this.state.movementAnnouncementPending = false;

        this.state.showFirstPlayerMessage = false;
        this.state.firstPlayerSessionId = "";
        this.state.firstPlayerName = "";

        this.state.showExtraRollMessage = false;
        this.state.extraRollPlayerName = "";
        this.state.extraRollPlayerSessionId = "";
        this.state.extraRollReason = "";

        this.state.showPrisonMessage = false;
        this.state.prisonPlayerName = "";
        this.state.prisonPlayerSessionId = "";

        this.playersResetForNewGame();

        void this.setMetadata({
            mode: this.mode,
            status: this.state.status
        });

        this.lock();

        console.log("Partida iniciada:", this.roomId);
        console.log("Aguardando 3 segundos para iniciar o anúncio...");

        this.startDelay?.clear();

        this.startDelay = this.clock.setTimeout(() => {
            this.startDelay = undefined;

            if (this.state.status !== "playing") {
                return;
            }

            console.log("Animação concluída! Sorteando primeiro jogador.");

            this.showFirstPlayerAnnouncement();
        }, GAME_START_DELAY);
    }

    // --------------------------------------------------
    // REINICIAR ESTADOS DOS JOGADORES
    // --------------------------------------------------

    private playersResetForNewGame() {
        this.state.players.forEach((player) => {
            player.extraRollAvailable = false;
            player.inPrison = false;
            player.prisonAttempts = 0;
            player.debtsAvailable = 1;
        });
    }

                client.send(

                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            const requestedColor = String(data?.color ?? "");

            if (!this.isValidColor(requestedColor)) {
                client.send("color_error", "Cor inválida.");
                return;
            }

            const occupied = [...this.state.players.keys()].some(
                sessionId => {
                    if (sessionId === client.sessionId) {
                        return false;
                    }

                    const otherPlayer = this.state.players.get(sessionId);

                    return otherPlayer?.color === requestedColor;
                }
            );

            if (occupied) {
                client.send(
                    "color_error",
                    "Essa cor já está sendo utilizada por outro jogador."
                );
                return;
            }

            player.color = requestedColor;

            console.log(
                `${player.name} escolheu a cor: ${requestedColor}`
            );
        },

        start: (client: Client) => {
            if (this.state.status !== "waiting") {
                client.send("start_error", "A partida já começou.");
                return;
            }

            if (this.mode !== "private") {
                client.send(
                    "start_error",
                    "Partidas públicas começam automaticamente."
                );
                return;
            }

            if (client.sessionId !== this.state.hostSessionId) {
                client.send(
                    "start_error",
                    "Somente o anfitrião pode iniciar."
                );
                return;
            }

            if (this.state.players.size < 2) {
                client.send(
                    "start_error",
                    "São necessários pelo menos 2 jogadores."
                );
                return;
            }

            this.startGame();
        },

        // --------------------------------------------------
        // ENVIAR TABULEIRO
        // --------------------------------------------------

        get_board: (client: Client) => {
            const boardData = this.board.map((cell, index) => ({
                position: index,
                name: cell.name,
                type: cell.type,
                spriteIndex: cell.spriteIndex
            }));

            client.send("board_data", JSON.stringify(boardData));

            console.log(
                `Tabuleiro enviado para ${client.sessionId}: ${boardData.length} casas.`
            );
        },

        // --------------------------------------------------
        // CONFIRMAR CHEGADA DA ANIMAÇÃO DE MOVIMENTO
        // --------------------------------------------------

        movement_arrived: (client: Client) => {
            const pending = this.pendingMovementAnnouncement;

            if (!pending) {
                return;
            }

            if (!this.state.players.has(client.sessionId)) {
                return;
            }

            pending.arrivedClients.add(client.sessionId);

            console.log(
                `Chegada confirmada: ${client.sessionId} (${pending.arrivedClients.size}/${this.clients.length})`
            );

            this.finishMovementAnnouncementIfReady();
        },

        // --------------------------------------------------
        // ROLAR DADOS NORMALMENTE
        // --------------------------------------------------

        roll_dice: (client: Client) => {
            if (this.state.status !== "playing") {
                client.send(
                    "action_error",
                    "A partida ainda não começou."
                );
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                client.send(
                    "action_error",
                    "Você precisa resolver sua situação na prisão primeiro."
                );
                return;
            }

            if (this.state.hasRolled) {
                client.send(
                    "action_error",
                    "Você já rolou os dados nesta rodada."
                );
                return;
            }

            if (this.state.hasMoved && !player.extraRollAvailable) {
                client.send(
                    "action_error",
                    "Você já movimentou nesta rodada."
                );
                return;
            }

            this.state.dice1 = Math.floor(Math.random() * 6) + 1;
            this.state.dice2 = Math.floor(Math.random() * 6) + 1;

            this.state.lastRollBy = client.sessionId;
            this.state.hasRolled = true;

            player.extraRollAvailable = false;
            this.state.hasMoved = false;

            console.log(
                `Dados de ${client.sessionId}: ${this.state.dice1} + ${this.state.dice2}`
            );
        },

        // --------------------------------------------------
        // TENTAR SAIR DA PRISÃO COM DUPLAS
        // --------------------------------------------------

        prison_try_doubles: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive() || this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (this.state.hasRolled || this.state.hasMoved) {
                client.send(
                    "action_error",
                    "Não é possível tentar sair da prisão agora."
                );
                return;
            }

            const dice1 = Math.floor(Math.random() * 6) + 1;
            const dice2 = Math.floor(Math.random() * 6) + 1;

            const rolledDoubles = dice1 === dice2;

            player.prisonAttempts++;

            // Os dados são compartilhados com o cliente para exibição.
            // O jogador continua preso durante esse período.
            this.state.dice1 = dice1;
            this.state.dice2 = dice2;
            this.state.lastRollBy = client.sessionId;
            this.state.hasRolled = true;
            this.state.hasMoved = false;

            // Pausar o limite do turno enquanto o resultado aparece.
            this.turnTimeout?.clear();
            this.turnTimeout = undefined;
            this.state.turnDeadline = 0;

            console.log("==================================");
            console.log("TENTATIVA DE SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log(`DADOS: ${dice1} + ${dice2}`);
            console.log("DUPLAS:", rolledDoubles);
            console.log(
                `TENTATIVA: ${player.prisonAttempts}/${MAX_PRISON_ATTEMPTS}`
            );
            console.log("==================================");

            // Aguardar a HUD exibir os dados e o resultado.
            this.prisonAttemptDelay?.clear();

            this.prisonAttemptDelay = this.clock.setTimeout(() => {
                this.prisonAttemptDelay = undefined;

                if (this.state.status !== "playing") {
                    return;
                }

                if (this.state.currentTurn !== client.sessionId) {
                    return;
                }

                const currentPlayer = this.state.players.get(client.sessionId);

                if (!currentPlayer || !currentPlayer.inPrison) {
                    return;
                }

                // Conseguiu duplas: liberar e permitir uma rolagem normal.
                if (rolledDoubles) {
                    console.log(
                        `${currentPlayer.name} conseguiu duplas e saiu da prisão.`
                    );

                    this.releasePlayerFromPrison(client.sessionId);
                    return;
                }

                // Terceira tentativa: libertar automaticamente.
                if (currentPlayer.prisonAttempts >= MAX_PRISON_ATTEMPTS) {
                    console.log(
                        `${currentPlayer.name} atingiu o limite de tentativas e foi liberado automaticamente.`
                    );

                    this.releasePlayerFromPrison(client.sessionId);
                    return;
                }

                // Falhou: informar e encerrar o turno.
                client.send(
                    "prison_attempt_failed",
                    {
                        attempts: currentPlayer.prisonAttempts,
                        maxAttempts: MAX_PRISON_ATTEMPTS
                    }
                );

                console.log(
                    `${currentPlayer.name} não conseguiu duplas. A próxima tentativa será no próximo turno.`
                );

                this.nextTurn();
            }, PRISON_ATTEMPT_RESULT_DURATION);
        },

        // --------------------------------------------------
        // PAGAR PARA SAIR DA PRISÃO
        // --------------------------------------------------

        prison_pay: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (player.money < PRISON_ESCAPE_COST) {
                client.send(
                    "action_error",
                    "Você não possui dinheiro suficiente para pagar a saída da prisão."
                );
                return;
            }

            player.money -= PRISON_ESCAPE_COST;

            console.log("==================================");
            console.log("JOGADOR PAGOU PARA SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log("VALOR PAGO:", PRISON_ESCAPE_COST);
            console.log("DINHEIRO RESTANTE:", player.money);
            console.log("==================================");

            this.releasePlayerFromPrison(client.sessionId);
        },

        // --------------------------------------------------
        // USAR DÍVIDA PARA SAIR DA PRISÃO
        // --------------------------------------------------

        prison_use_debt: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.prisonAttemptDelay) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                client.send("action_error", "Não é seu turno.");
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player || !player.inPrison) {
                client.send(
                    "action_error",
                    "Você não está na prisão."
                );
                return;
            }

            if (player.debtsAvailable <= 0) {
                client.send(
                    "action_error",
                    "Você não possui dívidas disponíveis."
                );
                return;
            }

            player.debtsAvailable--;

            console.log("==================================");
            console.log("JOGADOR USOU UMA DÍVIDA PARA SAIR DA PRISÃO");
            console.log("JOGADOR:", player.name);
            console.log(
                "DÍVIDAS RESTANTES:",
                player.debtsAvailable
            );
            console.log("==================================");

            this.releasePlayerFromPrison(client.sessionId);
        },

        // --------------------------------------------------
        // MOVIMENTAR JOGADOR
        // --------------------------------------------------

        move: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                client.send(
                    "action_error",
                    "Você precisa sair da prisão antes de movimentar."
                );
                return;
            }

            if (
                !this.state.hasRolled ||
                this.state.lastRollBy !== client.sessionId
            ) {
                client.send(
                    "action_error",
                    "Você precisa rolar os dados primeiro."
                );
                return;
            }

            if (this.state.hasMoved) {
                client.send(
                    "action_error",
                    "Você já movimentou nesta rodada."
                );
                return;
            }

            const dice1 = this.state.dice1;
            const dice2 = this.state.dice2;

            const rolledDoubles = dice1 === dice2;

            const steps = dice1 + dice2;
            const startingPosition = player.position;
            const newPosition = startingPosition + steps;

            const completedLaps = Math.floor(newPosition / BOARD_SIZE);
            const landedPosition = newPosition % BOARD_SIZE;

            player.position = landedPosition;
            player.laps += completedLaps;

            const landedCell = this.board[landedPosition];

            player.lastCellType = landedCell.type;
            player.lastCellName = landedCell.name;

            this.state.hasMoved = true;

            console.log(
                `${player.name} avançou ${steps} casas. Posição: ${player.position}`
            );

            console.log(
                `${player.name} caiu na casa ${player.position}: ${landedCell.name} (${landedCell.type})`
            );

            console.log(
                `Dados: ${dice1} + ${dice2} | Duplas: ${rolledDoubles}`
            );

            if (completedLaps > 0) {
                console.log(
                    `${player.name} completou ${completedLaps} volta(s). Total: ${player.laps}`
                );

                const totalMoneyReward =
                    START_CELL_MONEY_REWARD * completedLaps;

                player.money += totalMoneyReward;

                console.log(
                    `${player.name} recebeu $${totalMoneyReward.toLocaleString("pt-BR")} por completar ${completedLaps} volta(s).`
                );
            }

            // --------------------------------------------------
            // VERIFICAR SE CAIU NA PRISÃO
            // --------------------------------------------------

            const landedInPrison =
                landedCell.name === "Prisão" &&
                landedCell.type === "island";

            if (landedInPrison) {
                player.inPrison = true;
                player.prisonAttempts = 0;
                player.extraRollAvailable = false;

                this.state.hasRolled = false;
                this.state.hasMoved = true;
                this.state.lastRollBy = "";

                console.log(
                    `${player.name} foi enviado para a prisão. O turno será encerrado após o anúncio.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "prison"
                );

                return;
            }

            const landedExactlyOnStart =
                landedPosition === 0 &&
                landedCell.type === "start" &&
                landedCell.name === "Início";

            const receivesStartExtraRoll =
                landedExactlyOnStart &&
                completedLaps > 0 &&
                player.laps >= 1;

            let extraRollReason: ExtraRollReason | undefined;

            if (receivesStartExtraRoll) {
                extraRollReason = "start";
            }
            else if (rolledDoubles) {
                extraRollReason = "doubles";
            }

            const canReceiveExtraRoll =
                extraRollReason !== undefined;

            console.log(
                `[VERIFICAÇÃO ROLAGEM EXTRA] ${player.name} | ` +
                `posição inicial: ${startingPosition} | ` +
                `dados: ${dice1} + ${dice2} | ` +
                `duplas: ${rolledDoubles} | ` +
                `posição final: ${landedPosition} | ` +
                `voltas: ${player.laps} | ` +
                `terminou no início: ${receivesStartExtraRoll} | ` +
                `motivo: ${extraRollReason ?? "nenhum"} | ` +
                `recebe extra: ${canReceiveExtraRoll}`
            );

            if (canReceiveExtraRoll && extraRollReason) {
                player.extraRollAvailable = true;

                this.state.hasRolled = false;
                this.state.hasMoved = true;

                this.state.lastRollBy = "";

                console.log(
                    `${player.name} recebeu uma rolagem extra pelo motivo: ${extraRollReason}.`
                );

                this.queueMovementAnnouncement(
                    client.sessionId,
                    "extra",
                    extraRollReason
                );
            }
            else {
                // Mesmo sem popup, aguardar a animação terminar
                // antes de liberar as ações da HUD.
                this.queueMovementAnnouncement(
                    client.sessionId,
                    "none"
                );
            }
        },

        // --------------------------------------------------
        // ENCERRAR TURNO
        // --------------------------------------------------

        end_turn: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            if (!this.state.hasMoved) {
                client.send(
                    "action_error",
                    "Você precisa movimentar antes de encerrar a jogada."
                );
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                client.send(
                    "action_error",
                    "Você ainda está na prisão."
                );
                return;
            }

            if (player.extraRollAvailable) {
                client.send(
                    "action_error",
                    "Você precisa utilizar sua rolagem extra antes de encerrar o turno."
                );
                return;
            }

            console.log(
                `Jogador ${client.sessionId} encerrou sua jogada.`
            );

            this.nextTurn();
        },

        play: (client: Client) => {
            if (this.state.status !== "playing") {
                return;
            }

            if (this.isPopupActive()) {
                return;
            }

            if (this.state.currentTurn !== client.sessionId) {
                return;
            }

            const player = this.state.players.get(client.sessionId);

            if (!player) {
                return;
            }

            if (player.inPrison) {
                return;
            }

            player.score++;
            this.nextTurn();
        },
    };

    onAuth(client: Client, options: any) {

        if (this.state.status !== "waiting") {
            throw new ServerError(409, "Esta partida já começou.");
        }

        if (options?.mode !== this.mode) {
            throw new ServerError(
                403,
                "Esta sala não pertence a essa modalidade de partida."
            );
        }

        // --------------------------------------------------
        // VALIDAR NOME STEAM
        // --------------------------------------------------

        if (
            typeof options?.playerName !== "string"
        ) {
            throw new ServerError(400, "Nome Steam inválido.");
        }

        const playerName = String(options.playerName).trim();

        if (playerName.length === 0 || playerName.length > 32) {
            throw new ServerError(
                400,
                "O nome Steam deve possuir entre 1 e 32 caracteres."
            );
        }

        // --------------------------------------------------
        // VALIDAR STEAM ID
        // --------------------------------------------------

        const steamId = String(options?.steamId ?? "").trim();

        if (!/^\d{17}$/.test(steamId)) {
            throw new ServerError(
                400,
                "Steam ID inválido ou não recebido."
            );
        }

        // --------------------------------------------------
        // VALIDAR PERSONAGEM
        // --------------------------------------------------

        const selectedCharacter = Number(
            options?.selectedCharacter ?? 0
        );

        const validCharacter =
            Number.isInteger(selectedCharacter) &&
            selectedCharacter >= 0 &&
            selectedCharacter <= 3;

        if (!validCharacter) {
            throw new ServerError(400, "Personagem inválido.");
        }

        // --------------------------------------------------
        // VALIDAR COR
        // --------------------------------------------------

        const preferredColor =
            typeof options?.selectedColor === "string" &&
            this.isValidColor(options.selectedColor)
                ? options.selectedColor
                : "red";

        return {
            playerName,
            steamId,
            preferredColor,
            selectedCharacter
        };
    }

    onJoin(client: Client, options: any, auth: any) {

        const player = new Player();

        player.lastCellName = this.board[0].name;

        player.color = this.assignPlayerColor(
            auth.preferredColor ?? "red"
        );

        this.state.players.set(client.sessionId, player);

        this.playerJoinOrder.push(client.sessionId);

        if (this.state.hostSessionId === "") {
            this.state.hostSessionId = client.sessionId;
        }

        console.log("==================================");
        console.log(`${player.name} entrou na sala ${this.roomId}!`);
        console.log("STEAM ID:", player.steamId);
        console.log("SESSION ID:", client.sessionId);

        console.log("Personagem escolhido:", player.character);
        console.log("Dinheiro inicial:", player.money);
        console.log("Dívidas disponíveis:", player.debtsAvailable);

        console.log(
            "Ordem de entrada:",
            this.playerJoinOrder.join(" -> ")
        );

        console.log("==================================");

        this.resetAutoStartCountdown();
    }

    onLeave(client: Client, code: CloseCode) {
        console.log(client.sessionId, "saiu da sala!", code);

        const wasTheirTurn =
            this.state.currentTurn === client.sessionId;

        const wasHost =
            this.state.hostSessionId === client.sessionId;

        const wasSelectedFirstPlayer =
            this.state.firstPlayerSessionId === client.sessionId;

        const wasExtraRollPlayer =
            this.state.extraRollPlayerSessionId === client.sessionId;

        const wasPrisonMessagePlayer =
            this.state.prisonPlayerSessionId === client.sessionId;

        this.state.players.delete(client.sessionId);

        this.playerJoinOrder = this.playerJoinOrder.filter(
            sessionId => sessionId !== client.sessionId
        );

        // Atualizar a espera de chegada caso alguém saia da sala.
        if (this.pendingMovementAnnouncement) {
            if (
                this.pendingMovementAnnouncement.sessionId ===
                client.sessionId
            ) {
                this.pendingMovementAnnouncement = undefined;
                this.state.movementAnnouncementPending = false;
            }
            else {
                this.pendingMovementAnnouncement.arrivedClients.delete(
                    client.sessionId
                );

                this.finishMovementAnnouncementIfReady(
                    client.sessionId
                );
            }
        }

        if (
            this.state.showFirstPlayerMessage &&
            wasSelectedFirstPlayer
        ) {
            const newFirstPlayer = this.chooseRandomFirstPlayer();

            if (newFirstPlayer !== "") {
                this.updateFirstPlayerAnnouncement(newFirstPlayer);
            } else {
                this.state.firstPlayerSessionId = "";
                this.state.firstPlayerName = "";
            }
        }

        if (
            this.state.showExtraRollMessage &&
            wasExtraRollPlayer
        ) {
            this.extraRollMessageDelay?.clear();
            this.extraRollMessageDelay = undefined;

            this.state.showExtraRollMessage = false;
            this.state.extraRollPlayerName = "";
            this.state.extraRollPlayerSessionId = "";
            this.state.extraRollReason = "";

            if (this.state.status === "playing" && wasTheirTurn) {
                this.nextTurn();
            }
        }

        if (
            this.state.showPrisonMessage &&
            wasPrisonMessagePlayer
        ) {
            this.prisonMessageDelay?.clear();
            this.prisonMessageDelay = undefined;

            this.state.showPrisonMessage = false;
            this.state.prisonPlayerName = "";
            this.state.prisonPlayerSessionId = "";
        }

        if (
            this.state.status === "waiting" &&
            wasHost
        ) {
            const nextHost = this.playerJoinOrder.find(
                sessionId => this.state.players.has(sessionId)
            );

            this.state.hostSessionId = nextHost ?? "";

            if (nextHost) {
                const nextHostPlayer = this.state.players.get(nextHost);

                console.log("==================================");
