/// obj_menu - Create

enum telas {
    menu,
    menu_partida_privada,
    lobby,
    menu_partida_publica
}

enum personagem {
    boneco1,
    boneco2,
    boneco3,
    boneco4
}

tela_atual = telas.menu;

botoes_menu = ["Partida Privada", "Partida Publica"];
botoes_funcao = [
    telas.menu_partida_privada,
    telas.menu_partida_publica
];

// --------------------------------------------------
// PERSONAGENS
// --------------------------------------------------

// Lista de personagens disponíveis
personagens = [
    personagem.boneco1,
    personagem.boneco2,
    personagem.boneco3,
    personagem.boneco4
];

// Sprites correspondentes aos personagens
sprites_personagens = [
    spr_player1,
    spr_player2,
    spr_player3,
    spr_player4
];

// Personagem selecionado
personagem_indice = 0;
personagem_atual = personagens[personagem_indice];

// --------------------------------------------------
// CORES DOS JOGADORES
// --------------------------------------------------

cores_jogador = [
    "red",
    "blue",
    "green",
    "yellow"
];

cores_draw = [
    c_red,
    c_blue,
    c_green,
    c_yellow
];

// Índice da cor selecionada
cor_indice = 0;
cor_atual = cores_jogador[cor_indice];

// --------------------------------------------------
// NOME DO JOGADOR
// --------------------------------------------------

randomize();

nome_jogador = choose(
    "prikito",
    "caneta azul",
    "inxada",
    "tira o ja"
);

// --------------------------------------------------
// CÓDIGO DA SALA
// --------------------------------------------------

codigo_sala_privada = "";
code = "";
ativo_code = false;

// --------------------------------------------------
// MENSAGENS DA INTERFACE
// --------------------------------------------------

menu_mensagem = "";
menu_erro = "";

// --------------------------------------------------
// CONTROLE DE CONEXÃO
// --------------------------------------------------

menu_aguardando = false;
menu_conexao_anterior = false;

// --------------------------------------------------
// LISTAGEM PÚBLICA
// --------------------------------------------------

partidas_existentes = 0;
partida_selecionada = -1;

// --------------------------------------------------
// LOBBY
// --------------------------------------------------

qtd_players = 0;

obj_menu step:"
/// obj_menu - Step

var _net = instance_find(obj_network_manager, 0);

// --------------------------------------------------
// VERIFICAR SE A PARTIDA COMEÇOU
// --------------------------------------------------

if (_net != noone && _net.net_game_started) {

    _net.net_game_started = false;

    room_goto(rm_jogo);
    exit;
}

// --------------------------------------------------
// VERIFICAR CONEXÃO
// --------------------------------------------------

if (_net != noone) {

    if (_net.net_connected && !menu_conexao_anterior) {

        menu_conexao_anterior = true;
        menu_aguardando = false;

        tela_atual = telas.lobby;

        ativo_code = false;
        menu_erro = "";
        menu_mensagem = "";

        codigo_sala_privada = _net.net_room_id;

        cor_atual = _net.net_player_color;

        for (var i = 0; i < array_length(cores_jogador); i++) {

            if (cores_jogador[i] == cor_atual) {
                cor_indice = i;
                break;
            }
        }
    }

    if (_net.net_error != "") {

        menu_erro = _net.net_error;
        menu_aguardando = false;
    }

    partidas_existentes = array_length(_net.net_lobby_rooms);

    if (_net.net_connected) {
        qtd_players = array_length(_net.net_players);
    }
}

// --------------------------------------------------
// DIGITAÇÃO DOS CÓDIGOS
// --------------------------------------------------

if (ativo_code) {

    var _allowed = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    var _filtered_code = "";

    for (var i = 1; i <= string_length(keyboard_string); i++) {

        var _char = string_char_at(keyboard_string, i);

        if (string_pos(_char, _allowed) > 0) {

            if (string_length(_filtered_code) < 6) {
                _filtered_code += _char;
            }
        }
    }

    code = _filtered_code;
    keyboard_string = code;

    if (keyboard_check_pressed(vk_enter)) {

        ativo_code = false;
        code = string_trim(code);

        if (string_length(code) != 6) {

            menu_erro = "O código deve ter exatamente 6 caracteres.";
            menu_mensagem = "";

        }
        else if (_net == noone) {

            menu_erro = "Gerenciador de rede não encontrado.";

        }
        else {

            menu_erro = "";
            menu_mensagem = "Verificando sala...";
            menu_aguardando = true;

            if (tela_atual == telas.menu_partida_privada) {

                _net.net_join_private(
                    code,
                    nome_jogador,
                    cor_atual,
                    personagem_indice
                );

            }
            else if (tela_atual == telas.menu_partida_publica) {

                _net.net_join_public(
                    code,
                    nome_jogador,
                    cor_atual,
                    personagem_indice
                );

            }
        }
    }

    if (keyboard_check_pressed(vk_escape)) {

        ativo_code = false;
        code = "";
        keyboard_string = "";
    }
}

// --------------------------------------------------
// CLIQUES DO MENU
// --------------------------------------------------

if (mouse_check_button_pressed(mb_left)) {

    var _room_x = room_width;
    var _room_y = room_height;

    if (menu_aguardando) {
        exit;
    }

    switch (tela_atual) {

        // ==========================================
        // MENU PRINCIPAL
        // ==========================================

        case telas.menu:

            // Gerar nome aleatório.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                20, 20, 300, 70
            )) {

                randomize();

                nome_jogador = choose(
                    "prikito",
                    "neymar",
                    "inxada",
                    "tira o ja"
                );
            }

            // Trocar personagem.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                20, 80, 300, 130
            )) {

                personagem_indice++;

                if (personagem_indice >= array_length(personagens)) {
                    personagem_indice = 0;
                }

                personagem_atual = personagens[personagem_indice];
            }

            // Selecionar cor.
            for (var i = 0; i < array_length(cores_jogador); i++) {

                var _color_left = 20 + i * 60;
                var _color_right = _color_left + 50;

                if (point_in_rectangle(
                    mouse_x, mouse_y,
                    _color_left, 145,
                    _color_right, 195
                )) {

                    cor_indice = i;
                    cor_atual = cores_jogador[i];

                    break;
                }
            }

            var _b1 = (_room_y / 2) - 100;
            var _b2 = (_room_y / 2) - 25;
            var _b3 = (_room_x / 2) - 450;

            for (var i = 0; i < 2; i++) {

                var _left = _b3 + i * 300;
                var _right = _left + 250;

                if (point_in_rectangle(
                    mouse_x, mouse_y,
                    _left, _b1,
                    _right, _b2
                )) {

                    tela_atual = botoes_funcao[i];

                    ativo_code = false;
                    code = "";
                    keyboard_string = "";

                    menu_erro = "";
                    menu_mensagem = "";
                }
            }

            break;

        // ==========================================
        // MENU PRIVADO
        // ==========================================

        case telas.menu_partida_privada:

            // Criar partida privada.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                _room_x / 2 - 200,
                _room_y / 2 - 240,
                _room_x / 2 + 200,
                _room_y / 2 - 130
            )) {

                if (_net != noone) {

                    menu_erro = "";
                    menu_mensagem = "Criando partida...";
                    menu_aguardando = true;

                    _net.net_create_private(
                        nome_jogador,
                        cor_atual,
                        personagem_indice
                    );
                }
            }

            // Ativar campo de código privado.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                _room_x / 2 - 200,
                _room_y / 2 - 10,
                _room_x / 2 + 200,
                _room_y / 2 + 100
            )) {

                ativo_code = true;

                keyboard_string = code;
                menu_erro = "";
                menu_mensagem = "";
            }

            // Voltar.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                20, 670, 300, 740
            )) {

                ativo_code = false;
                code = "";
                keyboard_string = "";

                tela_atual = telas.menu;

                menu_erro = "";
                menu_mensagem = "";
            }

            break;

        // ==========================================
        // MENU PÚBLICO
        // ==========================================

        case telas.menu_partida_publica:

            // Voltar.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                20, 670, 300, 740
            )) {

                ativo_code = false;
                code = "";
                keyboard_string = "";

                tela_atual = telas.menu;

                menu_erro = "";
                menu_mensagem = "";

                        nome_jogador,
                        cor_atual,
                        personagem_indice
                    );
                }

                break;
            }

            // Entrar em uma partida pública da lista.
            if (_net != noone) {

                var _visible_index = 0;

                for (var i = 0; i < array_length(_net.net_lobby_rooms); i++) {

                    var _room_info = _net.net_lobby_rooms[i];

                    var _row_top = _room_y / 2 - 90 + (_visible_index * 100);
                    var _row_bottom = _row_top + 74;

                    if (_visible_index < 4) {

                        if (point_in_rectangle(
                            mouse_x, mouse_y,
                            _room_x / 2 + 210,
                            _row_top,
                            _room_x - 40,
                            _row_bottom
                        )) {

                            menu_erro = "";
                            menu_mensagem = "Entrando na partida...";
                            menu_aguardando = true;

                            _net.net_join_public(
                                _room_info.roomId,
                                nome_jogador,
                                cor_atual,
                                personagem_indice
                            );

                            break;
                        }
                    }

                    _visible_index++;
                }
            }

            break;

        // ==========================================
        // LOBBY
        // ==========================================

        case telas.lobby:

            // Selecionar uma cor no lobby.
            if (_net != noone
            && _net.net_connected
            && _net.net_game_status == "waiting") {

                for (var i = 0; i < array_length(cores_jogador); i++) {

                    var _color_left = 20 + i * 60;
                    var _color_right = _color_left + 50;

                    if (point_in_rectangle(
                        mouse_x, mouse_y,
                        _color_left, 150,
                        _color_right, 200
                    )) {

                        var _requested_color = cores_jogador[i];
                        var _color_occupied = false;

                        for (var j = 0; j < array_length(_net.net_players); j++) {

                            var _other_player = _net.net_players[j];

                            if (_other_player.session_id != _net.net_session_id) {

                                var _other_color = colyseus_schema_get(
                                    _other_player.schema,
                                    "color"
                                );

                                if (_other_color == _requested_color) {
                                    _color_occupied = true;
                                    break;
                                }
                            }
                        }

                        if (!_color_occupied) {

                            colyseus_send(
                                _net.net_room,
                                "select_color",
                                {
                                    color: _requested_color
                                }
                            );

                            menu_erro = "";
                        }

                        break;
                    }
                }
            }

            // Sair do lobby.
            if (point_in_rectangle(
                mouse_x, mouse_y,
                20, 670, 300, 740
            )) {

                if (_net != noone && _net.net_connected) {

                    if (_net.net_room != 0) {

                        if (colyseus_room_is_connected(_net.net_room)) {
                            colyseus_room_leave(_net.net_room);
                        }

                        colyseus_room_free(_net.net_room);
                    }

                    _net.net_room = 0;
                    _net.net_callbacks = 0;

                    _net.net_connected = false;
                    _net.net_busy = false;

                    _net.net_players = [];

                    _net.net_room_id = "";
                    _net.net_session_id = "";
                    _net.net_host_session_id = "";

                    _net.net_game_status = "waiting";
                    _net.net_game_started = false;

                    _net.net_room_mode = "";
                    _net.net_auto_start_countdown = 0;

                    _net.net_player_color = "red";
                }

                tela_atual = telas.menu;

                menu_conexao_anterior = false;
                menu_aguardando = false;

                ativo_code = false;
                code = "";
                keyboard_string = "";

                codigo_sala_privada = "";

                qtd_players = 0;

                menu_erro = "";
                menu_mensagem = "";

                break;
            }

            // Iniciar partida privada: somente o anfitrião.
            if (_net != noone
            && _net.net_connected
            && _net.net_room_mode == "private"
            && _net.net_session_id == _net.net_host_session_id
            && array_length(_net.net_players) >= 2) {

                if (point_in_rectangle(
                    mouse_x, mouse_y,
                    _room_x / 2 - 100,
                    670,
                    _room_x / 2 + 100,
                    740
                )) {

                    colyseus_send(
                        _net.net_room,
                        "start",
                        {}
                    );

                    menu_mensagem = "Iniciando partida...";
                }
            }

            break;
    }
}

                for (var j = 0; j < 4; j++) {

                    var _has_player = j < _room_info.clients;

                    draw_sprite_ext(
                        spr_players_icon,
                        _has_player,
                        _room_x - 150 + j * 25,
                        _row_top + 25,
                        2, 2, 0,
                        c_white, 1
                    );
                }
            }

            if (_room_count == 0) {

                draw_set_halign(fa_center);

                draw_text_transformed(
                    _room_x / 2 + 400,
                    _room_y / 2 + 50,
                    "Nenhuma partida encontrada",
                    1, 1, 0
                );

                draw_set_halign(fa_left);
            }
        }

        break;


    // --------------------------------------------------
    // LOBBY
    // --------------------------------------------------

    case telas.lobby:

        var _net = instance_find(obj_network_manager, 0);

        draw_set_halign(fa_center);

        draw_text_transformed(
            _room_x / 2,
            _room_y / 2 - 150,
            "Lobby",
            2, 2, 0
        );

        draw_text_transformed(
            _room_x / 2,
            _room_y / 2 - 100,
            "Codigo da sala: " + codigo_sala_privada,
            1, 1, 0
        );

        draw_text_transformed(
            _room_x / 2,
            _room_y / 2 - 60,
            "Jogadores: " + string(qtd_players) + " / 4",
            1, 1, 0
        );

        draw_set_halign(fa_left);

        // ------------------------------------------
        // SELEÇÃO DE CORES DO LOBBY
        // ------------------------------------------

        draw_text_transformed(
            20,
            125,
            "Escolha sua cor:",
            1, 1, 0
        );

        for (var i = 0; i < array_length(cores_jogador); i++) {

            var _color_left = 20 + i * 60;
            var _color_top = 150;
            var _color_right = _color_left + 50;
            var _color_bottom = _color_top + 50;

            var _color_occupied = false;

            // Verificar se algum jogador está usando essa cor
            if (_net != noone) {

                for (var j = 0; j < array_length(_net.net_players); j++) {

                    var _player_color = colyseus_schema_get(
                        _net.net_players[j].schema,
                        "color"
                    );

                    if (_player_color == cores_jogador[i]) {
                        _color_occupied = true;
                        break;
                    }
                }
            }

            // Desenhar quadrado colorido
            draw_set_alpha(1);
            draw_set_color(cores_draw[i]);

            draw_rectangle(
                _color_left,
                _color_top,
                _color_right,
                _color_bottom,
                false
            );

            // Escurecer cores ocupadas
            if (_color_occupied) {

                draw_set_color(c_black);
                draw_set_alpha(0.8);

                draw_rectangle(
                    _color_left,
                    _color_top,
                    _color_right,
                    _color_bottom,
                    false
                );
            }

            draw_set_alpha(1);
        }

        draw_set_color(c_white);

        // ------------------------------------------
        // CAIXAS DOS JOGADORES
        // ------------------------------------------

        var _player_x = -250;

        for (var i = 0; i < 4; i++) {

            var _box_left = _room_x / 2 + _player_x + i * 140;
            var _box_right = _box_left + 100;

            draw_rectangle(
                _box_left,
                _room_y / 2,
                _box_right,
                _room_y / 2 + 200,
                true
            );

            var _player_name = "Aguardando...";

            var _has_player = false;
            var _box_player_color = "";

            if (_net != noone) {

                if (i < array_length(_net.net_players)) {

                    _has_player = true;

                    _player_name = _net.net_players[i].name;

                    _box_player_color = colyseus_schema_get(
                        _net.net_players[i].schema,
                        "color"
                    );
                }
            }

            // Desenhar a faixa somente se existir jogador
            if (_has_player) {

                var _box_color = c_red;

                switch (_box_player_color) {

                    case "red":
                        _box_color = c_red;
                        break;

                    case "blue":
                        _box_color = c_blue;
                        break;

                    case "green":
                        _box_color = c_green;
                        break;

                    case "yellow":
                        _box_color = c_yellow;
                        break;
                }

                draw_set_alpha(1);
                draw_set_color(_box_color);

                draw_rectangle(
                    _box_left,
                    _room_y / 2,
                    _box_right,
                    _room_y / 2 + 12,
                    false
                );
            }

            draw_set_alpha(1);
            draw_set_color(c_white);

            draw_set_halign(fa_center);

            draw_text_transformed(
                _box_left + 50,
                _room_y / 2 + 75,
                _player_name,
                1, 1, 0
            );

            draw_set_halign(fa_left);
        }

        // ------------------------------------------
        // CONTAGEM AUTOMÁTICA DAS PARTIDAS PÚBLICAS
        // ------------------------------------------

        if (_net != noone
        && _net.net_connected
        && _net.net_room_mode == "public") {

            draw_set_halign(fa_center);

            if (_net.net_auto_start_countdown > 0) {

                draw_text_transformed(
                    _room_x / 2,
                    620,
                    "Partida iniciando em: " +
                    string(_net.net_auto_start_countdown) + "s",
                    1.5, 1.5, 0
                );

            }
            else if (array_length(_net.net_players) < 2) {

                draw_text_transformed(
                    _room_x / 2,
                    620,
                    "Aguardando mais jogadores...",
                    1.5, 1.5, 0
                );
            }

            draw_set_halign(fa_left);
        }

        // ------------------------------------------
        // BOTÃO SAIR DO LOBBY
        // ------------------------------------------

        if (point_in_rectangle(
            mouse_x, mouse_y,
            20, 670, 300, 740
        )) {
            draw_set_color(c_lime);
        }

        draw_rectangle(20, 670, 300, 740, true);

        draw_set_halign(fa_center);

        draw_text_transformed(
            160,
            690,
            "Sair do lobby",
            1, 1, 0
        );

        draw_set_halign(fa_left);
        draw_set_color(c_white);

        // ------------------------------------------
        // BOTÃO INICIAR PARTIDA
        // SOMENTE PARA PARTIDAS PRIVADAS
        // ------------------------------------------

        if (_net != noone
        && _net.net_connected
        && _net.net_room_mode == "private"
        && _net.net_session_id == _net.net_host_session_id
        && array_length(_net.net_players) >= 2) {

            if (point_in_rectangle(
                mouse_x, mouse_y,
                _room_x / 2 - 100,
                670,
                _room_x / 2 + 100,
                740
            )) {
                draw_set_color(c_lime);
            }

            draw_rectangle(
                _room_x / 2 - 100,
                670,
                _room_x / 2 + 100,
                740,
                true
            );

            draw_set_halign(fa_center);

            draw_text_transformed(
                _room_x / 2,
                690,
                "Iniciar partida",
                1, 1, 0
            );

            draw_set_halign(fa_left);
            draw_set_color(c_white);
        }

        break;
}
